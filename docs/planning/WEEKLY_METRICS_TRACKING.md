## 📊 DASHBOARD COMMANDS

### **Quick Status Check**
```bash
#!/bin/bash
# weekly_status.sh

echo "=== BEARDOG WEEKLY STATUS ==="
echo ""

# Coverage
echo "Test Coverage:"
cargo tarpaulin --workspace --out Json 2>/dev/null | \
  grep -o '"coverage":[0-9.]*' | \
  cut -d: -f2

# Unwraps
echo ""
echo "Production Unwraps:"
grep -r "\.unwrap()\|\.expect(" crates/ \
  --include="*.rs" \
  --exclude-dir="tests" | \
  grep -v "test" | wc -l

# Hardcoding
echo ""
echo "Hardcoded IPs/Ports:"
grep -rE "(localhost|127\.0\.0\.1|:8080|:8081)" crates/ \
  --include="*.rs" \
  --exclude-dir="tests" | wc -l

# Tests
echo ""
echo "Tests Passing:"
cargo test --workspace 2>&1 | \
  grep "test result" | tail -1

# Clippy
echo ""
echo "Clippy Warnings:"
cargo clippy --workspace --all-targets 2>&1 | \
  grep "warning:" | wc -l
```
