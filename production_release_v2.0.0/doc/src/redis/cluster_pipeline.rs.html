18        // All commands that start with "CONFIG"

20        "DBSIZE" |
21        "ECHO" | "EVALSHA" |
22        "FLUSHALL" | "FLUSHDB" |

24        "KEYS" |

44/// Represents a Redis Cluster command pipeline.
45#[derive(Clone)]
46pub struct ClusterPipeline {
47    commands: Vec<Cmd>,
48    ignored_commands: HashSet<usize>,
49}
50
51/// A cluster pipeline is almost identical to a normal [Pipeline](crate::pipeline::Pipeline), with two exceptions:
52/// * It does not support transactions
53/// * The following commands can not be used in a cluster pipeline:
54/// ```text
55/// BGREWRITEAOF, BGSAVE, BITOP, BRPOPLPUSH
56/// CLIENT GETNAME, CLIENT KILL, CLIENT LIST, CLIENT SETNAME, CONFIG GET,
57/// CONFIG RESETSTAT, CONFIG REWRITE, CONFIG SET
58/// DBSIZE
59/// ECHO, EVALSHA
60/// FLUSHALL, FLUSHDB
61/// INFO
62/// KEYS
63/// LASTSAVE
64/// MGET, MOVE, MSET, MSETNX
65/// PFMERGE, PFCOUNT, PING, PUBLISH
66/// RANDOMKEY, RENAME, RENAMENX, RPOPLPUSH
67/// SAVE, SCAN, SCRIPT EXISTS, SCRIPT FLUSH, SCRIPT KILL, SCRIPT LOAD, SDIFF, SDIFFSTORE,
68/// SENTINEL GET MASTER ADDR BY NAME, SENTINEL MASTER, SENTINEL MASTERS, SENTINEL MONITOR,
69/// SENTINEL REMOVE, SENTINEL SENTINELS, SENTINEL SET, SENTINEL SLAVES, SHUTDOWN, SINTER,
70/// SINTERSTORE, SLAVEOF, SLOWLOG GET, SLOWLOG LEN, SLOWLOG RESET, SMOVE, SORT, SUNION, SUNIONSTORE
71/// TIME
72/// ```
73impl ClusterPipeline {
74    /// Create an empty pipeline.
75    pub fn new() -> ClusterPipeline {
76        Self::with_capacity(0)
77    }
78
79    /// Creates an empty pipeline with pre-allocated capacity.
80    pub fn with_capacity(capacity: usize) -> ClusterPipeline {
81        ClusterPipeline {
82            commands: Vec::with_capacity(capacity),
83            ignored_commands: HashSet::new(),
84        }
85    }
86
87    pub(crate) fn commands(&self) -> &Vec<Cmd> {
88        &self.commands
89    }
90
91    /// Executes the pipeline and fetches the return values:
92    ///
93    /// ```rust,no_run
94    /// # let nodes = vec!["redis://127.0.0.1:6379/"];
95    /// # let client = redis::cluster::ClusterClient::new(nodes).unwrap();
96    /// # let mut con = client.get_connection().unwrap();
97    /// let mut pipe = redis::cluster::cluster_pipe();
98    /// let (k1, k2) : (i32, i32) = pipe
99    ///     .cmd("SET").arg("key_1").arg(42).ignore()
100    ///     .cmd("SET").arg("key_2").arg(43).ignore()
101    ///     .cmd("GET").arg("key_1")
102    ///     .cmd("GET").arg("key_2").query(&mut con).unwrap();
103    /// ```
104    #[inline]
105    pub fn query<T: FromRedisValue>(&self, con: &mut ClusterConnection) -> RedisResult<T> {
106        for cmd in &self.commands {
107            let cmd_name = std::str::from_utf8(cmd.arg_idx(0).unwrap_or(b""))
108                .unwrap_or("")
109                .trim()
110                .to_ascii_uppercase();
111
112            if is_illegal_cmd(&cmd_name) {
113                fail!((
114                    UNROUTABLE_ERROR.0,
115                    UNROUTABLE_ERROR.1,
116                    format!("Command '{cmd_name}' can't be executed in a cluster pipeline.")
117                ))
118            }
119        }
120
121        from_redis_value(
122            &(if self.commands.is_empty() {
123                Value::Bulk(vec![])
124            } else {
125                self.make_pipeline_results(con.execute_pipeline(self)?)
126            }),
127        )
128    }
129
130    /// This is a shortcut to `query()` that does not return a value and
131    /// will fail the task if the query of the pipeline fails.
132    ///
133    /// This is equivalent to a call to query like this:
134    ///
135    /// ```rust,no_run
136    /// # let nodes = vec!["redis://127.0.0.1:6379/"];
137    /// # let client = redis::cluster::ClusterClient::new(nodes).unwrap();
138    /// # let mut con = client.get_connection().unwrap();
139    /// let mut pipe = redis::cluster::cluster_pipe();
140    /// let _ : () = pipe.cmd("SET").arg("key_1").arg(42).ignore().query(&mut con).unwrap();
141    /// ```
142    #[inline]
143    pub fn execute(&self, con: &mut ClusterConnection) {
144        self.query::<()>(con).unwrap();
145    }
146}
147
148/// Shortcut for creating a new cluster pipeline.
149pub fn cluster_pipe() -> ClusterPipeline {
150    ClusterPipeline::new()
151}
