1//! This module extends the library to support Redis Cluster.
2//!
3//! Note that this module does not currently provide pubsub
4//! functionality.
5//!
6//! # Example
7//! ```rust,no_run
8//! use redis::Commands;
9//! use redis::cluster::ClusterClient;
10//!
11//! let nodes = vec!["redis://127.0.0.1:6379/", "redis://127.0.0.1:6378/", "redis://127.0.0.1:6377/"];
12//! let client = ClusterClient::new(nodes).unwrap();
13//! let mut connection = client.get_connection().unwrap();
14//!
15//! let _: () = connection.set("test", "test_data").unwrap();
16//! let rv: String = connection.get("test").unwrap();
17//!
18//! assert_eq!(rv, "test_data");
19//! ```
20//!
21//! # Pipelining
22//! ```rust,no_run
23//! use redis::Commands;
24//! use redis::cluster::{cluster_pipe, ClusterClient};
25//!
26//! let nodes = vec!["redis://127.0.0.1:6379/", "redis://127.0.0.1:6378/", "redis://127.0.0.1:6377/"];
27//! let client = ClusterClient::new(nodes).unwrap();
28//! let mut connection = client.get_connection().unwrap();
29//!
30//! let key = "test";
31//!
32//! let _: () = cluster_pipe()
33//!     .rpush(key, "123").ignore()
34//!     .ltrim(key, -10, -1).ignore()
35//!     .expire(key, 60).ignore()
36//!     .query(&mut connection).unwrap();
37//! ```

72#[derive(Clone)]
73enum Input<'a> {
74    Slice {
75        cmd: &'a [u8],
76        routable: Value,
77    },
78    Cmd(&'a Cmd),
79    Commands {
80        cmd: &'a [u8],
81        route: SingleNodeRoutingInfo,
82        offset: usize,
83        count: usize,
84    },
85}
86
87impl<'a> Input<'a> {
88    fn send(&'a self, connection: &mut impl ConnectionLike) -> RedisResult<Output> {
89        match self {
90            Input::Slice { cmd, routable: _ } => {
91                connection.req_packed_command(cmd).map(Output::Single)
92            }
93            Input::Cmd(cmd) => connection.req_command(cmd).map(Output::Single),
94            Input::Commands {
95                cmd,
96                route: _,
97                offset,
98                count,
99            } => connection
100                .req_packed_commands(cmd, *offset, *count)
101                .map(Output::Multi),
102        }
103    }
104}
105
106impl<'a> Routable for Input<'a> {
107    fn arg_idx(&self, idx: usize) -> Option<&[u8]> {
108        match self {
109            Input::Slice { cmd: _, routable } => routable.arg_idx(idx),
110            Input::Cmd(cmd) => cmd.arg_idx(idx),
111            Input::Commands { .. } => None,
112        }
113    }
114
115    fn position(&self, candidate: &[u8]) -> Option<usize> {
116        match self {
117            Input::Slice { cmd: _, routable } => routable.position(candidate),
118            Input::Cmd(cmd) => cmd.position(candidate),
119            Input::Commands { .. } => None,
120        }
121    }
122}

124enum Output {
125    Single(Value),
126    Multi(Vec<Value>),
127}

129impl From<Output> for Value {
130    fn from(value: Output) -> Self {
131        match value {
132            Output::Single(value) => value,
133            Output::Multi(values) => Value::Bulk(values),
134        }
135    }
136}

138impl From<Output> for Vec<Value> {
139    fn from(value: Output) -> Self {
140        match value {
141            Output::Single(value) => vec![value],
142            Output::Multi(values) => values,
143        }
144    }

147/// Implements the process of connecting to a Redis server
148/// and obtaining and configuring a connection handle.
149pub trait Connect: Sized {
150    /// Connect to a node, returning handle for command execution.
151    fn connect<T>(info: T, timeout: Option<Duration>) -> RedisResult<Self>
152    where
153        T: IntoConnectionInfo;
154
155    /// Sends an already encoded (packed) command into the TCP socket and
156    /// does not read a response.  This is useful for commands like
157    /// `MONITOR` which yield multiple items.  This needs to be used with
158    /// care because it changes the state of the connection.
159    fn send_packed_command(&mut self, cmd: &[u8]) -> RedisResult<()>;
160
161    /// Sets the write timeout for the connection.
162    ///
163    /// If the provided value is `None`, then `send_packed_command` call will
164    /// block indefinitely. It is an error to pass the zero `Duration` to this
165    /// method.
166    fn set_write_timeout(&self, dur: Option<Duration>) -> RedisResult<()>;
167
168    /// Sets the read timeout for the connection.
169    ///
170    /// If the provided value is `None`, then `recv_response` call will
171    /// block indefinitely. It is an error to pass the zero `Duration` to this
172    /// method.
173    fn set_read_timeout(&self, dur: Option<Duration>) -> RedisResult<()>;
174
175    /// Fetches a single response from the connection.  This is useful
176    /// if used in combination with `send_packed_command`.
177    fn recv_response(&mut self) -> RedisResult<Value>;
178}
179
180impl Connect for Connection {
181    fn connect<T>(info: T, timeout: Option<Duration>) -> RedisResult<Self>
182    where
183        T: IntoConnectionInfo,
184    {
185        connect(&info.into_connection_info()?, timeout)
186    }
187
188    fn send_packed_command(&mut self, cmd: &[u8]) -> RedisResult<()> {
189        Self::send_packed_command(self, cmd)
190    }
191
192    fn set_write_timeout(&self, dur: Option<Duration>) -> RedisResult<()> {
193        Self::set_write_timeout(self, dur)
194    }
195
196    fn set_read_timeout(&self, dur: Option<Duration>) -> RedisResult<()> {
197        Self::set_read_timeout(self, dur)
198    }
199
200    fn recv_response(&mut self) -> RedisResult<Value> {
201        Self::recv_response(self)
202    }
203}

205/// This represents a Redis Cluster connection. It stores the
206/// underlying connections maintained for each node in the cluster, as well
207/// as common parameters for connecting to nodes and executing commands.
208pub struct ClusterConnection<C = Connection> {
209    initial_nodes: Vec<ConnectionInfo>,

247    /// Sets the write timeout for the connection.
248    ///
249    /// If the provided value is `None`, then `send_packed_command` call will
250    /// block indefinitely. It is an error to pass the zero `Duration` to this
251    /// method.
252    pub fn set_write_timeout(&self, dur: Option<Duration>) -> RedisResult<()> {
253        // Check if duration is valid before updating local value.
254        if dur.is_some() && dur.unwrap().is_zero() {
255            return Err(RedisError::from((
256                ErrorKind::InvalidClientConfig,
257                "Duration should be None or non-zero.",
258            )));
259        }
260
261        let mut t = self.write_timeout.borrow_mut();
262        *t = dur;
263        let connections = self.connections.borrow();
264        for conn in connections.values() {
265            conn.set_write_timeout(dur)?;
266        }
267        Ok(())
268    }

272    /// If the provided value is `None`, then `recv_response` call will
273    /// block indefinitely. It is an error to pass the zero `Duration` to this
274    /// method.
275    pub fn set_read_timeout(&self, dur: Option<Duration>) -> RedisResult<()> {

303    /// Returns the connection status.
304    ///
305    /// The connection is open until any `read_response` call recieved an
306    /// invalid response from the server (most likely a closed or dropped
307    /// connection, otherwise a Redis protocol error). When using unix
308    /// sockets the connection is open until writing a command failed with a
309    /// `BrokenPipe` error.
310    fn create_initial_connections(&self) -> RedisResult<()> {
311        let mut connections = HashMap::with_capacity(self.initial_nodes.len());
312
313        for info in self.initial_nodes.iter() {
314            let addr = info.addr.to_string();
315
316            if let Ok(mut conn) = self.connect(&addr) {
317                if conn.check_connection() {
318                    connections.insert(addr, conn);
319                    break;
320                }
321            }
322        }
323
324        if connections.is_empty() {
325            return Err(RedisError::from((
326                ErrorKind::IoError,
327                "It failed to check startup nodes.",
328            )));
329        }
330
331        *self.connections.borrow_mut() = connections;
332        self.refresh_slots()?;
333        Ok(())
334    }
335
336    // Query a node to discover slot-> master mappings.
337    fn refresh_slots(&self) -> RedisResult<()> {
338        let mut slots = self.slots.borrow_mut();
339        *slots = self.create_new_slots()?;
340
341        let mut nodes = slots.values().flatten().collect::<Vec<_>>();
342        nodes.sort_unstable();
343        nodes.dedup();
344
345        let mut connections = self.connections.borrow_mut();
346        *connections = nodes
347            .into_iter()
348            .filter_map(|addr| {
349                if connections.contains_key(addr) {
350                    let mut conn = connections.remove(addr).unwrap();
351                    if conn.check_connection() {
352                        return Some((addr.to_string(), conn));
353                    }
354                }
355
356                if let Ok(mut conn) = self.connect(addr) {
357                    if conn.check_connection() {
358                        return Some((addr.to_string(), conn));
359                    }
360                }
361
362                None
363            })
364            .collect();
365
366        Ok(())
367    }

369    fn create_new_slots(&self) -> RedisResult<SlotMap> {
370        let mut connections = self.connections.borrow_mut();
371        let mut new_slots = None;
372        let mut rng = thread_rng();
373        let len = connections.len();
374        let mut samples = connections.values_mut().choose_multiple(&mut rng, len);
375
376        for conn in samples.iter_mut() {
377            let value = conn.req_command(&slot_cmd())?;
378            if let Ok(mut slots_data) = parse_slots(value, self.cluster_params.tls) {
379                slots_data.sort_by_key(|s| s.start());
380                let last_slot = slots_data.iter().try_fold(0, |prev_end, slot_data| {
381                    if prev_end != slot_data.start() {
382                        return Err(RedisError::from((
383                            ErrorKind::ResponseError,
384                            "Slot refresh error.",
385                            format!(
386                                "Received overlapping slots {} and {}..{}",
387                                prev_end,
388                                slot_data.start(),
389                                slot_data.end()
390                            ),
391                        )));
392                    }
393                    Ok(slot_data.end() + 1)
394                })?;
395
396                if last_slot != SLOT_SIZE {
397                    return Err(RedisError::from((
398                        ErrorKind::ResponseError,
399                        "Slot refresh error.",
400                        format!("Lacks the slots >= {last_slot}"),
401                    )));
402                }
403
404                new_slots = Some(SlotMap::from_slots(
405                    &slots_data,
406                    self.cluster_params.read_from_replicas,
407                ));
408                break;
409            }
410        }
411
412        match new_slots {
413            Some(new_slots) => Ok(new_slots),
414            None => Err(RedisError::from((
415                ErrorKind::ResponseError,
416                "Slot refresh error.",
417                "didn't get any slots from server".to_string(),
418            ))),
419        }
420    }

422    fn connect(&self, node: &str) -> RedisResult<C> {
423        let params = self.cluster_params.clone();
424        let info = get_connection_info(node, params)?;
425
426        let mut conn = C::connect(info, None)?;
427        if self.cluster_params.read_from_replicas {
428            // If READONLY is sent to primary nodes, it will have no effect
429            cmd("READONLY").query(&mut conn)?;
430        }
431        conn.set_read_timeout(*self.read_timeout.borrow())?;
432        conn.set_write_timeout(*self.write_timeout.borrow())?;
433        Ok(conn)
434    }

436    fn get_connection<'a>(
437        &self,
438        connections: &'a mut HashMap<String, C>,
439        route: &Route,
440    ) -> RedisResult<(String, &'a mut C)> {
441        let slots = self.slots.borrow();
442        if let Some(addr) = slots.slot_addr_for_route(route) {
443            Ok((
444                addr.to_string(),
445                self.get_connection_by_addr(connections, addr)?,
446            ))
447        } else {
448            // try a random node next.  This is safe if slots are involved
449            // as a wrong node would reject the request.
450            Ok(get_random_connection(connections))
451        }
452    }

454    fn get_connection_by_addr<'a>(
455        &self,
456        connections: &'a mut HashMap<String, C>,
457        addr: &str,
458    ) -> RedisResult<&'a mut C> {
459        if connections.contains_key(addr) {
460            Ok(connections.get_mut(addr).unwrap())
461        } else {
462            // Create new connection.
463            // TODO: error handling
464            let conn = self.connect(addr)?;
465            Ok(connections.entry(addr.to_string()).or_insert(conn))
466        }
467    }

469    fn get_addr_for_cmd(&self, cmd: &Cmd) -> RedisResult<String> {
470        let slots = self.slots.borrow();
471
472        let addr_for_slot = |route: Route| -> RedisResult<String> {
473            let slot_addr = slots
474                .slot_addr_for_route(&route)
475                .ok_or((ErrorKind::ClusterDown, "Missing slot coverage"))?;
476            Ok(slot_addr.to_string())
477        };
478
479        match RoutingInfo::for_routable(cmd) {
480            Some(RoutingInfo::SingleNode(SingleNodeRoutingInfo::Random)) => {
481                let mut rng = thread_rng();
482                Ok(addr_for_slot(Route::new(
483                    rng.gen_range(0..SLOT_SIZE),
484                    SlotAddr::Master,
485                ))?)
486            }
487            Some(RoutingInfo::SingleNode(SingleNodeRoutingInfo::SpecificNode(route))) => {
488                Ok(addr_for_slot(route)?)
489            }
490            _ => fail!(UNROUTABLE_ERROR),
491        }
492    }

494    fn map_cmds_to_nodes(&self, cmds: &[Cmd]) -> RedisResult<Vec<NodeCmd>> {
495        let mut cmd_map: HashMap<String, NodeCmd> = HashMap::new();
496
497        for (idx, cmd) in cmds.iter().enumerate() {
498            let addr = self.get_addr_for_cmd(cmd)?;
499            let nc = cmd_map
500                .entry(addr.clone())
501                .or_insert_with(|| NodeCmd::new(addr));
502            nc.indexes.push(idx);
503            cmd.write_packed_command(&mut nc.pipe);
504        }
505
506        let mut result = Vec::new();
507        for (_, v) in cmd_map.drain() {
508            result.push(v);
509        }
510        Ok(result)
511    }

513    fn execute_on_multiple_nodes(
514        &self,
515        input: Input,
516        routing: MultipleNodeRoutingInfo,
517        response_policy: Option<ResponsePolicy>,
518    ) -> RedisResult<Value> {
519        let mut connections = self.connections.borrow_mut();
520        let slots = self.slots.borrow_mut();
521        let addresses = slots.addresses_for_multi_routing(&routing);
522
523        // TODO: reconnect and shit
524        let results = addresses.iter().enumerate().map(|(index, addr)| {
525            if let Some(connection) = connections.get_mut(*addr) {
526                match &routing {
527                    MultipleNodeRoutingInfo::MultiSlot(vec) => {
528                        let (_, indices) = vec.get(index).unwrap();
529                        let cmd = crate::cluster_routing::command_for_multi_slot_indices(
530                            &input,
531                            indices.iter(),
532                        );
533                        connection.req_command(&cmd)
534                    }
535                    _ => match input {
536                        Input::Slice { cmd, routable: _ } => connection.req_packed_command(cmd),
537                        Input::Cmd(cmd) => connection.req_command(cmd),
538                        Input::Commands {
539                            cmd: _,
540                            route: _,
541                            offset: _,
542                            count: _,
543                        } => Err((
544                            ErrorKind::ClientError,
545                            "req_packed_commands isn't supported with multiple nodes",
546                        )
547                            .into()),
548                    },
549                }
550            } else {
551                Err((
552                    ErrorKind::IoError,
553                    "connection error",
554                    format!("Disconnected from {addr}"),
555                )
556                    .into())
557            }
558        });
559        match response_policy {
560            Some(ResponsePolicy::AllSucceeded) => {
561                for result in results {
562                    result?;
563                }
564
565                Ok(Value::Okay)
566            }
567            Some(ResponsePolicy::OneSucceeded) => {
568                let mut last_failure = None;
569
570                for result in results {
571                    match result {
572                        Ok(val) => return Ok(val),
573                        Err(err) => last_failure = Some(err),
574                    }
575                }
576
577                Err(last_failure
578                    .unwrap_or_else(|| (ErrorKind::IoError, "Couldn't find a connection").into()))
579            }
580            Some(ResponsePolicy::OneSucceededNonEmpty) => {
581                let mut last_failure = None;
582
583                for result in results {
584                    match result {
585                        Ok(Value::Nil) => continue,
586                        Ok(val) => return Ok(val),
587                        Err(err) => last_failure = Some(err),
588                    }
589                }
590                Err(last_failure
591                    .unwrap_or_else(|| (ErrorKind::IoError, "Couldn't find a connection").into()))
592            }
593            Some(ResponsePolicy::Aggregate(op)) => {
594                let results = results.collect::<RedisResult<Vec<_>>>()?;
595                crate::cluster_routing::aggregate(results, op)
596            }
597            Some(ResponsePolicy::AggregateLogical(op)) => {
598                let results = results.collect::<RedisResult<Vec<_>>>()?;
599                crate::cluster_routing::logical_aggregate(results, op)
600            }
601            Some(ResponsePolicy::CombineArrays) => {
602                let results = results.collect::<RedisResult<Vec<_>>>()?;
603                match routing {
604                    MultipleNodeRoutingInfo::MultiSlot(vec) => {
605                        crate::cluster_routing::combine_and_sort_array_results(
606                            results,
607                            vec.iter().map(|(_, indices)| indices),
608                        )
609                    }
610                    _ => crate::cluster_routing::combine_array_results(results),
611                }
612            }
613            Some(ResponsePolicy::Special) | None => {
614                // This is our assumption - if there's no coherent way to aggregate the responses, we just map each response to the sender, and pass it to the user.
615                // TODO - once RESP3 is merged, return a map value here.
616                // TODO - once Value::Error is merged, we can use join_all and report separate errors and also pass successes.
617                let results = results
618                    .enumerate()
619                    .map(|(index, result)| {
620                        let addr = addresses[index];
621                        result.map(|val| {
622                            Value::Bulk(vec![Value::Data(addr.as_bytes().to_vec()), val])
623                        })
624                    })
625                    .collect::<RedisResult<Vec<_>>>()?;
626                Ok(Value::Bulk(results))
627            }
628        }
629    }

631    #[allow(clippy::unnecessary_unwrap)]
632    fn request(&self, input: Input) -> RedisResult<Output> {
633        let route_option = match &input {
634            Input::Slice { cmd: _, routable } => RoutingInfo::for_routable(routable),
635            Input::Cmd(cmd) => RoutingInfo::for_routable(*cmd),
636            Input::Commands {
637                cmd: _,
638                route,
639                offset: _,
640                count: _,
641            } => Some(RoutingInfo::SingleNode(route.clone())),
642        };
643        let route = match route_option {
644            Some(RoutingInfo::SingleNode(SingleNodeRoutingInfo::Random)) => None,
645            Some(RoutingInfo::SingleNode(SingleNodeRoutingInfo::SpecificNode(route))) => {
646                Some(route)
647            }
648            Some(RoutingInfo::MultiNode((multi_node_routing, response_policy))) => {
649                return self
650                    .execute_on_multiple_nodes(input, multi_node_routing, response_policy)
651                    .map(Output::Single);
652            }
653            None => fail!(UNROUTABLE_ERROR),
654        };
655
656        let mut retries = 0;
657        let mut redirected = None::<Redirect>;
658
659        loop {
660            // Get target address and response.
661            let (addr, rv) = {
662                let mut connections = self.connections.borrow_mut();
663                let (addr, conn) = if let Some(redirected) = redirected.take() {
664                    let (addr, is_asking) = match redirected {
665                        Redirect::Moved(addr) => (addr, false),
666                        Redirect::Ask(addr) => (addr, true),
667                    };
668                    let conn = self.get_connection_by_addr(&mut connections, &addr)?;
669                    if is_asking {
670                        // if we are in asking mode we want to feed a single
671                        // ASKING command into the connection before what we
672                        // actually want to execute.
673                        conn.req_packed_command(&b"*1\r\n$6\r\nASKING\r\n"[..])?;
674                    }
675                    (addr.to_string(), conn)
676                } else if route.is_none() {
677                    get_random_connection(&mut connections)
678                } else {
679                    self.get_connection(&mut connections, route.as_ref().unwrap())?
680                };
681                (addr, input.send(conn))
682            };
683
684            match rv {
685                Ok(rv) => return Ok(rv),
686                Err(err) => {
687                    if retries == self.cluster_params.retry_params.number_of_retries {
688                        return Err(err);
689                    }
690                    retries += 1;
691
692                    match err.kind() {
693                        ErrorKind::Ask => {
694                            redirected = err
695                                .redirect_node()
696                                .map(|(node, _slot)| Redirect::Ask(node.to_string()));
697                        }
698                        ErrorKind::Moved => {
699                            // Refresh slots.
700                            self.refresh_slots()?;
701                            // Request again.
702                            redirected = err
703                                .redirect_node()
704                                .map(|(node, _slot)| Redirect::Moved(node.to_string()));
705                        }
706                        ErrorKind::TryAgain | ErrorKind::ClusterDown => {
707                            // Sleep and retry.
708                            let sleep_time = self
709                                .cluster_params
710                                .retry_params
711                                .wait_time_for_retry(retries);
712                            thread::sleep(sleep_time);
713                        }
714                        ErrorKind::IoError => {
715                            if *self.auto_reconnect.borrow() {
716                                if let Ok(mut conn) = self.connect(&addr) {
717                                    if conn.check_connection() {
718                                        self.connections.borrow_mut().insert(addr, conn);
719                                    }
720                                }
721                            }
722                        }
723                        _ => {
724                            if !err.is_retryable() {
725                                return Err(err);
726                            }
727                        }
728                    }
729                }
730            }
731        }
732    }
733
734    fn send_recv_and_retry_cmds(&self, cmds: &[Cmd]) -> RedisResult<Vec<Value>> {
735        // Vector to hold the results, pre-populated with `Nil` values. This allows the original
736        // cmd ordering to be re-established by inserting the response directly into the result
737        // vector (e.g., results[10] = response).
738        let mut results = vec![Value::Nil; cmds.len()];
739
740        let to_retry = self
741            .send_all_commands(cmds)
742            .and_then(|node_cmds| self.recv_all_commands(&mut results, &node_cmds))?;
743
744        if to_retry.is_empty() {
745            return Ok(results);
746        }
747
748        // Refresh the slots to ensure that we have a clean slate for the retry attempts.
749        self.refresh_slots()?;
750
751        // Given that there are commands that need to be retried, it means something in the cluster
752        // topology changed. Execute each command seperately to take advantage of the existing
753        // retry logic that handles these cases.
754        for retry_idx in to_retry {
755            let cmd = &cmds[retry_idx];
756            results[retry_idx] = self.request(Input::Cmd(cmd))?.into();
757        }
758        Ok(results)
759    }
760
761    // Build up a pipeline per node, then send it
762    fn send_all_commands(&self, cmds: &[Cmd]) -> RedisResult<Vec<NodeCmd>> {
763        let mut connections = self.connections.borrow_mut();
764
765        let node_cmds = self.map_cmds_to_nodes(cmds)?;
766        for nc in &node_cmds {
767            self.get_connection_by_addr(&mut connections, &nc.addr)?
768                .send_packed_command(&nc.pipe)?;
769        }
770        Ok(node_cmds)
771    }
772
773    // Receive from each node, keeping track of which commands need to be retried.
774    fn recv_all_commands(
775        &self,
776        results: &mut [Value],
777        node_cmds: &[NodeCmd],
778    ) -> RedisResult<Vec<usize>> {
779        let mut to_retry = Vec::new();
780        let mut connections = self.connections.borrow_mut();
781        let mut first_err = None;
782
783        for nc in node_cmds {
784            for cmd_idx in &nc.indexes {
785                match self
786                    .get_connection_by_addr(&mut connections, &nc.addr)?
787                    .recv_response()
788                {
789                    Ok(item) => results[*cmd_idx] = item,
790                    Err(err) if err.is_cluster_error() => to_retry.push(*cmd_idx),
791                    Err(err) => first_err = first_err.or(Some(err)),
792                }
793            }
794        }
795        match first_err {
796            Some(err) => Err(err),
797            None => Ok(to_retry),
798        }
799    }
800}

802const MULTI: &[u8] = "*1\r\n$5\r\nMULTI\r\n".as_bytes();
803impl<C: Connect + ConnectionLike> ConnectionLike for ClusterConnection<C> {
804    fn supports_pipelining(&self) -> bool {
805        false
806    }
807
808    fn req_command(&mut self, cmd: &Cmd) -> RedisResult<Value> {
809        self.request(Input::Cmd(cmd)).map(|res| res.into())
810    }
811
812    fn req_packed_command(&mut self, cmd: &[u8]) -> RedisResult<Value> {
813        let actual_cmd = if cmd.starts_with(MULTI) {
814            &cmd[MULTI.len()..]
815        } else {
816            cmd
817        };
818        let value = parse_redis_value(actual_cmd)?;
819        self.request(Input::Slice {
820            cmd,
821            routable: value,
822        })
823        .map(|res| res.into())
824    }
825
826    fn req_packed_commands(
827        &mut self,
828        cmd: &[u8],
829        offset: usize,
830        count: usize,
831    ) -> RedisResult<Vec<Value>> {
832        let actual_cmd = if cmd.starts_with(MULTI) {
833            &cmd[MULTI.len()..]
834        } else {
835            cmd
836        };
837        let value = parse_redis_value(actual_cmd)?;
838        let route = match RoutingInfo::for_routable(&value) {
839            Some(RoutingInfo::MultiNode(_)) => None,
840            Some(RoutingInfo::SingleNode(route)) => Some(route),
841            None => None,
842        }
843        .unwrap_or(SingleNodeRoutingInfo::Random);
844        self.request(Input::Commands {
845            cmd,
846            offset,
847            count,
848            route,
849        })
850        .map(|res| res.into())
851    }
852
853    fn get_db(&self) -> i64 {
854        0
855    }
856
857    fn is_open(&self) -> bool {
858        let connections = self.connections.borrow();
859        for conn in connections.values() {
860            if !conn.is_open() {
861                return false;
862            }
863        }
864        true
865    }
866
867    fn check_connection(&mut self) -> bool {
868        let mut connections = self.connections.borrow_mut();
869        for conn in connections.values_mut() {
870            if !conn.check_connection() {
871                return false;
872            }
873        }
874        true
875    }
876}

910// Parse slot data from raw redis value.
911pub(crate) fn parse_slots(raw_slot_resp: Value, tls: Option<TlsMode>) -> RedisResult<Vec<Slot>> {
912    // Parse response.
913    let mut result = Vec::with_capacity(2);
914
915    if let Value::Bulk(items) = raw_slot_resp {
916        let mut iter = items.into_iter();
917        while let Some(Value::Bulk(item)) = iter.next() {
918            if item.len() < 3 {
919                continue;
920            }
921
922            let start = if let Value::Int(start) = item[0] {
923                start as u16
924            } else {
925                continue;
926            };
927
928            let end = if let Value::Int(end) = item[1] {
929                end as u16
930            } else {
931                continue;
932            };
933
934            let mut nodes: Vec<String> = item
935                .into_iter()
936                .skip(2)
937                .filter_map(|node| {
938                    if let Value::Bulk(node) = node {
939                        if node.len() < 2 {
940                            return None;
941                        }
942
943                        let ip = if let Value::Data(ref ip) = node[0] {
944                            String::from_utf8_lossy(ip)
945                        } else {
946                            return None;
947                        };
948                        if ip.is_empty() {
949                            return None;
950                        }
951
952                        let port = if let Value::Int(port) = node[1] {
953                            port as u16
954                        } else {
955                            return None;
956                        };
957                        // This is only "stringifying" IP addresses, so `TLS parameters` are not required
958                        Some(get_connection_addr(ip.into_owned(), port, tls, None).to_string())
959                    } else {
960                        None
961                    }
962                })
963                .collect();
964
965            if nodes.is_empty() {
966                continue;
967            }
968
969            let replicas = nodes.split_off(1);
970            result.push(Slot::new(start, end, nodes.pop().unwrap(), replicas));
971        }
972    }
973
974    Ok(result)
975}

977// The node string passed to this function will always be in the format host:port as it is either:
978// - Created by calling ConnectionAddr::to_string (unix connections are not supported in cluster mode)
979// - Returned from redis via the ASK/MOVED response
980pub(crate) fn get_connection_info(
981    node: &str,
982    cluster_params: ClusterParams,
983) -> RedisResult<ConnectionInfo> {
984    let invalid_error = || (ErrorKind::InvalidClientConfig, "Invalid node string");
985
986    let (host, port) = node
987        .rsplit_once(':')
988        .and_then(|(host, port)| {
989            Some(host.trim_start_matches('[').trim_end_matches(']'))
990                .filter(|h| !h.is_empty())
991                .zip(u16::from_str(port).ok())
992        })
993        .ok_or_else(invalid_error)?;
994
995    Ok(ConnectionInfo {
996        addr: get_connection_addr(
997            host.to_string(),
998            port,
999            cluster_params.tls,
1000            cluster_params.tls_params,
1001        ),
1002        redis: RedisConnectionInfo {
1003            password: cluster_params.password,
1004            username: cluster_params.username,
1005            ..Default::default()
1006        },
1007    })
1008}

1010fn get_connection_addr(
1011    host: String,
1012    port: u16,
1013    tls: Option<TlsMode>,
1014    tls_params: Option<TlsConnParams>,
1015) -> ConnectionAddr {
1016    match tls {
1017        Some(TlsMode::Secure) => ConnectionAddr::TcpTls {
1018            host,
1019            port,
1020            insecure: false,
1021            tls_params,
1022        },
1023        Some(TlsMode::Insecure) => ConnectionAddr::TcpTls {
1024            host,
1025            port,
1026            insecure: true,
1027            tls_params,
1028        },
1029        _ => ConnectionAddr::Tcp(host, port),
1030    }
1031}
