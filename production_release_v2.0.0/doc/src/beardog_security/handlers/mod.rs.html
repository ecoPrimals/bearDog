43impl BearDogSecurityProvider {
44    /// Create a new security provider instance with configuration
45    pub async fn new_with_config(config: SecurityProviderConfig) -> BearDogResult<Self> {
46        let rate_limiter = RateLimiter {
47            config: config.rate_limit_config.clone(),
48            state: HashMap::new(),
49        };
50
51        // Initialize standalone memory key manager for "crypto in your pocket"
52        let _memory_key_manager = if config.memory_key_manager.is_some() {
53            let key_config = MemoryKeyConfig {
54                max_keys: 1000,              // Default value
55                enable_vault_sharing: false, // Default value
56                ..Default::default()
57            };
58            Some(MemoryKeyManager::new(key_config).await?)
59        } else {
60            None
61        };
62
63        // Initialize recovery system for distributed account recovery
64        let _recovery_manager = Some(crate::recovery::RecoveryManager::new());
65
66        Ok(Self {
67            config,
68            rate_limiter,
69            security_rules: SecurityRules {
70                rules: Vec::new(),
71                default_policy: PolicyDecision::Deny,
72                context: SecurityContext::default(),
73            },
74            locked_accounts: std::sync::Arc::new(tokio::sync::RwLock::new(HashMap::new())),
75            failed_attempts: std::sync::Arc::new(tokio::sync::RwLock::new(HashMap::new())),
76            metrics: SecurityProviderMetrics::default(),
77            session_store: SessionStore::new(),
78            audit_manager: AuditManager::new(),
79        })
80    }
