1//! Federation Recovery
2//!
3//! This module handles federation recovery between trusted BearDog instances.
4
5use chrono::{DateTime, Utc};
6use serde::{Deserialize, Serialize};
7
8/// Federation recovery configuration
9#[derive(Debug, Clone, Serialize, Deserialize)]
10pub struct FederationRecoveryConfig {
11    /// User ID this config belongs to
12    pub user_id: String,
13    /// List of trusted BearDog instances
14    pub trusted_instances: Vec<TrustedInstance>,
15    /// Minimum number of instances needed for recovery
16    pub min_instances_required: u32,
17    /// Whether federation recovery is enabled
18    pub enabled: bool,
19    /// Cross-instance verification settings
20    pub verification_settings: FederationVerificationSettings,
21}
22
23/// Trusted BearDog instance for federation recovery
24#[derive(Debug, Clone, Serialize, Deserialize)]
25pub struct TrustedInstance {
26    /// Instance ID
27    pub id: String,
28    /// Instance endpoint
29    pub endpoint: String,
30    /// Instance public key
31    pub public_key: String,
32    /// Trust level (0-100)
33    pub trust_level: u8,
34    /// When this instance was added to trusted list
35    pub added_at: DateTime<Utc>,
36    /// Whether this instance is currently active
37    pub active: bool,
38}
39
40/// Federation verification settings
41#[derive(Debug, Clone, Serialize, Deserialize)]
42pub struct FederationVerificationSettings {
43    /// Require cryptographic proof of identity
44    pub require_crypto_proof: bool,
45    /// Require reputation verification
46    pub require_reputation: bool,
47    /// Minimum reputation score required
48    pub min_reputation_score: u32,
49    /// Verification timeout in minutes
50    pub verification_timeout_minutes: u32,
51}
52
53impl Default for FederationRecoveryConfig {
54    fn default() -> Self {
55        Self {
56            user_id: String::new(),
57            trusted_instances: Vec::new(),
58            min_instances_required: 2,
59            enabled: false,
60            verification_settings: FederationVerificationSettings::default(),
61        }
62    }
63}
64
65impl Default for FederationVerificationSettings {
66    fn default() -> Self {
67        Self {
68            require_crypto_proof: true,
69            require_reputation: false,
70            min_reputation_score: 70,
71            verification_timeout_minutes: 30,
72        }
73    }
74}
