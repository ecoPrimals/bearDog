137/// Recovery audit entry
138#[derive(Debug, Clone, Serialize, Deserialize)]
139pub struct RecoveryAuditEntry {
140    /// Unique entry ID
141    pub id: String,
142    /// Type of recovery event
143    pub event_type: RecoveryEventType,
144    /// User ID associated with the event
145    pub user_id: String,
146    /// When the event occurred
147    pub timestamp: DateTime<Utc>,
148    /// Additional event details
149    pub details: HashMap<String, String>,
150    /// IP address of the request (if applicable)
151    pub ip_address: Option<String>,
152    /// User agent of the request (if applicable)
153    pub user_agent: Option<String>,
154    /// Session ID associated with the event (if applicable)
155    pub session_id: Option<String>,
156}
157
158impl RecoveryAuditEntry {
159    /// Create a new recovery audit entry
160    pub fn new(
161        event_type: RecoveryEventType,
162        user_id: String,
163        details: HashMap<String, String>,
164    ) -> Self {
165        Self {
166            id: uuid::Uuid::new_v4().to_string(),
167            event_type,
168            user_id,
169            timestamp: Utc::now(),
170            details,
171            ip_address: None,
172            user_agent: None,
173            session_id: None,
174        }
175    }
176
177    /// Set the IP address for this audit entry
178    pub fn with_ip_address(mut self, ip_address: String) -> Self {
179        self.ip_address = Some(ip_address);
180        self
181    }
182
183    /// Set the user agent for this audit entry
184    pub fn with_user_agent(mut self, user_agent: String) -> Self {
185        self.user_agent = Some(user_agent);
186        self
187    }
188
189    /// Set the session ID for this audit entry
190    pub fn with_session_id(mut self, session_id: String) -> Self {
191        self.session_id = Some(session_id);
192        self
193    }
194}
