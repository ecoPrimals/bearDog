1//! Address Management - Keys ARE the Authority
2//!
3//! This module implements BearDog's core principle: Keys are the source of authority,
4//! addresses are derived from keys, and no external validation is required.
5//!
6//! ## Core Principle: Self-Sovereign Address Generation
7//!
8//! - **Keys generate addresses** - Addresses are cryptographically derived from keys
9//! - **No external dependencies** - Address generation is purely mathematical
10//! - **Multiple formats supported** - Bitcoin, Ethereum, custom BearDog addresses
11//! - **Deterministic derivation** - Same key always generates same address
12//! - **Hierarchical support** - HD wallet-style key/address trees
13

30/// Cached address information
31#[derive(Debug, Clone)]
32struct CachedAddress {
33    /// The derived address
34    address: String,
35    /// Format used for derivation
36    format: AddressFormat,
37    /// Source key fingerprint (for cache invalidation)
38    #[allow(dead_code)]
39    key_fingerprint: String,
40    /// Derivation path (if applicable)
41    derivation_path: Option<String>,
42    /// Creation timestamp (for cache expiration)
43    #[allow(dead_code)]
44    created_at: chrono::DateTime<chrono::Utc>,
45}

47/// Address formats supported by BearDog
48#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
49pub enum AddressFormat {
50    /// Bitcoin Legacy (P2PKH) - starts with '1'
51    BitcoinLegacy,
52    /// Bitcoin SegWit (P2WPKH) - starts with 'bc1q'
53    BitcoinSegWit,
54    /// Ethereum addresses - starts with '0x'
55    Ethereum,
56    /// BearDog native format - starts with 'bear'
57    BearDogNative,
58    /// Ed25519 public key hash - starts with 'ed25519'
59    Ed25519Hash,
60    /// Custom format for specific use cases
61    Custom { name: String, prefix: String },
62}

105    /// Generate address from Ed25519 public key
106    pub fn generate_address_from_ed25519(
107        &mut self,
108        public_key: &[u8],
109        format: AddressFormat,
110    ) -> BearDogResult<AddressInfo> {
111        if public_key.len() != 32 {
112            return Err(BearDogError::InvalidInput {
113                message: format!("Invalid Ed25519 public key length: {}", public_key.len()),
114            });
115        }
116
117        let key_fingerprint = self.create_key_fingerprint(public_key);
118        let cache_key = format!("{key_fingerprint}:{format:?}");
119
120        // Check cache first
121        if let Some(cached) = self.address_cache.get(&cache_key) {
122            return Ok(AddressInfo {
123                address: cached.address.clone(),
124                format: cached.format.clone(),
125                public_key: Some(public_key.to_vec()),
126                derivation_path: cached.derivation_path.clone(),
127                metadata: HashMap::new(),
128                verified: true,
129            });
130        }
131
132        // Generate address based on format
133        let address = match format {
134            AddressFormat::BearDogNative => self.generate_beardog_address(public_key)?,
135            AddressFormat::Ed25519Hash => self.generate_ed25519_hash_address(public_key)?,
136            AddressFormat::Ethereum => self.generate_ethereum_style_address(public_key)?,
137            AddressFormat::BitcoinLegacy => self.generate_bitcoin_legacy_address(public_key)?,
138            AddressFormat::BitcoinSegWit => self.generate_bitcoin_segwit_address(public_key)?,
139            AddressFormat::Custom {
140                ref name,
141                ref prefix,
142            } => self.generate_custom_address(public_key, name, prefix)?,
143        };
144
145        // Cache the result
146        let cached_address = CachedAddress {
147            address: address.clone(),
148            format: format.clone(),
149            key_fingerprint,
150            derivation_path: None,
151            created_at: chrono::Utc::now(),
152        };
153        self.address_cache.insert(cache_key, cached_address);
154
155        Ok(AddressInfo {
156            address,
157            format,
158            public_key: Some(public_key.to_vec()),
159            derivation_path: None,
160            metadata: HashMap::new(),
161            verified: true,
162        })
163    }

165    /// Generate address from raw key material (for other key types)
166    pub fn generate_address_from_key_material(
167        &mut self,
168        key_material: &[u8],
169        key_type: &str,
170        format: AddressFormat,
171    ) -> BearDogResult<AddressInfo> {
172        match key_type {
173            "Ed25519" => {
174                if key_material.len() == 64 {
175                    // Assume it's a keypair, extract public key
176                    self.generate_address_from_ed25519(&key_material[32..], format)
177                } else if key_material.len() == 32 {
178                    // Assume it's just the public key
179                    self.generate_address_from_ed25519(key_material, format)
180                } else {
181                    Err(BearDogError::InvalidInput {
182                        message: format!(
183                            "Invalid Ed25519 key material length: {}",
184                            key_material.len()
185                        ),
186                    })
187                }
188            }
189            "AES-256" => {
190                // For symmetric keys, generate address from key hash
191                self.generate_address_from_hash(key_material, format)
192            }
193            _ => {
194                // For unknown key types, use hash-based address generation
195                self.generate_address_from_hash(key_material, format)
196            }
197        }
198    }

200    /// Verify that an address was derived from a specific key
201    pub fn verify_address_ownership(
202        &mut self,
203        address: &str,
204        public_key: &[u8],
205        format: AddressFormat,
206    ) -> BearDogResult<bool> {
207        // Generate address from the key and compare
208        let derived_address = self.generate_address_from_ed25519(public_key, format)?;
209        Ok(derived_address.address == address)
210    }

212    /// Generate hierarchical deterministic address
213    pub fn generate_hd_address(
214        &mut self,
215        master_key: &[u8],
216        derivation_path: &DerivationPath,
217        format: AddressFormat,
218    ) -> BearDogResult<AddressInfo> {
219        // Derive child key from master key using derivation path
220        let child_key = self.derive_child_key(master_key, derivation_path)?;
221
222        // Generate address from child key
223        let mut address_info = self.generate_address_from_ed25519(&child_key, format)?;
224        address_info.derivation_path = Some(self.derivation_path_to_string(derivation_path));
225
226        Ok(address_info)
227    }

249    /// Generate BearDog native address format
250    fn generate_beardog_address(&self, public_key: &[u8]) -> BearDogResult<String> {
251        // BearDog format: bear + base58(hash160(pubkey)) + checksum
252        let hash160 = self.hash160(public_key);
253        let address_bytes = self.add_checksum(&hash160, b"bear")?;
254        Ok(format!("bear{}", self.base58_encode(&address_bytes)))
255    }

257    /// Generate Ed25519 hash address
258    fn generate_ed25519_hash_address(&self, public_key: &[u8]) -> BearDogResult<String> {
259        // Ed25519 format: ed25519 + hex(sha256(pubkey)[:20])
260        let mut hasher = Sha256::new();
261        hasher.update(public_key);
262        let hash = hasher.finalize();
263        Ok(format!("ed25519{}", hex::encode(&hash[..20])))
264    }

266    /// Generate Ethereum-style address
267    fn generate_ethereum_style_address(&self, public_key: &[u8]) -> BearDogResult<String> {
268        // Ethereum format: 0x + hex(keccak256(pubkey)[-20:])
269        let hash = self.keccak256(public_key);
270        Ok(format!("0x{}", hex::encode(&hash[12..])))
271    }

273    /// Generate Bitcoin legacy address (P2PKH)
274    fn generate_bitcoin_legacy_address(&self, public_key: &[u8]) -> BearDogResult<String> {
275        // Bitcoin format: base58(version + hash160(pubkey) + checksum)
276        let mut address_bytes = vec![0x00]; // Version byte for mainnet
277        address_bytes.extend_from_slice(&self.hash160(public_key));
278        let checksum = self.double_sha256_checksum(&address_bytes);
279        address_bytes.extend_from_slice(&checksum[..4]);
280        Ok(self.base58_encode(&address_bytes))
281    }

283    /// Generate Bitcoin SegWit address (P2WPKH)
284    fn generate_bitcoin_segwit_address(&self, public_key: &[u8]) -> BearDogResult<String> {
285        // SegWit format: bech32 encoding of witness program
286        let hash160 = self.hash160(public_key);
287        Ok(format!("bc1q{}", self.bech32_encode(&hash160)))
288    }

290    /// Generate custom format address
291    fn generate_custom_address(
292        &self,
293        public_key: &[u8],
294        name: &str,
295        prefix: &str,
296    ) -> BearDogResult<String> {
297        // Custom format: prefix + hash + checksum
298        let mut hasher = Sha256::new();
299        hasher.update(name.as_bytes());
300        hasher.update(public_key);
301        let hash = hasher.finalize();
302        Ok(format!("{}{}", prefix, hex::encode(&hash[..16])))
303    }

305    /// Generate address from hash (for non-public key material)
306    fn generate_address_from_hash(
307        &self,
308        data: &[u8],
309        format: AddressFormat,
310    ) -> BearDogResult<AddressInfo> {
311        let hash = {
312            let mut hasher = Sha256::new();
313            hasher.update(data);
314            hasher.finalize().to_vec()
315        };
316
317        let address = match format {
318            AddressFormat::BearDogNative => {
319                format!("bear{}", hex::encode(&hash[..20]))
320            }
321            AddressFormat::Ed25519Hash => {
322                format!("ed25519{}", hex::encode(&hash[..20]))
323            }
324            AddressFormat::Ethereum => {
325                format!("0x{}", hex::encode(&hash[..20]))
326            }
327            _ => {
328                format!("hash{}", hex::encode(&hash[..20]))
329            }
330        };
331
332        Ok(AddressInfo {
333            address,
334            format,
335            public_key: None,
336            derivation_path: None,
337            metadata: HashMap::new(),
338            verified: false, // Cannot verify hash-based addresses
339        })
340    }

349    /// Derive child key using hierarchical deterministic derivation
350    fn derive_child_key(&self, master_key: &[u8], path: &DerivationPath) -> BearDogResult<Vec<u8>> {
351        let mut current_key = master_key.to_vec();
352
353        for (i, &component) in path.components.iter().enumerate() {
354            let hardened = path.hardened.get(i).copied().unwrap_or(false);
355            current_key = self.derive_single_child(&current_key, component, hardened)?;
356        }
357
358        // Return the public key portion (last 32 bytes for Ed25519)
359        if current_key.len() >= 32 {
360            Ok(current_key[current_key.len() - 32..].to_vec())
361        } else {
362            Err(BearDogError::Crypto {
363                message: "Invalid derived key length".to_string(),
364            })
365        }
366    }

368    /// Derive single child key
369    fn derive_single_child(
370        &self,
371        parent_key: &[u8],
372        index: u32,
373        hardened: bool,
374    ) -> BearDogResult<Vec<u8>> {
375        let mut hasher = Sha256::new();
376        hasher.update(parent_key);
377        hasher.update(if hardened { "hardened" } else { "normal" });
378        hasher.update(index.to_be_bytes());
379        Ok(hasher.finalize().to_vec())
380    }

382    /// Convert derivation path to string representation
383    fn derivation_path_to_string(&self, path: &DerivationPath) -> String {
384        let mut result = String::from("m");
385        for (i, &component) in path.components.iter().enumerate() {
386            let hardened = path.hardened.get(i).copied().unwrap_or(false);
387            if hardened {
388                result.push_str(&format!("/{component}'"));
389            } else {
390                result.push_str(&format!("/{component}"));
391            }
392        }
393        result
394    }

398    /// Calculate RIPEMD160(SHA256(data))
399    fn hash160(&self, data: &[u8]) -> Vec<u8> {
400        let sha256_hash = {
401            let mut hasher = Sha256::new();
402            hasher.update(data);
403            hasher.finalize()
404        };
405
406        let mut ripemd_hasher = Ripemd160::new();
407        ripemd_hasher.update(sha256_hash);
408        ripemd_hasher.finalize().to_vec()
409    }

412    fn keccak256(&self, data: &[u8]) -> Vec<u8> {
413        // Simplified Keccak256 - in production, use proper Keccak implementation
414        let mut hasher = Sha256::new();
415        hasher.update(b"keccak256:");
416        hasher.update(data);
417        hasher.finalize().to_vec()

420    /// Add checksum to address bytes
421    fn add_checksum(&self, data: &[u8], prefix: &[u8]) -> BearDogResult<Vec<u8>> {
422        let mut hasher = Sha256::new();
423        hasher.update(prefix);
424        hasher.update(data);
425        let hash = hasher.finalize();
426
427        let mut result = data.to_vec();
428        result.extend_from_slice(&hash[..4]);
429        Ok(result)
430    }

432    /// Calculate double SHA256 checksum (Bitcoin style)
433    fn double_sha256_checksum(&self, data: &[u8]) -> Vec<u8> {
434        let first_hash = {
435            let mut hasher = Sha256::new();
436            hasher.update(data);
437            hasher.finalize()
438        };
439
440        let mut second_hasher = Sha256::new();
441        second_hasher.update(first_hash);
442        second_hasher.finalize().to_vec()
443    }

445    /// Base58 encoding (simplified)
446    fn base58_encode(&self, data: &[u8]) -> String {
447        // Simplified base58 encoding - in production, use proper base58 library
448        const ALPHABET: &[u8] = b"123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
449
450        if data.is_empty() {
451            return String::new();
452        }
453
454        // Convert to base58 (simplified implementation)
455        let mut result = String::new();
456        for chunk in data.chunks(8) {
457            let mut num = 0u64;
458            for &byte in chunk {
459                num = num * 256 + byte as u64;
460            }
461
462            while num > 0 {
463                result.push(ALPHABET[(num % 58) as usize] as char);
464                num /= 58;
465            }
466        }
467
468        result.chars().rev().collect()
469    }

484impl DerivationPath {
485    /// Create new derivation path
486    pub fn new(components: Vec<u32>, hardened: Vec<bool>) -> Self {
487        Self {
488            components,
489            hardened,
490        }
491    }
492
493    /// Parse derivation path from string (e.g., "m/44'/0'/0'/0/0")
494    pub fn from_string(path: &str) -> BearDogResult<Self> {
495        if !path.starts_with("m/") && !path.starts_with("M/") {
496            return Err(BearDogError::InvalidInput {
497                message: "Derivation path must start with 'm/'".to_string(),
498            });
499        }
500
501        let parts: Vec<&str> = path[2..].split('/').collect();
502        let mut components = Vec::new();
503        let mut hardened = Vec::new();
504
505        for part in parts {
506            if part.is_empty() {
507                continue;
508            }
509
510            let (num_str, is_hardened) = if part.ends_with('\'') || part.ends_with('h') {
511                (&part[..part.len() - 1], true)
512            } else {
513                (part, false)
514            };
515
516            let component = num_str
517                .parse::<u32>()
518                .map_err(|_| BearDogError::InvalidInput {
519                    message: format!("Invalid derivation path component: {part}"),
520                })?;
521
522            components.push(component);
523            hardened.push(is_hardened);
524        }
525
526        Ok(Self {
527            components,
528            hardened,
529        })
530    }
