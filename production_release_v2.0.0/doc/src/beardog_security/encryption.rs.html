33/// Supported encryption algorithms
34#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, Default)]
35pub enum EncryptionAlgorithm {
36    /// AES-256 in GCM mode (recommended for most use cases)
37    #[default]
38    Aes256Gcm,
39    /// ChaCha20-Poly1305 (recommended for high-performance scenarios)
40    ChaCha20Poly1305,
41    /// Quantum-resistant encryption
42    QuantumResistant,
43}
44
45impl fmt::Display for EncryptionAlgorithm {
46    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
47        match self {
48            EncryptionAlgorithm::Aes256Gcm => write!(f, "AES256-GCM"),
49            EncryptionAlgorithm::ChaCha20Poly1305 => write!(f, "ChaCha20-Poly1305"),
50            EncryptionAlgorithm::QuantumResistant => write!(f, "Quantum-Resistant"),
51        }
52    }
53}

126    pub async fn generate_key(
127        &self,
128        algorithm: String,
129        purpose: String,
130    ) -> BearDogResult<(String, Vec<u8>)> {
131        info!(
132            "🔑 Generating encryption key: {} for {}",
133            algorithm, purpose
134        );
135
136        let key_id = uuid::Uuid::new_v4().to_string();
137        let key = match algorithm.as_str() {
138            "AES256" => crate::crypto_utils::BearDogCrypto::secure_random_bytes(32),
139            "RSA2048" => {
140                // Generate RSA-2048 equivalent key material (256 bytes)
141                // In a real implementation, this would use proper RSA key generation
142
143                crate::crypto_utils::BearDogCrypto::secure_random_bytes(256)
144            }
145            "RSA4096" => {
146                // Generate RSA-4096 equivalent key material (512 bytes)
147                // In a real implementation, this would use proper RSA key generation
148
149                crate::crypto_utils::BearDogCrypto::secure_random_bytes(512)
150            }
151            "ECDSA_P256" => {
152                // Generate ECDSA P-256 equivalent key material (32 bytes for private key)
153                // In a real implementation, this would use proper ECDSA key generation
154
155                crate::crypto_utils::BearDogCrypto::secure_random_bytes(32)
156            }
157            "ECDSA_P384" => {
158                // Generate ECDSA P-384 equivalent key material (48 bytes for private key)
159                // In a real implementation, this would use proper ECDSA key generation
160
161                crate::crypto_utils::BearDogCrypto::secure_random_bytes(48)
162            }
163            "ECDSA_P521" => {
164                // Ring doesn't support P-521, so we'll use a different approach or fallback
165                // P-521 private key size
166                crate::crypto_utils::BearDogCrypto::secure_random_bytes(66)
167            }
168            _ => {
169                return Err(BearDogError::Encryption {
170                    operation: "key_generation".to_string(),
171                    message: "Unsupported algorithm".to_string(),
172                })
173            }
174        };
175
176        // Store the key
177        let key_bytes = key?;
178        self.keys
179            .write()
180            .await
181            .insert(key_id.clone(), key_bytes.clone());
182
183        info!("✅ Generated key: {}", key_id);
184        Ok((key_id, key_bytes))
185    }

203    /// Encrypt data with a specific key
204    pub async fn encrypt_with_key(&self, data: &[u8], key: &[u8]) -> BearDogResult<Vec<u8>> {
205        // Use AES-GCM for encryption
206        let key = Key::<Aes256Gcm>::from_slice(&key[..32]);
207        let cipher = Aes256Gcm::new(key);
208        let nonce = Aes256Gcm::generate_nonce(&mut OsRng);
209
210        let ciphertext = cipher
211            .encrypt(&nonce, data)
212            .map_err(|e| BearDogError::Encryption {
213                operation: "encrypt_with_key".to_string(),
214                message: format!("AES encryption failed: {e}"),
215            })?;
216
217        // Prepend nonce to ciphertext
218        let mut result = nonce.to_vec();
219        result.extend_from_slice(&ciphertext);
220        Ok(result)
221    }

223    /// Decrypt data with a specific key
224    pub async fn decrypt_with_key(&self, data: &[u8], key: &[u8]) -> BearDogResult<Vec<u8>> {
225        if data.len() < 12 {
226            return Err(BearDogError::Encryption {
227                operation: "decrypt_with_key".to_string(),
228                message: "Invalid ciphertext length".to_string(),
229            });
230        }
231
232        let (nonce_bytes, ciphertext) = data.split_at(12);
233        let nonce = Nonce::from_slice(nonce_bytes);
234
235        let key = Key::<Aes256Gcm>::from_slice(&key[..32]);
236        let cipher = Aes256Gcm::new(key);
237        let plaintext =
238            cipher
239                .decrypt(nonce, ciphertext)
240                .map_err(|e| BearDogError::Encryption {
241                    operation: "decrypt_with_key".to_string(),
242                    message: format!("AES decryption failed: {e}"),
243                })?;
244
245        Ok(plaintext)
246    }

248    /// Derive key from password using Argon2
249    pub async fn derive_key_from_password(
250        &self,
251        password: &str,
252        params: Option<KeyDerivationParams>,
253    ) -> BearDogResult<Vec<u8>> {
254        let params = params.unwrap_or_else(|| KeyDerivationParams {
255            salt: self.generate_salt(),
256            iterations: self.config.key_derivation_iterations,
257            memory_cost: 65536, // 64 MB
258            parallelism: 1,
259        });
260
261        let salt = SaltString::encode_b64(&params.salt)
262            .map_err(|e| BearDogError::internal(format!("Salt encoding failed: {e}")))?;
263
264        let password_hash = Argon2::default()
265            .hash_password(password.as_bytes(), &salt)
266            .map_err(|e| BearDogError::KeyDerivation {
267                message: format!("Password hashing failed: {e:?}"),
268            })?;
269
270        password_hash
271            .hash
272            .ok_or_else(|| BearDogError::KeyDerivation {
273                message: "Password hash generation failed".to_string(),
274            })
275            .map(|hash| hash.as_bytes().to_vec())
276    }

345
346        let key_bytes =
347            base64::Engine::decode(&base64::engine::general_purpose::STANDARD, key_bytes).map_err(

451    /// Encrypt using Quantum-Resistant encryption
452    async fn encrypt_quantum_resistant(&self, data: &[u8]) -> BearDogResult<EncryptedData> {
453        // Multi-layer encryption approach for quantum resistance
454        // Layer 1: AES-256-GCM (still secure against quantum attacks with larger key sizes)
455        let aes_result = self.encrypt_aes256_gcm(data).await?;
456
457        // Layer 2: ChaCha20-Poly1305 (different mathematical foundation)
458        let chacha_result = self
459            .encrypt_chacha20_poly1305(&aes_result.ciphertext)
460            .await?;
461
462        // Create combined nonce from both layers
463        let mut combined_nonce = Vec::new();
464        combined_nonce.extend_from_slice(&aes_result.nonce);
465        combined_nonce.extend_from_slice(&chacha_result.nonce);
466
467        // Add metadata about the layered approach
468        let mut metadata = HashMap::new();
469        metadata.insert("layers".into(), "2".into());
470        metadata.insert("layer1".into(), "AES-256-GCM".into());
471        metadata.insert("layer2".into(), "ChaCha20-Poly1305".into());
472        metadata.insert("quantum_resistant".into(), "true".into());
473
474        Ok(EncryptedData {
475            algorithm: EncryptionAlgorithm::QuantumResistant,
476            ciphertext: chacha_result.ciphertext,
477            nonce: combined_nonce,
478            tag: None,
479            metadata,
480            key_id: None,
481        })
482    }

484    /// Decrypt using Quantum-Resistant encryption
485    async fn decrypt_quantum_resistant(
486        &self,
487        encrypted_data: &EncryptedData,
488    ) -> BearDogResult<Vec<u8>> {
489        // Verify this is a valid quantum-resistant encrypted data
490        if !encrypted_data.metadata.contains_key("layers")
491            || encrypted_data.metadata.get("layers") != Some(&"2".to_string())
492        {
493            return Err(BearDogError::Encryption {
494                operation: "quantum_decryption".to_string(),
495                message: "Invalid quantum-resistant encrypted data format".to_string(),
496            });
497        }
498
499        // Extract nonces for both layers
500        if encrypted_data.nonce.len() != 24 {
501            // 12 bytes for AES + 12 bytes for ChaCha20
502            return Err(BearDogError::Encryption {
503                operation: "quantum_decryption".to_string(),
504                message: "Invalid nonce length for quantum-resistant decryption".to_string(),
505            });
506        }
507
508        let (aes_nonce, chacha_nonce) = encrypted_data.nonce.split_at(12);
509
510        // Layer 2: Decrypt ChaCha20-Poly1305 first (reverse order)
511        let chacha_encrypted = EncryptedData {
512            algorithm: EncryptionAlgorithm::ChaCha20Poly1305,
513            ciphertext: encrypted_data.ciphertext.clone(),
514            nonce: chacha_nonce.to_vec(),
515            tag: None,
516            metadata: HashMap::new(),
517            key_id: None,
518        };
519
520        let chacha_result = self.decrypt_chacha20_poly1305(&chacha_encrypted).await?;
521
522        // Layer 1: Decrypt AES-256-GCM
523        let aes_encrypted = EncryptedData {
524            algorithm: EncryptionAlgorithm::Aes256Gcm,
525            ciphertext: chacha_result,
526            nonce: aes_nonce.to_vec(),
527            tag: None,
528            metadata: HashMap::new(),
529            key_id: None,
530        };
531
532        self.decrypt_aes256_gcm(&aes_encrypted).await
533    }

565    /// Hash a password with Argon2
566    pub fn hash_password(&self, password: &str) -> BearDogResult<Vec<u8>> {
567        use ring::rand::SecureRandom;
568
569        let mut salt = [0u8; 32];
570        self.rng
571            .fill(&mut salt)
572            .map_err(|e| BearDogError::Encryption {
573                operation: "hash_password".to_string(),
574                message: format!("Salt generation failed: {e}"),
575            })?;
576
577        let salt_string = SaltString::encode_b64(&salt).map_err(|e| BearDogError::Encryption {
578            operation: "hash_password".to_string(),
579            message: format!("Salt encoding failed: {e}"),
580        })?;
581
582        let password_hash = Argon2::default()
583            .hash_password(password.as_bytes(), &salt_string)
584            .map_err(|e| BearDogError::Encryption {
585                operation: "hash_password".to_string(),
586                message: format!("Password hashing failed: {e}"),
587            })?;
588
589        match password_hash.hash {
590            Some(hash) => Ok(hash.as_bytes().to_vec()),
591            None => Err(BearDogError::Encryption {
592                operation: "hash_password".to_string(),
593                message: "Password hash generation failed".to_string(),
594            }),
595        }
596    }

841    async fn test_concurrent_encryption() {
842        let config = create_test_config();
843        let engine = std::sync::Arc::new(EncryptionEngine::new(config).await.unwrap());
844
845        let mut handles = vec![];
846
847        for i in 0..10 {
848            let engine_clone = engine.clone();
849            let handle = tokio::spawn(async move {
850                let plaintext = format!("test message {i}");
851                let encrypted = engine_clone
852                    .encrypt(plaintext.as_bytes(), Some(EncryptionAlgorithm::Aes256Gcm))
853                    .await
854                    .unwrap();
855                let decrypted = engine_clone.decrypt(&encrypted).await.unwrap();
856                assert_eq!(decrypted, plaintext.as_bytes());
857            });
858            handles.push(handle);
859        }
860
861        for handle in handles {
862            handle.await.unwrap();
863        }
864    }
