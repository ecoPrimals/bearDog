52/// Bulk data handler with zero-copy optimization
53pub async fn bulk_data_handler_zero_copy(
54    State(context): State<Arc<ZeroCopyHandlerContext>>,
55    Json(request): Json<BulkDataRequest>,
56) -> BearDogResult<Response> {
57    trace!(
58        "Processing bulk data request with {} items",
59        request.items.len()
60    );
61
62    // Process each item according to the requested processing options
63    let processed_data: Vec<BulkDataItem> = request
64        .items
65        .into_iter()
66        .enumerate()
67        .map(|(index, item)| {
68            // Apply processing based on options
69            let processed_content = if let Some(ref options) = request.processing_options {
70                process_data_item(&item, options)
71            } else {
72                // Default processing: basic validation and formatting
73                process_data_item_default(&item)
74            };
75
76            BulkDataItem {
77                id: format!("processed_{index}"),
78                data: processed_content,
79                processed_at: chrono::Utc::now().to_rfc3339(),
80            }
81        })
82        .collect();
83
84    // Use streaming response for large datasets to maintain zero-copy performance
85    if processed_data.len() > 100 {
86        context
87            .response_builder
88            .streaming_response(processed_data, StatusCode::OK)
89            .await
90    } else {
91        context
92            .response_builder
93            .json_response(&processed_data, StatusCode::OK)
94            .await
95    }
96}

98/// Process a data item with specific options
99fn process_data_item(item: &str, options: &HashMap<String, String>) -> String {
100    let mut result = item.to_string();
101
102    // Apply security-focused processing based on options
103    if let Some(operation) = options.get("operation") {
104        match operation.as_str() {
105            "hash" => {
106                // Compute SHA-256 hash for security validation
107                let mut hasher = Sha256::new();
108                hasher.update(item.as_bytes());
109                result = format!("sha256:{:x}", hasher.finalize());
110            }
111            "validate" => {
112                // Validate data format and return status
113                result = if is_valid_data_format(item) {
114                    format!("valid:{item}")
115                } else {
116                    format!("invalid:{item}")
117                };
118            }
119            "sanitize" => {
120                // Sanitize potentially dangerous content
121                result = sanitize_data_content(item);
122            }
123            "encrypt" => {
124                // Basic encryption placeholder - in production would use proper encryption
125                result = format!(
126                    "encrypted:{}",
127                    base64::engine::general_purpose::STANDARD.encode(item)
128                );
129            }
130            _ => {
131                // Unknown operation, return as-is with warning
132                tracing::warn!("Unknown processing operation: {}", operation);
133                result = format!("unprocessed:{item}");
134            }
135        }
136    }
137
138    // Apply additional transformations based on other options
139    if let Some(format) = options.get("format") {
140        match format.as_str() {
141            "uppercase" => result = result.to_uppercase(),
142            "lowercase" => result = result.to_lowercase(),
143            "base64" => result = base64::engine::general_purpose::STANDARD.encode(result),
144            _ => {} // Keep as-is for unknown formats
145        }
146    }
147
148    result
149}

151/// Default processing for items without specific options
152fn process_data_item_default(item: &str) -> String {
153    // Default: validate and sanitize the input
154    if is_valid_data_format(item) {
155        sanitize_data_content(item)
156    } else {
157        format!(
158            "invalid_format:{}",
159            item.chars().take(50).collect::<String>()
160        )
161    }
162}

164/// Validate data format (basic security check)
165fn is_valid_data_format(data: &str) -> bool {
166    // Basic validation: not empty, reasonable length, no null bytes
167    !data.is_empty() && data.len() <= 10000 && !data.contains('\0') && data.is_ascii()
168}
169
170/// Sanitize potentially dangerous content
171fn sanitize_data_content(data: &str) -> String {
172    // Remove potentially dangerous characters and sequences
173    data.chars()
174        .filter(|c| c.is_ascii_alphanumeric() || " .-_@".contains(*c))
175        .collect::<String>()
176        .trim()
177        .to_string()
178}

180/// Create default handler context for zero-copy operations
181impl Default for ZeroCopyHandlerContext {
182    fn default() -> Self {
183        let buffer_pool = Arc::new(HttpBufferPool::new());
184        let serializer = Arc::new(ZeroCopyJsonSerializer::new(buffer_pool.clone()));
185        let request_parser = Arc::new(ZeroCopyRequestParser::new(
186            buffer_pool.clone(),
187            serializer.clone(),
188        ));
189        let response_builder = Arc::new(ZeroCopyResponseBuilder::new());
190
191        Self {
192            response_builder,
193            request_parser,
194            buffer_pool,
195            stats: super::zero_copy::types::ZeroCopyHandlerStats::default(),
196        }
197    }
198}
