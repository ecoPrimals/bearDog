1//! Authentication & Authorization API Module
2//!
3//! Comprehensive REST API for BearDog's authentication, authorization, and user
4//! management capabilities supporting multiple authentication methods, MFA, and RBAC.
5
6use super::*;
7use axum::{
8    routing::{delete, get, post, put},
9    Router,
10};
11
12// Re-export types and handlers from sub-modules
13pub use authentication::*;
14pub use authorization::*;
15pub use mfa::*;
16pub use models::*;
17pub use users::*;
18
19// Sub-modules with focused responsibilities
20pub mod authentication; // Core auth: login, logout, tokens, sessions
21pub mod authorization; // Roles, permissions, API keys
22pub mod improved_authentication; // Improved auth with rich context patterns
23pub mod mfa; // Multi-factor authentication
24pub mod models; // All request/response models

27/// Create authentication API routes
28pub fn create_routes() -> Router<AppState> {
29    Router::new()
30        // Authentication routes
31        .route("/login", post(authenticate_user))
32        .route("/logout", post(logout_user))
33        .route("/refresh", post(refresh_token))
34        .route("/validate", post(validate_token))
35        .route("/session", get(get_current_session))
36        .route("/session", delete(invalidate_session))
37        // Multi-Factor Authentication routes
38        .route("/mfa/setup", post(setup_mfa))
39        .route("/mfa/verify", post(verify_mfa))
40        .route("/mfa/backup-codes", get(get_backup_codes))
41        .route("/mfa/backup-codes", post(generate_backup_codes))
42        .route("/mfa/status", get(get_mfa_status))
43        // User Management routes
44        .route("/users", get(list_users))
45        .route("/users", post(create_user))
46        .route("/users/:user_id", get(get_user))
47        .route("/users/:user_id", put(update_user))
48        .route("/users/:user_id", delete(delete_user))
49        .route("/users/:user_id/activate", post(activate_user))
50        .route("/users/:user_id/deactivate", post(deactivate_user))
51        .route("/users/:user_id/reset-password", post(reset_password))
52        // Role-Based Access Control routes
53        .route("/roles", get(list_roles))
54        .route("/roles", post(create_role))
55        .route("/roles/:role_id", get(get_role))
56        .route("/roles/:role_id", put(update_role))
57        .route("/roles/:role_id", delete(delete_role))
58        .route("/roles/:role_id/permissions", get(get_role_permissions))
59        .route("/roles/:role_id/permissions", put(update_role_permissions))
60        // User-Role Assignment routes
61        .route("/users/:user_id/roles", get(get_user_roles))
62        .route("/users/:user_id/roles", post(assign_user_roles))
63        .route("/users/:user_id/roles/:role_id", delete(remove_user_role))
64        // Permissions & Authorization routes
65        .route("/permissions", get(list_permissions))
66        .route("/permissions/check", post(check_permission))
67        .route("/permissions/bulk-check", post(bulk_check_permissions))
68        .route("/users/:user_id/permissions", get(get_user_permissions))
69        // API Keys & Service Accounts routes
70        .route("/api-keys", get(list_api_keys))
71        .route("/api-keys", post(create_api_key))
72        .route("/api-keys/:key_id", get(get_api_key))
73        .route("/api-keys/:key_id", delete(revoke_api_key))
74        .route("/api-keys/:key_id/rotate", post(rotate_api_key))
75        // Audit & Security routes
76        .route("/audit/login-attempts", get(get_login_attempts))
77        .route("/audit/security-events", get(get_security_events))
78        .route("/audit/user-activity", get(get_user_activity))
79        .route("/security/password-policy", get(get_password_policy))
80        .route("/security/password-policy", put(update_password_policy))
81        .route("/security/lockout-policy", get(get_lockout_policy))
82        .route("/security/lockout-policy", put(update_lockout_policy))
83}
