18/// List all users with filtering and pagination
19pub async fn list_users(
20    State(_state): State<AppState>,
21    Query(params): Query<UserListQuery>,
22) -> Result<Json<ApiResponse<UserListResponse>>, StatusCode> {
23    let start_time = Instant::now();
24    let request_id = uuid::Uuid::new_v4().to_string();
25
26    info!("👥 Listing users with filters: {:?}", params);
27
28    let users = vec![
29        UserSummary {
30            user_id: "user_12345".to_string(),
31            username: "admin".to_string(),
32            display_name: "System Administrator".to_string(),
33            email: "admin@beardog.com".to_string(),
34            status: "active".to_string(),
35            roles: vec!["admin".to_string(), "user".to_string()],
36            created_at: chrono::Utc::now().to_rfc3339(),
37            last_login: Some(chrono::Utc::now().to_rfc3339()),
38            mfa_enabled: true,
39        },
40        UserSummary {
41            user_id: "user_67890".to_string(),
42            username: "john.doe".to_string(),
43            display_name: "John Doe".to_string(),
44            email: "john.doe@example.com".to_string(),
45            status: "active".to_string(),
46            roles: vec!["user".to_string()],
47            created_at: (chrono::Utc::now() - chrono::Duration::days(30)).to_rfc3339(),
48            last_login: Some((chrono::Utc::now() - chrono::Duration::days(1)).to_rfc3339()),
49            mfa_enabled: false,
50        },
51    ];
52
53    let response = UserListResponse {
54        users,
55        total_count: 2,
56        page: params.page.unwrap_or(1),
57        per_page: params.per_page.unwrap_or(20),
58        total_pages: 1,
59        has_more: false,
60        filters_applied: UserListFilters {
61            status: params.status,
62            role: params.role,
63            search: params.search,
64        },
65    };
66
67    let processing_time = start_time.elapsed().as_millis() as u64;
68    Ok(Json(success_response(
69        response,
70        request_id,
71        processing_time,
72        true,
73    )))
74}

76/// Create new user account
77pub async fn create_user(
78    State(_state): State<AppState>,
79    Json(request): Json<CreateUserRequest>,
80) -> Result<Json<ApiResponse<CreateUserResponse>>, StatusCode> {
81    let start_time = Instant::now();
82    let request_id = uuid::Uuid::new_v4().to_string();
83
84    info!("➕ Creating new user: {}", request.username);
85
86    let user_id = uuid::Uuid::new_v4().to_string();
87    let response = CreateUserResponse {
88        user_id: user_id.clone(),
89        username: request.username,
90        email: request.email,
91        status: "pending_verification".to_string(),
92        created_at: chrono::Utc::now().to_rfc3339(),
93        verification_required: true,
94        verification_email_sent: true,
95        default_roles: vec!["user".to_string()],
96        password_reset_required: request.temporary_password.is_some(),
97    };
98
99    let processing_time = start_time.elapsed().as_millis() as u64;
100    Ok(Json(success_response(
101        response,
102        request_id,
103        processing_time,
104        false,
105    )))
106}
