1//! Individual Sovereignty API Routes
2//!
3//! Routes for empowering individuals to control their compute, data, and identity
4//! through consent-based peer-to-peer interactions.
5
6use crate::api::AppState;
7use axum::{
8    routing::{delete, get, post},
9    Router,
10};
11
12use super::handlers::*;
13
14/// Create individual sovereignty API routes
15///
16/// These routes embody BearDog's core mission: empowering individuals
17/// to control their own resources and share them with friends through
18/// explicit consent mechanisms.
19///
20/// # Route Structure
21/// - `/sharing` - Peer-to-peer resource sharing with consent
22/// - `/recovery` - Friend-based account recovery workflows  
23/// - `/identity` - Self-sovereign identity management
24/// - `/privacy` - Anti-surveillance privacy protections
25/// - `/consent` - Consent management for all interactions
26pub fn create_sovereignty_routes() -> Router<AppState> {
27    Router::new()
28        // 🤝 PEER-TO-PEER RESOURCE SHARING
29        .route("/sharing/request", post(request_resource_sharing))
30        .route("/sharing/offers", get(list_sharing_offers))
31        .route("/sharing/offers/:offer_id", get(get_sharing_offer))
32        .route(
33            "/sharing/offers/:offer_id/accept",
34            post(accept_sharing_offer),
35        )
36        .route(
37            "/sharing/offers/:offer_id/decline",
38            post(decline_sharing_offer),
39        )
40        .route("/sharing/active", get(list_active_shares))
41        .route("/sharing/active/:share_id", delete(revoke_resource_share))
42        // 🆘 FRIEND-BASED RECOVERY
43        .route("/recovery/request", post(request_friend_recovery))
44        .route("/recovery/requests", get(list_recovery_requests))
45        .route(
46            "/recovery/requests/:request_id/assist",
47            post(assist_with_recovery),
48        )
49        .route(
50            "/recovery/shards/distribute",
51            post(distribute_recovery_shards),
52        )
53        .route("/recovery/shards/collect", post(collect_recovery_shards))
54        // 🎭 SELF-SOVEREIGN IDENTITY
55        .route("/identity/keys", get(list_my_identity_keys))
56        .route("/identity/keys", post(generate_identity_key))
57        .route("/identity/keys/:key_id", delete(revoke_identity_key))
58        .route("/identity/verify", post(verify_identity_claim))
59        .route("/identity/attest", post(create_identity_attestation))
60        // 🔒 ANTI-SURVEILLANCE PRIVACY
61        .route("/privacy/status", get(privacy_status_check))
62        .route("/privacy/audit", get(privacy_audit_trail))
63        .route("/privacy/purge", post(privacy_data_purge))
64        .route("/privacy/anonymize", post(anonymize_personal_data))
65        // ✋ CONSENT MANAGEMENT
66        .route("/consent/active", get(list_active_consents))
67        .route("/consent/:consent_id", get(get_consent_details))
68        .route("/consent/:consent_id/revoke", post(revoke_consent))
69        .route("/consent/grants", get(list_consent_grants))
70        .route("/consent/requests", get(list_consent_requests))
71}
