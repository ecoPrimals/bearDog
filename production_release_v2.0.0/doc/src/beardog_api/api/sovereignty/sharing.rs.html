1//! Peer-to-Peer Resource Sharing
2//!
3//! **Empowering individuals to share resources with friends through explicit consent**
4//!
5//! This module implements Beardog's core peer-to-peer sharing capabilities:
6//! - Friend-to-friend compute sharing (CPU, GPU, memory)
7//! - Community storage pools with encryption
8//! - Consent-based resource lending
9//! - Usage tracking and fair allocation
10//! - Automated friend network discovery
11//!
12//! ## Core Principles
13//! - **Individual Control**: Only you decide what to share and with whom
14//! - **Explicit Consent**: Every resource share requires explicit friend consent
15//! - **Privacy First**: All sharing is encrypted and logged locally
16//! - **Fair Usage**: Built-in mechanisms to prevent abuse
17//! - **Human Dignity**: Technology serves the individual, not corporations

115    /// Create a resource sharing offer to friends
116    pub async fn create_sharing_offer(
117        &self,
118        from_node_id: String,
119        from_display_name: String,
120        resource_type: ResourceType,
121        resource_amount: ResourceAmount,
122        personal_message: String,
123        expires_hours: Option<i64>,
124    ) -> Result<SharingOffer, Box<dyn std::error::Error + Send + Sync>> {
125        info!(
126            "🎁 Creating resource sharing offer from {}",
127            from_display_name
128        );
129
130        let offer_id = Uuid::new_v4().to_string();
131        let now = Utc::now();
132        let expires_at = expires_hours.map(|h| now + Duration::hours(h));
133
134        // Create temporary offer for workflow
135        let temp_offer = SharingOffer {
136            offer_id: offer_id.clone(),
137            from_node_id: from_node_id.clone(),
138            from_display_name: from_display_name.clone(),
139            resource_type: resource_type.clone(),
140            resource_amount: resource_amount.clone(),
141            personal_message: personal_message.clone(),
142            created_at: now.to_rfc3339(),
143            expires_at: expires_at.map(|dt| dt.to_rfc3339()),
144            terms: self.generate_sharing_terms(&resource_type),
145        };
146
147        // Create consent workflow for friend approval
148        let workflow_id = self.create_consent_workflow(&temp_offer).await?;
149
150        let offer_internal = SharingOfferInternal {
151            offer_id: offer_id.clone(),
152            from_node_id: from_node_id.clone(),
153            from_display_name: from_display_name.clone(),
154            resource_type: resource_type.clone(),
155            resource_amount: resource_amount.clone(),
156            personal_message: personal_message.clone(),
157            created_at: now,
158            expires_at,
159            terms: self.generate_sharing_terms(&resource_type),
160            consent_required: true,
161            friends_notified: Vec::new(),
162            workflow_id: Some(workflow_id),
163        };
164
165        // Store the offer
166        {
167            let mut offers = self.active_offers.write().await;
168            offers.insert(offer_id.clone(), offer_internal);
169        }
170
171        // Notify friends about the offer
172        self.notify_friends_of_offer(&offer_id).await?;
173
174        // Convert to public format (clone resource_type to avoid move issues)
175        let terms = self.generate_sharing_terms(&resource_type);
176        Ok(SharingOffer {
177            offer_id,
178            from_node_id,
179            from_display_name,
180            resource_type,
181            resource_amount,
182            personal_message,
183            created_at: now.to_rfc3339(),
184            expires_at: expires_at.map(|dt| dt.to_rfc3339()),
185            terms,
186        })
187    }

189    /// Accept a resource sharing offer from a friend
190    pub async fn accept_sharing_offer(
191        &self,
192        offer_id: &str,
193        _accepting_node_id: String,
194        _accepting_display_name: String,
195    ) -> Result<ActiveShare, Box<dyn std::error::Error + Send + Sync>> {
196        info!("✅ Accepting resource sharing offer: {}", offer_id);
197
198        // Find the offer
199        let offer = {
200            let offers = self.active_offers.read().await;
201            offers
202                .get(offer_id)
203                .cloned()
204                .ok_or("Sharing offer not found")?
205        };
206
207        // Check if offer is still valid
208        if let Some(expires_at) = offer.expires_at {
209            if Utc::now() > expires_at {
210                return Err("Sharing offer has expired".into());
211            }
212        }
213
214        // Create active share
215        let share_id = Uuid::new_v4().to_string();
216        let consent_record_id = Uuid::new_v4().to_string();
217
218        let active_share_internal = ActiveShareInternal {
219            share_id: share_id.clone(),
220            friend_node_id: offer.from_node_id.clone(),
221            friend_display_name: offer.from_display_name.clone(),
222            resource_type: offer.resource_type.clone(),
223            usage_stats: ResourceUsageStatsInternal {
224                allocated: offer.resource_amount.maximum,
225                current_usage: 0,
226                peak_usage: 0,
227                usage_history: Vec::new(),
228                fair_share_score: 1.0,
229            },
230            started_at: Utc::now(),
231            expires_at: offer.expires_at,
232            revocable: true,
233            consent_record_id: consent_record_id.clone(),
234        };
235
236        // Store the active share
237        {
238            let mut shares = self.active_shares.write().await;
239            shares.insert(share_id.clone(), active_share_internal);
240        }
241
242        // Remove the offer since it's been accepted
243        {
244            let mut offers = self.active_offers.write().await;
245            offers.remove(offer_id);
246        }
247
248        // Update friend network with successful sharing
249        self.update_friend_trust_score(&offer.from_node_id, 0.1)
250            .await?;
251
252        info!("🎉 Resource sharing activated: {}", share_id);
253
254        // Convert to public format
255        Ok(ActiveShare {
256            share_id,
257            friend_node_id: offer.from_node_id,
258            friend_display_name: offer.from_display_name,
259            resource_type: offer.resource_type,
260            usage_stats: ResourceUsageStats {
261                allocated: offer.resource_amount.maximum,
262                current_usage: 0,
263                peak_usage: 0,
264                usage_history: Vec::new(),
265            },
266            started_at: Utc::now().to_rfc3339(),
267            expires_at: offer.expires_at.map(|dt| dt.to_rfc3339()),
268            revocable: true,
269        })
270    }

386    /// Update usage statistics for an active share
387    pub async fn update_usage_stats(
388        &self,
389        share_id: &str,
390        current_usage: u64,
391    ) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
392        debug!("📈 Updating usage statistics for share: {}", share_id);
393
394        let mut shares = self.active_shares.write().await;
395        if let Some(share) = shares.get_mut(share_id) {
396            share.usage_stats.current_usage = current_usage;
397
398            if current_usage > share.usage_stats.peak_usage {
399                share.usage_stats.peak_usage = current_usage;
400            }
401
402            // Add data point to history
403            share
404                .usage_stats
405                .usage_history
406                .push(UsageDataPointInternal {
407                    timestamp: Utc::now(),
408                    usage_amount: current_usage,
409                    efficiency_score: self
410                        .calculate_efficiency_score(current_usage, share.usage_stats.allocated),
411                });
412
413            // Update fair share score
414            share.usage_stats.fair_share_score =
415                self.calculate_fair_share_score(&share.usage_stats);
416
417            Ok(())
418        } else {
419            Err("Resource share not found".into())
420        }
421    }

447    /// Generate standard sharing terms based on resource type
448    fn generate_sharing_terms(&self, resource_type: &ResourceType) -> Vec<String> {
449        let mut terms = vec![
450            "Explicit consent required for all usage".to_string(),
451            "Resources can be revoked at any time".to_string(),
452            "Usage will be monitored for fair allocation".to_string(),
453            "All data processed remains encrypted".to_string(),
454        ];
455
456        match resource_type {
457            ResourceType::Compute { .. } => {
458                terms.push("Compute resources subject to availability".to_string());
459                terms.push("No cryptocurrency mining allowed".to_string());
460            }
461            ResourceType::Storage { encrypted, .. } => {
462                if *encrypted {
463                    terms.push("All stored data is encrypted at rest".to_string());
464                }
465                terms.push("No illegal content allowed".to_string());
466            }
467            ResourceType::Network { .. } => {
468                terms.push("Network usage subject to bandwidth limits".to_string());
469                terms.push("No illegal network activities allowed".to_string());
470            }
471        }
472
473        terms
474    }

495    /// Update a friend's trust score based on sharing interactions
496    async fn update_friend_trust_score(
497        &self,
498        friend_id: &str,
499        delta: f64,
500    ) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
501        let mut network = self.friend_network.write().await;
502        if let Some(friend) = network.get_mut(friend_id) {
503            friend.trust_score = (friend.trust_score + delta).clamp(0.0, 5.0);
504            friend.last_seen = Utc::now();
505        }
506        Ok(())
507    }

509    /// Calculate efficiency score for resource usage
510    fn calculate_efficiency_score(&self, used: u64, allocated: u64) -> f64 {
511        if allocated == 0 {
512            return 0.0;
513        }
514
515        let usage_ratio = used as f64 / allocated as f64;
516
517        // Efficiency peaks at around 70-80% usage
518        if usage_ratio <= 0.8 {
519            usage_ratio / 0.8
520        } else {
521            1.0 - ((usage_ratio - 0.8) / 0.2) * 0.5
522        }
523    }
524
525    /// Calculate fair share score based on usage patterns
526    fn calculate_fair_share_score(&self, stats: &ResourceUsageStatsInternal) -> f64 {
527        if stats.usage_history.is_empty() {
528            return 1.0;
529        }
530
531        let avg_efficiency: f64 = stats
532            .usage_history
533            .iter()
534            .map(|dp| dp.efficiency_score)
535            .sum::<f64>()
536            / stats.usage_history.len() as f64;
537
538        // Penalty for consistently low usage
539        let usage_ratio = stats.current_usage as f64 / stats.allocated as f64;
540        if usage_ratio < 0.1 && stats.usage_history.len() > 10 {
541            avg_efficiency * 0.7
542        } else {
543            avg_efficiency
544        }
545    }
