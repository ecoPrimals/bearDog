56impl Default for ApiServerConfig {

58        Self {
59            bind_address: std::env::var("BEARDOG_API_BIND_ADDRESS").unwrap_or_else(|_| {
60                format!("{}:{}", "0.0.0.0", beardog_config::constants::ports::API)

62            request_timeout_seconds: std::env::var("BEARDOG_API_REQUEST_TIMEOUT")

67                .ok()
68                .and_then(|v| v.parse().ok())
69                .unwrap_or(16 * 1024 * 1024), // 16MB
70            compression_enabled: std::env::var("BEARDOG_API_COMPRESSION_ENABLED")
71                .ok()
72                .and_then(|v| v.parse().ok())
73                .unwrap_or(true),
74            cors_enabled: std::env::var("BEARDOG_API_CORS_ENABLED")
75                .ok()
76                .and_then(|v| v.parse().ok())
77                .unwrap_or(true),
78            rate_limiting_enabled: std::env::var("BEARDOG_API_RATE_LIMITING_ENABLED")

80                .and_then(|v| v.parse().ok())

105impl BearDogApiServer {
106    /// Create new API server with comprehensive middleware

108        Self::new_with_config(core, ApiServerConfig::default()).await

111    /// Create new API server with custom configuration
112    pub async fn new_with_config(
113        core: Arc<dyn std::any::Any + Send + Sync>,
114        config: ApiServerConfig,
115    ) -> BearDogResult<Self> {
116        // Initialize cache provider
117        let cache = Arc::new(crate::api::cache::CacheProviderType::InMemory(
118            crate::api::cache::InMemoryCache::new(),
119        ));
120
121        // Initialize rate limiter
122        let rate_limiter = Arc::new(crate::api::rate_limiting::RateLimiterType::new());
123
124        Ok(Self {
125            core,
126            cache,
127            rate_limiter,
128            config,
129        })
130    }

132    /// Create comprehensive application router
133    pub fn create_router(&self) -> Router {
134        let app_state = AppState {
135            core: self.core.clone(),
136            cache: self.cache.clone(),
137            rate_limiter: self.rate_limiter.clone(),
138            config: self.config.clone(),
139            zero_copy_context: Arc::new(ZeroCopyHandlerContext::new()),
140        };
141
142        // Return the configured app
143        Router::new()
144            // Use zero-copy optimized health endpoint for better performance
145            .route("/health", get(zero_copy_health_wrapper))
146            .route("/info", get(self::handlers::server_info))
147            // Performance monitoring endpoint for zero-copy statistics
148            .route(
149                "/api/v1/performance/zero-copy-stats",
150                get(zero_copy_stats_handler),
151            )
152            // API v1 routes
153            .nest("/v1", self.create_v1_routes())
154            // Apply basic middleware layers
155            .layer(middleware::from_fn(request_id_middleware))
156            .layer(middleware::from_fn_with_state(
157                app_state.clone(),
158                rate_limiting_middleware,
159            ))
160            .layer(middleware::from_fn_with_state(
161                app_state.clone(),
162                performance_middleware,
163            ))
164            .with_state(app_state)
165    }

187    }
188
189    /// Start the API server
190    pub async fn start(&self, bind_address: &str) -> BearDogResult<()> {
191        info!(
192            "🚀 Starting high-performance BearDog API server on {}",
193            bind_address
194        );
195
196        let listener = TcpListener::bind(bind_address).await.map_err(|e| {
197            BearDogError::internal(format!("Failed to bind to {bind_address}: {e}"))
198        })?;
199
200        let app = self.create_router();
201
202        info!("✅ BearDog API server ready - AI-first design with full BearDog capabilities");
203        info!(
204            "📊 Performance features: caching={}, rate_limiting={}, compression={}",
205            self.config.caching_enabled,
206            self.config.rate_limiting_enabled,
207            self.config.compression_enabled
208        );
209
210        axum::serve(listener, app)
211            .with_graceful_shutdown(shutdown_signal())
212            .await
213            .map_err(|e| BearDogError::internal(format!("Server error: {e}")))?;
214
215        info!("🛑 BearDog API server shutdown completed");
216        Ok(())
217    }

298/// Request performance middleware
299async fn performance_middleware(
300    State(_state): State<AppState>,
301    mut request: Request,
302    next: Next,
303) -> Response {
304    let start = Instant::now();
305
306    // Add start time to request extensions
307    request.extensions_mut().insert(start);
308
309    let mut response = next.run(request).await;
310
311    let duration = start.elapsed();
312
313    // Add performance headers to the response
314    if let Ok(header_value) =
315        HeaderValue::from_str(&format!("{:.2}", duration.as_secs_f64() * 1000.0))
316    {
317        response
318            .headers_mut()
319            .insert("X-Response-Time-Ms", header_value);
320    }
321
322    // Log performance for monitoring
323    tracing::debug!("Request took {:.2}ms", duration.as_secs_f64() * 1000.0);
324
325    response
326}

328/// Rate limiting middleware
329async fn rate_limiting_middleware(
330    State(state): State<AppState>,
331    request: Request,
332    next: Next,
333) -> Result<Response, StatusCode> {
334    if !state.config.rate_limiting_enabled {
335        return Ok(next.run(request).await);
336    }
337
338    // Extract client identifier (IP or API key)
339    let client_id = extract_client_id(&request);
340
341    // Check rate limit
342    if !state.rate_limiter.check_limit(&client_id).await {
343        warn!("Rate limit exceeded for client: {}", client_id);
344        return Err(StatusCode::TOO_MANY_REQUESTS);
345    }
346
347    Ok(next.run(request).await)
348}

350/// Request ID middleware
351async fn request_id_middleware(mut request: Request, next: Next) -> Response {
352    let request_id = uuid::Uuid::new_v4().to_string();
353
354    // Add request ID to headers
355    if let Ok(header_value) = request_id.parse() {
356        request.headers_mut().insert("X-Request-ID", header_value);
357    }
358
359    // Store in extensions for handlers
360    request.extensions_mut().insert(request_id.clone());
361
362    let mut response = next.run(request).await;
363
364    // Add request ID to response headers
365    if let Ok(header_value) = request_id.parse() {
366        response.headers_mut().insert("X-Request-ID", header_value);
367    }
368
369    response
370}

372/// Extract client identifier for rate limiting
373fn extract_client_id(request: &Request) -> String {
374    // Try API key first
375    if let Some(api_key) = request.headers().get("X-API-Key") {
376        if let Ok(key_str) = api_key.to_str() {
377            return format!("api_key:{key_str}");
378        }
379    }
380
381    // Fall back to IP address
382    if let Some(forwarded) = request.headers().get("X-Forwarded-For") {
383        if let Ok(ip_str) = forwarded.to_str() {
384            return format!(
385                "ip:{}",
386                ip_str.split(',').next().unwrap_or("unknown").trim()
387            );
388        }
389    }
390
391    // Default fallback
392    "unknown".to_string()
393}

395/// Graceful shutdown signal
396async fn shutdown_signal() {
397    let ctrl_c = async {
398        match tokio::signal::ctrl_c().await {
399            Ok(()) => {}
400            Err(e) => {
401                error!("Failed to install Ctrl+C handler: {}", e);
402                // Continue without ctrl+c handling rather than panic
403            }
404        }
405    };
406
407    #[cfg(unix)]
408    let terminate = async {
409        match tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate()) {
410            Ok(mut signal) => {
411                signal.recv().await;
412            }
413            Err(e) => {
414                error!("Failed to install signal handler: {}", e);
415                // Continue without terminate signal handling rather than panic
416            }
417        }
418    };
419
420    #[cfg(not(unix))]
421    let terminate = std::future::pending::<()>();
422
423    tokio::select! {
424        _ = ctrl_c => {
425            info!("Received Ctrl+C signal, initiating graceful shutdown...");
426        },
427        _ = terminate => {
428            info!("Received terminate signal, initiating graceful shutdown...");
429        },
430    }
431}

439    /// Health check endpoint
440    pub async fn health_check(State(state): State<AppState>) -> Result<Json<Value>, StatusCode> {
441        // Attempt to downcast the core to BearDogCore for proper health checking
442        use beardog_core::core::BearDogCore;
443
444        match state.core.downcast_ref::<BearDogCore>() {
445            Some(core) => {
446                // Perform actual health check using the BearDog core
447                match core.health_check().await {
448                    Ok(health_check) => {
449                        // Return detailed health information
450                        Ok(Json(json!({
451                            "status": "healthy", // Simplified for canonical modernization
452                            "timestamp": chrono::Utc::now(),
453                            "version": env!("CARGO_PKG_VERSION"),
454                            "api_version": API_VERSION,
455                            "uptime_seconds": health_check.uptime.map(|u| u.num_seconds()).unwrap_or(0),
456                                            "component_name": health_check.component_name,
457                            "metrics": {
458                                                                            "details": health_check.details,
459                                "error_rate_percent": 0.0 // SystemMetrics doesn't have error_rate_percent
460                            }
461                        })))
462                    }
463                    Err(e) => {
464                        tracing::error!("Health check failed: {}", e);
465                        Ok(Json(json!({
466                            "status": "unhealthy",
467                            "timestamp": chrono::Utc::now(),
468                            "version": env!("CARGO_PKG_VERSION"),
469                            "api_version": API_VERSION,
470                            "error": format!("Health check failed: {}", e)
471                        })))
472                    }
473                }
474            }
475            None => {
476                // Fallback to basic health check if core type is not BearDogCore
477                tracing::warn!("Core is not BearDogCore type, using basic health check");
478                Ok(Json(json!({
479                    "status": "healthy",
480                    "timestamp": chrono::Utc::now(),
481                    "version": env!("CARGO_PKG_VERSION"),
482                    "api_version": API_VERSION,
483                    "note": "Basic health check - core type not recognized"
484                })))
485            }
486        }
487    }
