1//! Zero-Copy Response Builder
2//!
3//! Builds HTTP responses with minimal allocations using buffer pooling.
4
5use super::{HttpBufferPool, ZeroCopyJsonSerializer};
6use axum::{
7    body::Body,
8    http::{HeaderValue, StatusCode},
9    response::Response,
10};
11use beardog_errors::BearDogResult;
12use serde::Serialize;
13use std::collections::HashMap;
14use std::sync::{
15    atomic::{AtomicU64, Ordering},
16    Arc,
17};
18use tokio::sync::RwLock;
19
20/// Zero-copy response builder
21pub struct ZeroCopyResponseBuilder {
22    #[allow(dead_code)] // Will be used when zero-copy response optimizations are fully implemented
23    buffer_pool: Arc<HttpBufferPool>,
24    serializer: Arc<ZeroCopyJsonSerializer>,
25    #[allow(dead_code)] // Reserved for future header optimization
26    header_cache: Arc<RwLock<HashMap<String, HeaderValue>>>,
27    stats: ZeroCopyResponseStats,
28}
29
30#[derive(Debug, Default)]
31pub struct ZeroCopyResponseStats {
32    pub responses_built: AtomicU64,
33    pub header_cache_hits: AtomicU64,
34    pub header_cache_misses: AtomicU64,
35    pub streaming_responses: AtomicU64,
36    pub content_bytes_served: AtomicU64,
37}
38
39impl Default for ZeroCopyResponseBuilder {
40    fn default() -> Self {
41        Self::new()
42    }
43}
44
45impl ZeroCopyResponseBuilder {
46    /// Create new zero-copy response builder
47    pub fn new() -> Self {
48        let buffer_pool = Arc::new(HttpBufferPool::new());
49        let serializer = Arc::new(ZeroCopyJsonSerializer::new(buffer_pool.clone()));
50
51        Self {
52            buffer_pool,
53            serializer,
54            header_cache: Arc::new(RwLock::new(HashMap::new())),
55            stats: ZeroCopyResponseStats::default(),
56        }
57    }
58
59    /// Build JSON response with zero-copy optimization
60    pub async fn json_response<T: Serialize>(
61        &self,
62        data: &T,
63        status: StatusCode,
64    ) -> BearDogResult<Response> {
65        self.stats.responses_built.fetch_add(1, Ordering::Relaxed);
66
67        // Serialize using zero-copy JSON serializer
68        let json_bytes = self.serializer.serialize_zero_copy(data).await?;
69
70        self.stats
71            .content_bytes_served
72            .fetch_add(json_bytes.len() as u64, Ordering::Relaxed);
73
74        // Build response with cached headers
75        let response = Response::builder()
76            .status(status)
77            .header("content-type", "application/json")
78            .header("content-length", json_bytes.len())
79            .body(Body::from(json_bytes))
80            .map_err(|e| beardog_errors::BearDogError::Internal {
81                message: format!("Failed to build response: {e}"),
82            })?;
83
84        Ok(response)
85    }
86
87    /// Build streaming response for large datasets
88    pub async fn streaming_response<T: Serialize>(
89        &self,
90        data: Vec<T>,
91        status: StatusCode,
92    ) -> BearDogResult<Response> {
93        self.stats
94            .streaming_responses
95            .fetch_add(1, Ordering::Relaxed);
96
97        // For large datasets, serialize in chunks
98        let json_bytes = self.serializer.serialize_zero_copy(&data).await?;
99
100        self.stats
101            .content_bytes_served
102            .fetch_add(json_bytes.len() as u64, Ordering::Relaxed);
103
104        let response = Response::builder()
105            .status(status)
106            .header("content-type", "application/json")
107            .header("transfer-encoding", "chunked")
108            .body(Body::from(json_bytes))
109            .map_err(|e| beardog_errors::BearDogError::Internal {
110                message: format!("Failed to build streaming response: {e}"),
111            })?;
112
113        Ok(response)
114    }
115
116    /// Build error response
117    pub async fn error_response(
118        &self,
119        error_code: &str,
120        error_message: &str,
121        status: StatusCode,
122    ) -> BearDogResult<Response> {
123        let error_data = serde_json::json!({
124            "error": {
125                "code": error_code,
126                "message": error_message,
127                "timestamp": chrono::Utc::now().to_rfc3339()
128            }
129        });
130
131        self.json_response(&error_data, status).await
132    }
133
134    /// Get response builder statistics
135    pub fn get_stats(&self) -> &ZeroCopyResponseStats {
136        &self.stats
137    }
138}
