1//! Ecosystem RPC Integration - Dynamic Primal Discovery
2//!
3//! Rust RPC integration that enables sovereign participation in the
4//! ecoPrimals ecosystem while maintaining complete independence and human dignity.
5//!
6//! ## Core Principles
7//! - **Sovereignty First**: Primal remains fully functional standalone
8//! - **Dynamic Discovery**: Discover other primals through capabilities, not hardcoded names
9//! - **Capability-Based**: Discovery and integration based on what services can do
10//! - **Human-Centered**: All RPC operations serve individual human dignity
11

31/// Create the RPC router with all ecosystem integration endpoints
32pub fn create_rpc_router() -> Router<AppState> {
33    Router::new()
34        // Registration & Discovery
35        .route("/rpc/register", post(register_with_ecosystem))
36        .route("/rpc/status", get(get_primal_status))
37        .route("/rpc/capabilities", get(get_self_primal_capabilities))
38        .route("/rpc/discover", post(discover_capabilities))
39        // Collaboration Services
40        .route("/rpc/services/security", post(provide_security_service))
41        .route(
42            "/rpc/services/sovereignty",
43            post(provide_sovereignty_service),
44        )
45        .route("/rpc/network-effects", get(get_network_effects))
46        // Cross-Primal Operations
47        .route(
48            "/rpc/operations/execute",
49            post(execute_cross_primal_operation),
50        )
51        .route("/rpc/resources/compute", post(request_compute_resources))
52        .route("/rpc/resources/storage", post(request_storage_resources))
53        .route("/rpc/operations/call", post(call_ecosystem_service))
54        .route("/rpc/operations/broadcast", post(broadcast_to_ecosystem))
55        // Management
56        .route("/rpc/deregister", post(deregister_from_ecosystem))
57        .route("/rpc/metadata", get(get_primal_metadata))
58        .route("/rpc/health", get(primal_health_check))
59        .route("/rpc/services/discover", post(discover_ecosystem_services))
60        .route("/rpc/services/subscribe", post(subscribe_to_service))
61        .route("/rpc/services/unsubscribe", post(unsubscribe_from_service))
62        .route("/rpc/metrics", get(get_ecosystem_metrics))
63        .route("/rpc/topology", get(get_ecosystem_topology))
64}
