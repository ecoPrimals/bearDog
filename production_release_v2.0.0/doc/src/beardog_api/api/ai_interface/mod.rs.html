31/// Create AI-first API router
32pub fn create_ai_router() -> Router<Arc<BearDogCore>> {
33    Router::new()
34        // Health and status endpoints
35        .route("/ai/health", get(health_check))
36        .route("/ai/status", get(system_status))
37        .route("/ai/metrics", get(metrics))
38        // Security endpoints
39        .route("/ai/encrypt", post(encrypt))
40        .route("/ai/decrypt", post(decrypt))
41        .route("/ai/sign", post(sign))
42        .route("/ai/verify", post(verify))
43        .route("/ai/keygen", post(generate_key))
44        // Batch security endpoints
45        .route("/ai/batch/security", post(batch_security))
46        // Genetics endpoints
47        .route("/ai/spawn", post(spawn_node))
48        .route("/ai/batch/spawn", post(batch_spawn))
49        .route("/ai/spawn/status/:node_id", get(spawn_status))
50        // HSM endpoints
51        .route("/ai/hsm/status", get(hsm_status))
52        .route("/ai/hsm/tiers", get(hsm_tiers))
53        .route("/ai/hsm/select", post(select_hsm_tier))
54        // Streaming endpoints
55        .route("/ai/stream/status", get(stream_status))
56        .route("/ai/stream/metrics", get(stream_metrics))
57}
