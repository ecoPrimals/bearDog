1//! Type definitions and data structures for proof verification
2//!
3//! Contains all structs, enums, and type aliases for verification operations.
4
5use chrono::{DateTime, Utc};
6use serde::{Deserialize, Serialize};
7use std::collections::HashMap;
8
9// Import auth types that verification depends on
10use crate::auth::{BearDogGenetics, ResourcePermission};
11
12/// Configuration for proof verification operations
13#[derive(Debug, Clone, Serialize, Deserialize)]
14pub struct VerificationConfig {
15    /// Whether to enable proof caching
16    pub enable_proof_caching: bool,
17    /// Maximum cache size for verification results
18    pub max_cache_size: usize,
19    /// Cache TTL in seconds
20    pub cache_ttl_seconds: u64,
21    /// Whether to perform deep verification
22    pub deep_verification: bool,
23    /// Cryptographic strength requirements
24    pub min_crypto_strength: f64,
25}
26
27impl Default for VerificationConfig {
28    fn default() -> Self {
29        Self {
30            enable_proof_caching: true,
31            max_cache_size: 10000,
32            cache_ttl_seconds: 3600, // 1 hour
33            deep_verification: true,
34            min_crypto_strength: 0.8,
35        }
36    }
37}
38
39/// Verification result for authorization proofs
40#[derive(Debug, Clone, Serialize, Deserialize)]
41pub struct VerificationResult {
42    /// Whether the proof is valid
43    pub valid: bool,
44    /// Confidence score (0.0-1.0)
45    pub confidence: f64,
46    /// Details about the verification
47    pub details: String,
48    /// Timestamp of verification
49    pub verified_at: DateTime<Utc>,
50    /// Verification method used
51    pub method: VerificationMethod,
52}
53
54/// Methods used for verification
55#[derive(Debug, Clone, Serialize, Deserialize)]
56pub enum VerificationMethod {
57    /// Cryptographic signature verification
58    CryptographicSignature,
59    /// Genetic consistency check
60    GeneticConsistency,
61    /// Timestamp verification
62    TimestampVerification,
63    /// Cross-reference validation
64    CrossReference,
65    /// Combined verification
66    Combined(Vec<VerificationMethod>),
67}
68
69/// Cached verification result
70#[derive(Debug, Clone)]
71pub struct CachedVerification {
72    /// The verification result
73    pub result: VerificationResult,
74    /// When this was cached
75    pub cached_at: DateTime<Utc>,
76    /// Cache expiry time
77    pub expires_at: DateTime<Utc>,
78}
79
80/// Proof verification context
81#[derive(Debug, Clone)]
82pub struct VerificationContext {
83    /// Node genetics for genetic verification
84    pub node_genetics: Option<BearDogGenetics>,
85    /// Required permission level
86    pub required_permission: Option<ResourcePermission>,
87    /// Additional context data
88    pub context_data: HashMap<String, String>,
89}
90
91/// Verification statistics
92#[derive(Debug, Clone, Serialize, Deserialize)]
93pub struct VerificationStats {
94    /// Total verifications performed
95    pub total_verifications: u64,
96    /// Successful verifications
97    pub successful_verifications: u64,
98    /// Failed verifications
99    pub failed_verifications: u64,
100    /// Cache hits
101    pub cache_hits: u64,
102    /// Cache misses
103    pub cache_misses: u64,
104    /// Average verification time in milliseconds
105    pub avg_verification_time_ms: f64,
106}
