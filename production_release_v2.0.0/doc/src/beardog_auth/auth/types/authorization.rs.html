10/// Configuration for cross-node authorization

13/// policies for proof verification, genetic spawning, consensus requirements,

16pub struct CrossNodeAuthConfig {

21    /// Enable genetic spawning authorization

47/// Cross-node authorization structure
48///
49/// Represents an authorization grant from one node to another, including
50/// permissions, conditions, expiration, and cryptographic validation.
51#[derive(Debug, Clone, Serialize, Deserialize)]
52pub struct CrossNodeAuthorization {
53    /// Unique authorization ID
54    pub id: String,
55    /// Requesting node ID
56    pub requester_node_id: String,
57    /// Resource owner node ID
58    pub resource_owner_node_id: String,
59    /// Resource identifier
60    pub resource_id: String,
61    /// Granted permissions
62    pub permissions: Vec<ResourcePermission>,
63    /// Access conditions
64    pub conditions: Vec<AccessCondition>,
65    /// Authorization creation time
66    pub created_at: DateTime<Utc>,
67    /// Authorization expiration time
68    pub expires_at: DateTime<Utc>,
69    /// Cryptographic signature
70    pub signature: String,
71    /// Whether the authorization is currently active
72    pub is_active: bool,
73}
74
75impl CrossNodeAuthorization {
76    /// Check if authorization is currently valid
77    pub fn is_valid(&self) -> bool {
78        self.is_active && Utc::now() < self.expires_at
79    }
80
81    /// Check if authorization grants specific permission
82    pub fn has_permission(&self, permission: &ResourcePermission) -> bool {
83        self.permissions.iter().any(|p| p.implies(permission))
84    }
85}

87/// Resource permission levels
88///
89/// Defines the different types of permissions that can be granted
90/// for accessing resources across nodes.
91#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
92pub enum ResourcePermission {
93    /// Read access to the resource
94    Read,
95    /// Write access to modify the resource
96    Write,
97    /// Delete permission to remove the resource
98    Delete,
99    /// Administrative access for full control
100    Admin,
101    /// Execute permission for running operations
102    Execute,
103    /// Permission to create new resources
104    Create,
105    /// Permission to share resources with other nodes
106    Share,
107    /// Permission to backup the resource
108    Backup,
109    /// Permission to restore from backup
110    Restore,
111    /// Permission to audit resource access
112    Audit,
113    /// Permission to replicate data across nodes
114    Replicate,
115    /// Permission to spawn new BearDog instances
116    Spawn,
117    /// Permission to modify genetic algorithms
118    GeneticModify,
119    /// Permission to participate in consensus
120    Consensus,
121    /// Permission to perform compliance operations
122    Compliance,
123}
124
125impl ResourcePermission {
126    /// Check if permission implies another permission
127    pub fn implies(&self, other: &Self) -> bool {
128        match (self, other) {
129            (ResourcePermission::Admin, _) => true,
130            (ResourcePermission::Write, ResourcePermission::Read) => true,
131            (ResourcePermission::Delete, ResourcePermission::Write) => true,
132            (ResourcePermission::Spawn, ResourcePermission::Create) => true,
133            (a, b) => a == b,
134        }
135    }
136
137    /// Get the security level of the permission (higher = more sensitive)
138    pub fn security_level(&self) -> u8 {
139        match self {
140            ResourcePermission::Read => 1,
141            ResourcePermission::Audit => 2,
142            ResourcePermission::Backup => 3,
143            ResourcePermission::Execute => 4,
144            ResourcePermission::Create => 5,
145            ResourcePermission::Write => 6,
146            ResourcePermission::Share => 7,
147            ResourcePermission::Replicate => 8,
148            ResourcePermission::Restore => 9,
149            ResourcePermission::Consensus => 10,
150            ResourcePermission::Compliance => 11,
151            ResourcePermission::GeneticModify => 12,
152            ResourcePermission::Spawn => 13,
153            ResourcePermission::Delete => 14,
154            ResourcePermission::Admin => 15,
155        }
156    }
157}

159/// Access condition modifiers
160///
161/// Defines conditions that must be met for an authorization to be valid,
162/// such as time windows, IP restrictions, and usage limits.
163#[derive(Debug, Clone, Serialize, Deserialize)]
164pub enum AccessCondition {
165    /// Time-based access window
166    TimeWindow {
167        /// Start time of the access window
168        start: DateTime<Utc>,
169        /// End time of the access window
170        end: DateTime<Utc>,
171    },
172    /// IP address restriction
173    IpAddress(String),
174    /// Require multi-factor authentication
175    RequireMfa,
176    /// Maximum usage count
177    MaxUsage(u32),
178    /// Rate limiting
179    RateLimit {
180        /// Maximum requests allowed
181        max_requests: u32,
182        /// Time window in seconds
183        window_seconds: u32,
184    },
185    /// Require consensus from other nodes
186    RequireConsensus {
187        /// Required consensus threshold (0.0 to 1.0)
188        threshold: f64,
189        /// List of nodes that must participate
190        nodes: Vec<String>,
191    },
192}
