1//! Workflow system types for cross-node operations
2//!
3//! This module contains all types related to workflow management,
4//! including workflow requests, definitions, automated checks, and escalation conditions.
5
6use chrono::{DateTime, Utc};
7use serde::{Deserialize, Serialize};
8
9use super::authorization::ResourcePermission;
10use super::genetics::NodeCapability;
11use super::spawning::SpawnPurpose;
12
13/// Cross-node workflow request
14///
15/// Represents a request to execute a workflow across multiple nodes,
16/// including automation checks and escalation conditions.
17#[derive(Debug, Clone, Serialize, Deserialize)]
18pub struct CrossNodeWorkflowRequest {
19    /// Unique request identifier
20    pub id: String,
21    /// Type of workflow requested
22    pub workflow_type: BearDogWorkflowType,
23    /// Node ID making the request
24    pub requester_node_id: String,
25    /// Target nodes for the workflow
26    pub target_nodes: Vec<String>,
27    /// Required permissions for the workflow
28    pub required_permissions: Vec<ResourcePermission>,
29    /// Automated checks to perform
30    pub automated_checks: Vec<AutomatedCheck>,
31    /// Conditions that trigger escalation
32    pub escalation_conditions: Vec<EscalationCondition>,
33    /// When the request was created
34    pub created_at: DateTime<Utc>,
35    /// When the request expires
36    pub expires_at: DateTime<Utc>,
37}
38
39/// Workflow types for cross-node operations
40#[derive(Debug, Clone, Serialize, Deserialize)]
41pub enum BearDogWorkflowType {
42    /// Data backup workflow
43    DataBackup {
44        /// Source node for the backup
45        source_node: String,
46        /// Destination nodes for backup storage
47        backup_nodes: Vec<String>,
48        /// Whether encryption is required
49        encryption_required: bool,
50    },
51    /// Compliance audit workflow
52    ComplianceAudit {
53        /// Scope of the audit
54        audit_scope: Vec<String>,
55        /// Standards to audit against
56        standards: Vec<String>,
57        /// Whether automated remediation is enabled
58        automated_remediation: bool,
59    },
60    /// Security incident response workflow
61    SecurityIncidentResponse {
62        /// Threat level (1-10)
63        threat_level: u8,
64        /// Affected resources
65        affected_resources: Vec<String>,
66        /// Response team members
67        response_team: Vec<String>,
68    },
69    /// Genetic spawning workflow
70    GeneticSpawning {
71        /// Parent genetics to use
72        parent_genetics: Vec<String>,
73        /// Purpose for spawning
74        spawn_purpose: SpawnPurpose,
75        /// Target capabilities for spawned instance
76        target_capabilities: Vec<NodeCapability>,
77    },
78}

80/// Automated checks for workflow approval
81#[derive(Debug, Clone, Serialize, Deserialize)]
82pub enum AutomatedCheck {
83    /// Check resource availability
84    ResourceAvailability,
85    /// Check security clearance
86    SecurityClearance,
87    /// Validate compliance requirements
88    ComplianceValidation,
89    /// Verify trust relationships
90    TrustVerification,
91    /// Match capabilities
92    CapabilityMatch,
93}
94
95/// Conditions that trigger escalation to human approval
96#[derive(Debug, Clone, Serialize, Deserialize)]
97pub enum EscalationCondition {
98    /// High risk operation detected
99    HighRiskOperation,
100    /// Compliance violation detected
101    ComplianceViolation,
102    /// Unknown node detected
103    UnknownNode,
104    /// Resource exhaustion detected
105    ResourceExhaustion,
106    /// Security threat detected
107    SecurityThreat,
108}
109
110/// Workflow status enumeration
111#[derive(Debug, Clone, Serialize, Deserialize)]
112pub enum WorkflowStatus {
113    /// Workflow is pending
114    Pending,
115    /// Workflow is in progress
116    InProgress,
117    /// Workflow has completed successfully
118    Completed,
119    /// Workflow has failed with an error
120    Failed(String),
121    /// Workflow requires human approval
122    RequiresApproval,
123}
