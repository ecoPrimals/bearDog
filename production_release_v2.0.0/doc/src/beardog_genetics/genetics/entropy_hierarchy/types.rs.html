11/// Three-tier entropy hierarchy: Human > Human-Supervised > Store-Bought
12#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
13pub enum EntropyClass {
14    /// HIGHEST TIER: Human-Lived Experience Entropy
15    /// - From microphone, camera, haptic sensors
16    /// - Irreproducible and uniquely owned
17    /// - Cannot be replicated by machines
18    HumanLivedExperience {
19        /// The type of human entropy source used
20        source_type: HumanEntropySource,
21        /// Timestamp when the entropy was captured
22        capture_timestamp: DateTime<Utc>,
23        /// Biometric signature for authentication
24        biometric_signature: BiometricHash,
25        /// Proof of ownership for this entropy
26        ownership_proof: OwnershipProof,
27    },
28
29    /// MID-TIER: Human-Supervised Machine Entropy
30    /// - Machine-generated but human-validated
31    /// - Reproducible but authenticated
32    HumanSupervisedMachine {
33        /// The machine source that generated the entropy
34        machine_source: MachineEntropySource,
35        /// Human validator who supervised the generation
36        human_validator: HumanIdentity,
37        /// Timestamp when the entropy was validated
38        validation_timestamp: DateTime<Utc>,
39    },
40
41    /// LOWEST TIER: Store-Bought Machine Entropy
42    /// - Standard CSPRNG, hardware RNG
43    /// - Reproducible and "store-bought compute random"
44    /// - Default for all-machine operations
45    StoreBoughtMachine {
46        /// The type of machine entropy source
47        source_type: MachineEntropySource,
48        /// Timestamp when the entropy was generated
49        generation_timestamp: DateTime<Utc>,
50        /// Reproducibility index (0.0-1.0, higher = more reproducible)
51        reproducibility_index: f64,
52    },
53}
54
55impl PartialOrd for EntropyClass {
56    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
57        // Define precedence levels: HumanLivedExperience > HumanSupervisedMachine > StoreBoughtMachine
58        let self_precedence = match self {
59            EntropyClass::HumanLivedExperience { .. } => 3,
60            EntropyClass::HumanSupervisedMachine { .. } => 2,
61            EntropyClass::StoreBoughtMachine { .. } => 1,
62        };
63
64        let other_precedence = match other {
65            EntropyClass::HumanLivedExperience { .. } => 3,
66            EntropyClass::HumanSupervisedMachine { .. } => 2,
67            EntropyClass::StoreBoughtMachine { .. } => 1,
68        };
69
70        self_precedence.partial_cmp(&other_precedence)
71    }
72}

74/// Sources of human entropy

76pub enum HumanEntropySource {
77    /// Microphone entropy from ambient sound

92        resolution: (u32, u32),
93        /// Lighting variation measurements
94        lighting_variations: Vec<f32>,
95    },
96
97    /// Haptic entropy from touch and motion
98    Haptic {
99        /// Duration of haptic capture in milliseconds
100        duration_ms: u32,
101        /// Touch points on the interface
102        touch_points: Vec<(f32, f32)>,
103        /// Motion patterns detected
104        motion_patterns: Vec<f32>,
105    },
106
107    /// Biometric entropy (privacy-preserving)
108    Biometric {
109        /// Entropy hash derived from biometric data
110        entropy_hash: Vec<u8>,
111        /// Type of biometric data used
112        biometric_type: String,
113        /// Quality score of the biometric data (0.0-1.0)
114        quality_score: f64,
115    },
116
117    /// Multi-modal fusion of multiple sources
118    MultiModalHuman {
119        /// List of human entropy sources being fused
120        sources: Vec<HumanEntropySource>,
121        /// Algorithm used for fusion
122        fusion_algorithm: FusionAlgorithm,
123        /// Confidence score of the fusion result (0.0-1.0)
124        confidence_score: f64,
125    },
126}

128impl PartialOrd for HumanEntropySource {
129    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
130        use HumanEntropySource::*;
131        let self_quality = match self {
132            MultiModalHuman {
133                confidence_score, ..
134            } => *confidence_score,
135            Biometric { quality_score, .. } => *quality_score,
136            Microphone { .. } => 0.8,
137            Camera { .. } => 0.7,
138            Haptic { .. } => 0.6,
139        };
140
141        let other_quality = match other {
142            MultiModalHuman {
143                confidence_score, ..
144            } => *confidence_score,
145            Biometric { quality_score, .. } => *quality_score,
146            Microphone { .. } => 0.8,
147            Camera { .. } => 0.7,
148            Haptic { .. } => 0.6,
149        };
150
151        self_quality.partial_cmp(&other_quality)
152    }
153}

453/// Secure container for secret bytes with automatic zeroization
454#[derive(Debug, Clone, Serialize, Deserialize)]
455pub struct SecretBytes {
456    /// Raw bytes of the secret data
457    #[serde(skip)]
458    bytes: Vec<u8>,
459}
460
461impl SecretBytes {
462    /// Create a new SecretBytes container with the given bytes
463    pub fn new(bytes: Vec<u8>) -> Self {
464        Self { bytes }
465    }
466
467    /// Get a reference to the underlying bytes
468    pub fn as_bytes(&self) -> &[u8] {
469        &self.bytes
470    }
471
472    /// Check if the container is empty
473    pub fn is_empty(&self) -> bool {
474        self.bytes.is_empty()
475    }
476}
477
478impl Zeroize for SecretBytes {
479    fn zeroize(&mut self) {
480        self.bytes.zeroize();
481    }

499/// Proof that entropy cannot be reproduced
500#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
501pub struct IrreproducibilityProof {
502    /// Cryptographic commitment to the entropy value
503    pub entropy_commitment: Vec<u8>,
504    /// Proof that the entropy was generated at a specific time
505    pub temporal_proof: Vec<u8>,
506    /// Proof that the entropy is unique and cannot be reproduced
507    pub uniqueness_proof: Vec<u8>,
508}
509
510impl Zeroize for IrreproducibilityProof {
511    fn zeroize(&mut self) {
512        self.entropy_commitment.zeroize();
513        self.temporal_proof.zeroize();
514        self.uniqueness_proof.zeroize();
515    }
516}
