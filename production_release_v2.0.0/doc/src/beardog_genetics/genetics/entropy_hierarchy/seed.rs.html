15    /// Create a new human entropy seed with cryptographic proofs
16    pub async fn new_human_entropy(
17        entropy_class: EntropyClass,
18        lifetime_policy: SeedLifetimePolicy,
19        owner_identity: HumanIdentity,
20        seed_bytes: Vec<u8>,
21        entropy_manager: &crate::genetics::entropy_hierarchy::EntropyHierarchyManager,
22    ) -> BearDogResult<Self> {
23        let seed_id = Uuid::new_v4();
24
25        // Generate ownership proof
26        let ownership_proof = entropy_manager
27            .generate_ownership_proof(&owner_identity, &seed_bytes)
28            .await?;
29
30        // Generate irreproducibility proof
31        let irreproducibility_proof = entropy_manager
32            .generate_irreproducibility_proof(&seed_bytes, &entropy_class)
33            .await?;
34
35        // Create usage policy based on entropy class
36        let usage_policy = match &entropy_class {
37            EntropyClass::HumanLivedExperience { .. } => SeedUsagePolicy {
38                allowed_operations: vec![
39                    "key_derivation".to_string(),
40                    "signing".to_string(),
41                    "encryption".to_string(),
42                ],
43                max_uses: None, // No limit for human entropy
44                requires_approval: false,
45            },
46            _ => SeedUsagePolicy {
47                allowed_operations: vec!["key_derivation".to_string()],
48                max_uses: Some(100),
49                requires_approval: true,
50            },
51        };
52
53        Ok(EntropySeed {
54            id: seed_id,
55            seed_bytes: SecretBytes::new(seed_bytes),
56            entropy_class,
57            generation_time: Utc::now(),
58            lifetime_policy,
59            ownership: SeedOwnership::HumanOwned {
60                owner_identity,
61                ownership_proof,
62                transfer_count: 0,
63            },
64            irreproducibility_proof,
65            usage_policy,
66            usage_history: Vec::new(),
67            social_context: None,
68        })
69    }

71    /// Create a new event-based seed for social contexts
72    pub async fn new_event_seed(
73        event_context: SocialContext,
74        sharing_policy: SharingPolicy,
75        owner_identity: HumanIdentity,
76        seed_bytes: Vec<u8>,
77        entropy_manager: &crate::genetics::entropy_hierarchy::EntropyHierarchyManager,
78    ) -> BearDogResult<Self> {
79        let seed_id = Uuid::new_v4();
80
81        // Generate ownership proof
82        let ownership_proof = entropy_manager
83            .generate_ownership_proof(&owner_identity, &seed_bytes)
84            .await?;
85
86        // Generate biometric signature from event context (clone needed values first)
87        let location_clone = event_context.location.clone();
88        let timestamp = event_context.event_timestamp;
89        let biometric_data = format!(
90            "{}:{}:{}",
91            timestamp.timestamp(),
92            location_clone.unwrap_or_else(|| "Unknown".to_string()),
93            0.5 // Default quality score during refactor
94        );
95
96        // Create cryptographic hash of biometric-like data
97        use sha2::{Digest, Sha256};
98        let mut hasher = Sha256::new();
99        hasher.update(biometric_data.as_bytes());
100        let biometric_hash = hasher.finalize().to_vec();
101
102        // Create entropy class for event seed
103        let entropy_class = EntropyClass::HumanLivedExperience {
104            source_type: HumanEntropySource::MultiModalHuman {
105                sources: vec![], // Would be populated with actual sources
106                fusion_algorithm: FusionAlgorithm::CryptographicMixing,
107                confidence_score: 0.9,
108            },
109            capture_timestamp: event_context.event_timestamp,
110            biometric_signature: BiometricHash(biometric_hash),
111            ownership_proof: ownership_proof.clone(),
112        };
113
114        // Generate irreproducibility proof
115        let irreproducibility_proof = entropy_manager
116            .generate_irreproducibility_proof(&seed_bytes, &entropy_class)
117            .await?;
118
119        // Create event-based lifetime policy
120        // Clone event_context before using it
121        let event_context_clone = event_context.clone();
122
123        let lifetime_policy = SeedLifetimePolicy::EventBased {
124            event_id: format!("event_{seed_id}"),
125            event_type: event_context.event_type.clone(),
126            sharing_policy: sharing_policy.clone(),
127            event_expiration: sharing_policy.sharing_expiration,
128        };
129
130        Ok(EntropySeed {
131            id: seed_id,
132            seed_bytes: SecretBytes::new(seed_bytes),
133            entropy_class,
134            generation_time: Utc::now(),
135            lifetime_policy,
136            ownership: SeedOwnership::SharedOwnership {
137                primary_owner: owner_identity,
138                shared_with: event_context.participants.clone(),
139                sharing_terms: SharingTerms {
140                    max_participants: sharing_policy.max_shares,
141                    expiration: sharing_policy.sharing_expiration,
142                    permissions: sharing_policy.allowed_operations.clone(),
143                },
144            },
145            irreproducibility_proof,
146            usage_policy: SeedUsagePolicy {
147                allowed_operations: sharing_policy.allowed_operations,
148                max_uses: None,
149                requires_approval: sharing_policy.require_permission,
150            },
151            usage_history: Vec::new(),
152            social_context: Some(event_context_clone), // Use cloned context
153        })
154    }

156    /// Validate that the seed is still valid and usable
157    pub fn is_valid(&self) -> bool {
158        // Check expiration based on lifetime policy
159        match &self.lifetime_policy {
160            SeedLifetimePolicy::Ephemeral {
161                expiration_time, ..
162            } => Utc::now() < *expiration_time,
163            SeedLifetimePolicy::Persistent {
164                ownership_expiration,
165                ..
166            } => ownership_expiration.map_or(true, |exp| Utc::now() < exp),
167            SeedLifetimePolicy::EventBased {
168                event_expiration, ..
169            } => event_expiration.map_or(true, |exp| Utc::now() < exp),
170            SeedLifetimePolicy::SelfSovereign { .. } => true, // Self-sovereign seeds don't expire
171        }
172    }

174    /// Transfer ownership of the seed to a new owner
175    pub fn transfer_ownership(&mut self, new_owner: HumanIdentity) -> BearDogResult<()> {
176        // Check if transfer is allowed
177        match &self.lifetime_policy {
178            SeedLifetimePolicy::Persistent {
179                ownership_transfer_allowed,
180                ..
181            } => {
182                if !ownership_transfer_allowed {
183                    return Err(BearDogError::Internal {
184                        message: "Ownership transfer not allowed for this seed".to_string(),
185                    });
186                }
187            }
188            SeedLifetimePolicy::SelfSovereign {
189                transfer_permissions,
190                ..
191            } => {
192                if !transfer_permissions.transferable {
193                    return Err(BearDogError::Internal {
194                        message: "This self-sovereign seed is not transferable".to_string(),
195                    });
196                }
197            }
198            _ => {
199                return Err(BearDogError::Internal {
200                    message: "Ownership transfer not supported for this seed type".to_string(),
201                });
202            }
203        }
204
205        // Record the transfer before moving new_owner
206        let transfer_context = format!("Transferred to {}", new_owner.identity_id);
207
208        // Update ownership
209        match &mut self.ownership {
210            SeedOwnership::HumanOwned { transfer_count, .. } => {
211                *transfer_count += 1;
212                self.ownership = SeedOwnership::HumanOwned {
213                    owner_identity: new_owner, // Move happens here
214                    ownership_proof: OwnershipProof {
215                        signature: vec![0u8; 64], // Would be generated properly
216                        timestamp: Utc::now(),
217                        verification_key: vec![0u8; 32],
218                    },
219                    transfer_count: *transfer_count,
220                };
221            }
222            _ => {
223                return Err(BearDogError::Internal {
224                    message: "Cannot transfer ownership from non-human owner".to_string(),
225                });
226            }
227        }
228
229        // Record usage event using the saved context
230        self.usage_history.push(SeedUsageEvent {
231            timestamp: Utc::now(),
232            operation: "ownership_transfer".to_string(),
233            context: transfer_context,  // Use the saved context
234            result_hash: vec![0u8; 32], // Would be hash of transfer
235        });
236
237        Ok(())
238    }

240    /// Expire ownership and transition to machine ownership
241    pub fn expire_ownership(&mut self) -> BearDogResult<()> {
242        let previous_owner = match &self.ownership {
243            SeedOwnership::HumanOwned { owner_identity, .. } => Some(owner_identity.clone()),
244            SeedOwnership::SharedOwnership { primary_owner, .. } => Some(primary_owner.clone()),
245            _ => None,
246        };
247
248        self.ownership = SeedOwnership::MachineOwned {
249            previous_owner,
250            ownership_transition: OwnershipTransition::OwnershipExpired,
251            self_sovereign: matches!(
252                self.lifetime_policy,
253                SeedLifetimePolicy::SelfSovereign { .. }
254            ),
255        };
256
257        // Record the expiration
258        self.usage_history.push(SeedUsageEvent {
259            timestamp: Utc::now(),
260            operation: "ownership_expiration".to_string(),
261            context: "Ownership expired due to policy".to_string(),
262            result_hash: vec![0u8; 32],
263        });
264
265        Ok(())
266    }

268    /// Use the seed for a cryptographic operation
269    pub fn use_for_operation(&mut self, operation: &str) -> BearDogResult<Vec<u8>> {
270        // Check if the seed is valid
271        if !self.is_valid() {
272            return Err(BearDogError::Internal {
273                message: "Seed has expired and cannot be used".to_string(),
274            });
275        }
276
277        // Check if operation is allowed
278        if !self
279            .usage_policy
280            .allowed_operations
281            .contains(&operation.to_string())
282        {
283            return Err(BearDogError::Internal {
284                message: format!("Operation '{operation}' not allowed for this seed"),
285            });
286        }
287
288        // Check usage limits
289        if let Some(max_uses) = self.usage_policy.max_uses {
290            let operation_count = self
291                .usage_history
292                .iter()
293                .filter(|event| event.operation == operation)
294                .count() as u32;
295
296            if operation_count >= max_uses {
297                return Err(BearDogError::Internal {
298                    message: format!(
299                        "Maximum usage limit ({max_uses}) reached for operation '{operation}'"
300                    ),
301                });
302            }
303        }
304
305        // Perform the operation (simplified - would use actual crypto)
306        let mut hasher = Sha3_256::new();
307        hasher.update(self.seed_bytes.as_bytes());
308        hasher.update(operation.as_bytes());
309        hasher.update(Utc::now().timestamp().to_le_bytes());
310        let result = hasher.finalize().to_vec();
311
312        // Record the usage
313        let result_hash = {
314            let mut hash_hasher = Sha3_256::new();
315            hash_hasher.update(&result);
316            hash_hasher.finalize().to_vec()
317        };
318
319        self.usage_history.push(SeedUsageEvent {
320            timestamp: Utc::now(),
321            operation: operation.to_string(),
322            context: "Cryptographic operation".to_string(),
323            result_hash,
324        });
325
326        Ok(result)
327    }

329    /// Securely destroy the seed
330    pub fn destroy(&mut self) {
331        self.seed_bytes.zeroize();
332        self.irreproducibility_proof.zeroize();
333
334        // Mark as destroyed in usage history
335        self.usage_history.push(SeedUsageEvent {
336            timestamp: Utc::now(),
337            operation: "destroy".to_string(),
338            context: "Seed securely destroyed".to_string(),
339            result_hash: vec![0u8; 32],
340        });
341    }

352    /// Check if seed can be shared with others
353    pub fn can_be_shared(&self) -> bool {
354        match &self.lifetime_policy {
355            SeedLifetimePolicy::EventBased { sharing_policy, .. } => {
356                sharing_policy.max_shares.is_some()
357            }
358            SeedLifetimePolicy::SelfSovereign {
359                transfer_permissions,
360                ..
361            } => transfer_permissions.transferable,
362            _ => false,
363        }
364    }

366    /// Get current owner identity if human-owned
367    pub fn get_current_owner(&self) -> Option<&HumanIdentity> {
368        match &self.ownership {
369            SeedOwnership::HumanOwned { owner_identity, .. } => Some(owner_identity),
370            SeedOwnership::SharedOwnership { primary_owner, .. } => Some(primary_owner),
371            SeedOwnership::CommunityOwned { .. } => None, // Community ownership
372            SeedOwnership::MachineOwned { .. } => None,   // Machine ownership

376    /// Get usage statistics for this seed
377    pub fn get_usage_stats(&self) -> HashMap<String, u32> {
378        let mut stats = HashMap::new();
379        for event in &self.usage_history {
380            *stats.entry(event.operation.clone()).or_insert(0) += 1;
381        }
382        stats
383    }

390    /// Get remaining usage count for operations with limits
391    pub fn get_remaining_uses(&self, operation: &str) -> Option<u32> {
392        self.usage_policy.max_uses.map(|max_uses| {
393            let used = self
394                .usage_history
395                .iter()
396                .filter(|event| event.operation == operation)
397                .count() as u32;
398            max_uses.saturating_sub(used)
399        })
