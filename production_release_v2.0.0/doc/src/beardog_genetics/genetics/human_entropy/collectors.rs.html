159    /// Validate user consent
160    async fn validate_consent(&self, consent: &InformedConsent) -> BearDogResult<()> {
161        // Check if consent is recent (within 24 hours)
162        let now = Utc::now();
163        let consent_age = now.signed_duration_since(consent.consent_timestamp);
164
165        if consent_age > chrono::Duration::hours(24) {
166            return Err(BearDogError::config("Consent expired"));
167        }
168
169        // Validate consent signature (simplified - in real implementation, verify cryptographic signature)
170        if consent.consent_signature.signature_bytes.is_empty() {
171            return Err(BearDogError::config("Invalid consent"));
172        }
173
174        Ok(())
175    }

535impl MultiModalHumanEntropyCollector {
536    /// Create a new multi-modal entropy collector
537    pub fn new(config: HumanEntropyConfig) -> Self {
538        let microphone_collector = if config.audio_config.enabled {
539            Some(MicrophoneEntropyCollector::new(config.audio_config.clone()))
540        } else {
541            None
542        };
543
544        let camera_collector = if config.visual_config.enabled {
545            Some(CameraEntropyCollector::new(config.visual_config.clone()))
546        } else {
547            None
548        };
549
550        let haptic_collector = if config.haptic_config.enabled {
551            Some(HapticEntropyCollector::new(config.haptic_config.clone()))
552        } else {
553            None
554        };
555
556        Self {
557            config,
558            microphone_collector,
559            camera_collector,
560            haptic_collector,
561            fusion_algorithm: HumanPreservingFusion::new(),
562        }
563    }
564
565    /// Collect multimodal entropy from all available sources
566    pub async fn collect_multimodal_entropy(
567        &self,
568        consent: &InformedConsent,
569    ) -> BearDogResult<FusedEntropyResult> {
570        let mut entropy_sources = Vec::new();
571        let mut source_types = Vec::new();
572
573        // Collect from audio if enabled
574        if let Some(ref collector) = self.microphone_collector {
575            match collector
576                .collect_audio_entropy(self.config.collection_duration, consent)
577                .await
578            {
579                Ok(audio_entropy) => {
580                    entropy_sources.push(audio_entropy.entropy_bytes);
581                    source_types.push("audio".to_string());
582                }
583                Err(e) => {
584                    println!("Audio entropy collection failed: {e:?}");
585                }
586            }
587        }
588
589        // Collect from visual if enabled
590        if let Some(ref collector) = self.camera_collector {
591            match collector
592                .collect_visual_entropy(self.config.collection_duration, consent)
593                .await
594            {
595                Ok(visual_entropy) => {
596                    entropy_sources.push(visual_entropy.entropy_bytes);
597                    source_types.push("visual".to_string());
598                }
599                Err(e) => {
600                    println!("Visual entropy collection failed: {e:?}");
601                }
602            }
603        }
604
605        // Collect from haptic if enabled
606        if let Some(ref collector) = self.haptic_collector {
607            match collector
608                .collect_haptic_entropy(self.config.collection_duration, consent)
609                .await
610            {
611                Ok(haptic_entropy) => {
612                    entropy_sources.push(haptic_entropy.entropy_bytes);
613                    source_types.push("haptic".to_string());
614                }
615                Err(e) => {
616                    println!("Haptic entropy collection failed: {e:?}");
617                }
618            }
619        }
620
621        // Ensure we have at least one source
622        if entropy_sources.is_empty() {
623            return Err(BearDogError::config("No entropy sources available"));
624        }
625
626        // Fuse entropy sources
627        let fused_entropy = self
628            .fusion_algorithm
629            .fuse_entropy_sources(&entropy_sources)
630            .await?;
631        let fusion_quality = self.assess_fusion_quality(&entropy_sources).await?;
632
633        Ok(FusedEntropyResult {
634            entropy_bytes: fused_entropy,
635            component_count: entropy_sources.len(),
636            fusion_timestamp: Utc::now(),
637            source_types,
638            fusion_quality,
639        })
640    }
641
642    /// Assess the quality of entropy fusion
643    async fn assess_fusion_quality(&self, entropy_sources: &[SecretBytes]) -> BearDogResult<f64> {
644        // Simple quality assessment based on diversity and quantity
645        let source_count = entropy_sources.len();
646        let total_entropy = entropy_sources.iter().map(|s| s.len()).sum::<usize>();
647
648        let diversity_score = source_count as f64 / 3.0; // Max 3 sources
649        let quantity_score = (total_entropy as f64 / 100.0).min(1.0); // Normalize to 1.0
650
651        Ok((diversity_score + quantity_score) / 2.0)
652    }
