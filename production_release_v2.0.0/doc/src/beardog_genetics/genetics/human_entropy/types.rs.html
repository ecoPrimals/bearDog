11/// Secret bytes that are automatically zeroed on drop
12///
13/// This type provides secure handling of sensitive entropy data by automatically
14/// zeroing the underlying bytes when the instance is dropped, preventing
15/// sensitive data from remaining in memory.
16#[derive(Debug, Clone)]
17pub struct SecretBytes(Vec<u8>);
18
19impl SecretBytes {
20    /// Create a new SecretBytes instance
21    pub fn new(bytes: Vec<u8>) -> Self {
22        Self(bytes)
23    }
24
25    /// Get a reference to the underlying bytes
26    pub fn as_bytes(&self) -> &[u8] {
27        &self.0
28    }
29
30    /// Get the length of the underlying bytes
31    pub fn len(&self) -> usize {
32        self.0.len()
33    }
34
35    /// Check if the bytes are empty
36    pub fn is_empty(&self) -> bool {
37        self.0.is_empty()
38    }
39}
40
41impl Drop for SecretBytes {
42    fn drop(&mut self) {
43        self.0.zeroize();
44    }
45}
