1//! Ethical framework for human entropy collection
2//!
3//! This module defines comprehensive ethical guidelines and policies for
4//! collecting human-generated entropy, ensuring privacy, consent, and user
5//! control at all times.
6
7use chrono::{DateTime, Utc};
8use serde::{Deserialize, Serialize};
9
10/// Ethical framework for human entropy collection
11///
12/// Defines the comprehensive ethical guidelines and policies that must be
13/// followed during human entropy collection, ensuring privacy, consent,
14/// and user control at all times.
15#[derive(Debug, Clone, Serialize, Deserialize)]
16pub struct EntropyCollectionEthics {
17    /// Informed consent requirements and validation
18    pub informed_consent: InformedConsent,
19    /// Privacy protection policies and measures
20    pub privacy_protection: PrivacyProtection,
21    /// Data minimization principles and enforcement
22    pub data_minimization: DataMinimization,
23    /// User control capabilities and rights
24    pub user_control: UserControl,
25    /// Purpose limitation and scope definitions
26    pub purpose_limitation: PurposeLimitation,
27    /// Transparency policies and reporting
28    pub transparency: TransparencyPolicy,
29}
30
31/// Informed consent for entropy collection
32///
33/// Represents the legally binding informed consent given by a user for
34/// entropy collection activities. Includes full disclosure of collection
35/// methods, usage, and retention policies.
36#[derive(Debug, Clone, Serialize, Deserialize)]
37pub struct InformedConsent {
38    /// Detailed description of what data is collected and how
39    pub collection_description: String,
40    /// Explanation of how the collected data will be used
41    pub usage_description: String,
42    /// How long the processed data will be retained
43    pub retention_period: chrono::Duration,
44    /// User's withdrawal rights and procedures
45    pub withdrawal_rights: WithdrawalRights,
46    /// Timestamp when consent was granted
47    pub consent_timestamp: DateTime<Utc>,
48    /// Digital signature validating the consent
49    pub consent_signature: DigitalSignature,
50}
51
52/// User withdrawal rights and procedures
53#[derive(Debug, Clone, Serialize, Deserialize)]
54pub struct WithdrawalRights {
55    /// Whether the user can withdraw consent
56    pub can_withdraw: bool,
57    /// Process for withdrawing consent
58    pub withdrawal_process: String,
59    /// Timeline for data deletion after withdrawal
60    pub data_deletion_timeline: chrono::Duration,
61}
62
63/// Digital signature for consent validation
64#[derive(Debug, Clone, Serialize, Deserialize)]
65pub struct DigitalSignature {
66    /// Raw signature bytes
67    pub signature_bytes: Vec<u8>,
68    /// Signature algorithm used
69    pub algorithm: String,
70    /// Identifier of the signing key
71    pub key_id: String,
72}
73
74/// Privacy protection policies
75#[derive(Debug, Clone, Serialize, Deserialize)]
76pub struct PrivacyProtection {
77    /// Whether data minimization is applied
78    pub data_minimization: bool,
79    /// Whether raw data is immediately deleted after processing
80    pub immediate_raw_deletion: bool,
81    /// Level of anonymization applied to processed data
82    pub anonymization_level: AnonymizationLevel,
83    /// Whether data is encrypted at rest
84    pub encryption_at_rest: bool,
85}
86
87/// Levels of data anonymization
88#[derive(Debug, Clone, Serialize, Deserialize)]
89pub enum AnonymizationLevel {
90    /// No anonymization applied
91    None,
92    /// Basic anonymization (identifiers removed)
93    Low,
94    /// Moderate anonymization (additional noise added)
95    Medium,
96    /// High anonymization (significant data transformation)
97    High,
98    /// Maximum anonymization (only aggregate patterns preserved)
99    Maximum,
100}
101
102/// Data minimization principles
103#[derive(Debug, Clone, Serialize, Deserialize)]
104pub struct DataMinimization {
105    /// Only collect data that is strictly necessary
106    pub collect_only_necessary: bool,
107    /// Only extract features, not store raw data
108    pub feature_extraction_only: bool,
109    /// How long raw data is retained (should be minimal)
110    pub raw_data_retention: chrono::Duration,
111}
112
113/// User control capabilities
114#[derive(Debug, Clone, Serialize, Deserialize)]
115pub struct UserControl {
116    /// User can pause data collection at any time
117    pub can_pause_collection: bool,
118    /// User can review what data has been collected
119    pub can_review_data: bool,
120    /// User can delete their data
121    pub can_delete_data: bool,
122    /// User has granular control over what is collected
123    pub granular_permissions: bool,
124}
125
126/// Purpose limitation for data collection
127#[derive(Debug, Clone, Serialize, Deserialize)]
128pub struct PurposeLimitation {
129    /// Primary purpose for data collection
130    pub primary_purpose: String,
131    /// Any secondary purposes (requires explicit consent)
132    pub secondary_purposes: Vec<String>,
133    /// Whether users are notified of purpose changes
134    pub purpose_change_notification: bool,
135}
136
137/// Transparency policies
138#[derive(Debug, Clone, Serialize, Deserialize)]
139pub struct TransparencyPolicy {
140    /// Description of algorithms used for processing
141    pub algorithm_description: String,
142    /// Documentation of data flow and processing
143    pub data_flow_documentation: String,
144    /// Security measures in place
145    pub security_measures: String,
146    /// Whether users can access audit logs
147    pub audit_log_access: bool,
148}
149
150/// Privacy metadata for collected data
151#[derive(Debug, Clone, Serialize, Deserialize)]
152pub struct PrivacyMetadata {
153    /// Whether data minimization was applied
154    pub data_minimization_applied: bool,
155    /// Whether raw data was deleted
156    pub raw_data_deleted: bool,
157    /// Level of anonymization applied
158    pub anonymization_level: AnonymizationLevel,
159    /// When the data was processed
160    pub processing_timestamp: DateTime<Utc>,
161}
