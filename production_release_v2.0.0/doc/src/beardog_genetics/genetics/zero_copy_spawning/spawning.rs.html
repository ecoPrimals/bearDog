42    /// Spawn new genetics using zero-copy operations
43    pub async fn spawn_genetics(
44        &self,
45        parent_a: &BearDogGenetics,
46        parent_b: &BearDogGenetics,
47    ) -> BearDogResult<BearDogGenetics> {
48        debug!("🧬 Spawning genetics with zero-copy optimization");
49
50        // Get pooled genetics structure for reuse, or create new one
51        let mut child_genetics = self
52            .genetics_pool
53            .get_genetics()
54            .await
55            .unwrap_or_else(BearDogGenetics::default);
56
57        // Perform genetic recombination in-place (zero-copy)
58        self.recombine_genetics(&mut child_genetics, parent_a, parent_b)
59            .await?;
60
61        // Update lineage tracking - use correct method name
62        {
63            let mut lineage = self.lineage_tracker.write().await;
64            lineage.track_spawn(parent_a.id.clone(), child_genetics.id.clone());
65            lineage.track_spawn(parent_b.id.clone(), child_genetics.id.clone());
66        }
67
68        // Update statistics
69        self.stats.record_spawn();
70
71        Ok(child_genetics)
72    }

74    /// Perform genetic recombination using zero-copy techniques
75    async fn recombine_genetics(
76        &self,
77        child: &mut BearDogGenetics,
78        parent_a: &BearDogGenetics,
79        parent_b: &BearDogGenetics,
80    ) -> BearDogResult<()> {
81        // Modernized genetic recombination logic
82        // This is a simplified implementation - in practice would use
83        // sophisticated genetic algorithms with zero-copy optimizations
84
85        // Combine capabilities from both parents
86        child.capabilities = parent_a.capabilities.clone();
87        child.capabilities.extend(parent_b.capabilities.clone());
88
89        // Inherit security traits with mutations - clone to avoid move
90        child.security_clearance = if parent_a.security_clearance >= parent_b.security_clearance {
91            parent_a.security_clearance.clone()
92        } else {
93            parent_b.security_clearance.clone()
94        };
95
96        // Generate new node ID for child
97        child.id = format!("child_{}_{}", parent_a.id, parent_b.id);
98
99        // Set generation (max of parents + 1)
100        child.generation = std::cmp::max(parent_a.generation, parent_b.generation) + 1;
101
102        // Record parent genetics
103        child.parent_genetics = Some(vec![parent_a.id.clone(), parent_b.id.clone()]);
104
105        debug!("🧬 Genetic recombination completed for child: {}", child.id);
106        Ok(())
107    }
