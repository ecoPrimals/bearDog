1//! Genetic Lineage Tracking for Zero-Copy Operations
2//!
3//! Efficient tracking of genetic lineage without unnecessary allocations.
4//! Provides copy-on-write tracking for genetic inheritance.
5
6use serde::{Deserialize, Serialize};
7use std::collections::HashMap;
8
9/// Tracks genetic lineage with copy-on-write semantics
10#[derive(Debug, Clone, Serialize, Deserialize)]
11pub struct LineageTracker {
12    /// Parent-child relationships
13    pub lineage_map: HashMap<String, Vec<String>>,
14}
15
16impl Default for LineageTracker {
17    fn default() -> Self {
18        Self::new()
19    }
20}
21
22impl LineageTracker {
23    /// Create new lineage tracker
24    pub fn new() -> Self {
25        Self {
26            lineage_map: HashMap::new(),
27        }
28    }
29
30    /// Track spawning from parent to child
31    pub fn track_spawn(&mut self, parent_id: String, child_id: String) {
32        self.lineage_map
33            .entry(parent_id)
34            .or_default()
35            .push(child_id);
36    }
37
38    /// Get children of a genetic entity
39    pub fn get_children(&self, parent_id: &str) -> Vec<String> {
40        self.lineage_map
41            .get(parent_id)
42            .cloned()
43            .unwrap_or_else(Vec::new)
44    }
45
46    /// Get lineage depth (generation count)
47    pub fn get_lineage_depth(&self, entity_id: &str) -> u32 {
48        let mut depth = 0;
49        let mut current_children = self.get_children(entity_id);
50
51        while !current_children.is_empty() {
52            depth += 1;
53            let mut next_generation = Vec::new();
54
55            for child_id in current_children {
56                let mut child_children = self.get_children(&child_id);
57                next_generation.append(&mut child_children);
58            }
59
60            current_children = next_generation;
61
62            // Prevent infinite loops in case of circular references
63            if depth > 100 {
64                break;
65            }
66        }
67
68        depth
69    }
70
71    /// Get total descendants count
72    pub fn get_descendant_count(&self, entity_id: &str) -> u32 {
73        let mut count = 0;
74        let mut to_process = vec![entity_id.to_string()];
75
76        while let Some(current_id) = to_process.pop() {
77            if let Some(children) = self.lineage_map.get(&current_id) {
78                count += children.len() as u32;
79                for child_id in children {
80                    to_process.push(child_id.clone());
81                }
82            }
83        }
84
85        count
86    }
87
88    /// Clear lineage data to free memory
89    pub fn clear(&mut self) {
90        self.lineage_map.clear();
91    }
92
93    /// Get lineage statistics
94    pub fn get_stats(&self) -> LineageStats {
95        let total_entities = self.lineage_map.len();
96        let total_relationships: usize = self.lineage_map.values().map(|v| v.len()).sum();
97
98        let max_children = self
99            .lineage_map
100            .values()
101            .map(|v| v.len())
102            .max()
103            .unwrap_or(0);
104
105        LineageStats {
106            total_entities,
107            total_relationships,
108            max_children,
109        }
110    }
111}
112
113/// Statistics about genetic lineage tracking
114#[derive(Debug, Clone, Serialize, Deserialize)]
115pub struct LineageStats {
116    /// Total number of entities being tracked
117    pub total_entities: usize,
118    /// Total number of parent-child relationships
119    pub total_relationships: usize,
120    /// Maximum children any single entity has
121    pub max_children: usize,
122}
