1//! Genetic Analysis Components
2//!
3//! Provides fitness analysis and population analysis functionality.
4
5use beardog_auth::auth::NodeCapability;
6use std::collections::HashMap;
7
8/// Cached fitness analysis to avoid recomputation
9#[derive(Debug, Clone)]
10pub struct CachedFitnessAnalysis {
11    pub fitness_score: f64,
12    pub generation: u32,
13    pub capabilities: Vec<NodeCapability>,
14    pub genetic_hash: String,
15    pub timestamp: chrono::DateTime<chrono::Utc>,
16}
17
18/// Analysis results for a chunk of genetic data
19#[derive(Debug, Default)]
20pub struct ChunkAnalysis {
21    pub nodes_analyzed: usize,
22    pub fitness_sum: f64,
23    pub generation_sum: u64,
24    pub max_generation: u32,
25    pub capabilities_distribution: HashMap<NodeCapability, u32>,
26}
27
28/// Complete population analysis results
29#[derive(Debug, Default)]
30pub struct PopulationAnalysis {
31    pub total_nodes: usize,
32    pub average_fitness: f64,
33    pub average_generation: f64,
34    pub max_generation: u32,
35    pub diversity_index: f64,
36    pub capabilities_distribution: HashMap<NodeCapability, u32>,
37}
38
39impl ChunkAnalysis {
40    /// Create new empty chunk analysis
41    pub fn new() -> Self {
42        Self::default()
43    }
44
45    /// Merge this analysis with another chunk
46    pub fn merge(&mut self, other: &ChunkAnalysis) {
47        self.nodes_analyzed += other.nodes_analyzed;
48        self.fitness_sum += other.fitness_sum;
49        self.generation_sum += other.generation_sum;
50        self.max_generation = self.max_generation.max(other.max_generation);
51
52        // Merge capability distributions
53        for (capability, count) in &other.capabilities_distribution {
54            *self
55                .capabilities_distribution
56                .entry(capability.clone())
57                .or_insert(0) += count;
58        }
59    }
60}
61
62impl PopulationAnalysis {
63    /// Create new population analysis from chunk analyses
64    pub fn from_chunks(chunks: &[ChunkAnalysis]) -> Self {
65        let total_nodes: usize = chunks.iter().map(|c| c.nodes_analyzed).sum();
66        let fitness_sum: f64 = chunks.iter().map(|c| c.fitness_sum).sum();
67        let generation_sum: u64 = chunks.iter().map(|c| c.generation_sum).sum();
68        let max_generation = chunks.iter().map(|c| c.max_generation).max().unwrap_or(0);
69
70        let mut capabilities_distribution = HashMap::new();
71        for chunk in chunks {
72            for (capability, count) in &chunk.capabilities_distribution {
73                *capabilities_distribution
74                    .entry(capability.clone())
75                    .or_insert(0) += count;
76            }
77        }
78
79        // Calculate diversity index based on capability distribution
80        let diversity_index =
81            Self::calculate_diversity_index(&capabilities_distribution, total_nodes);
82
83        Self {
84            total_nodes,
85            average_fitness: if total_nodes > 0 {
86                fitness_sum / total_nodes as f64
87            } else {
88                0.0
89            },
90            average_generation: if total_nodes > 0 {
91                generation_sum as f64 / total_nodes as f64
92            } else {
93                0.0
94            },
95            max_generation,
96            diversity_index,
97            capabilities_distribution,
98        }
99    }
100
101    /// Calculate genetic diversity index
102    fn calculate_diversity_index(distribution: &HashMap<NodeCapability, u32>, total: usize) -> f64 {
103        if total == 0 {
104            return 0.0;
105        }
106
107        // Shannon diversity index
108        let mut diversity = 0.0;
109        for count in distribution.values() {
110            if *count > 0 {
111                let p = *count as f64 / total as f64;
112                diversity -= p * p.ln();
113            }
114        }
115        diversity
116    }
117}
