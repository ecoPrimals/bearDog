1//! Genetic Lineage Tracking
2//!
3//! Provides efficient tracking of genetic lineages with minimal overhead.
4
5use std::collections::HashMap;
6use tracing::debug;
7
8/// Tracks genetic lineages with zero-copy optimizations
9#[derive(Debug)]
10pub struct LineageTracker {
11    /// Lineage records using Arc for sharing
12    lineages: HashMap<String, Vec<String>>,
13    /// Lineage statistics
14    stats: LineageStats,
15}
16
17impl Default for LineageTracker {
18    fn default() -> Self {
19        Self::new()
20    }
21}
22
23impl LineageTracker {
24    pub fn new() -> Self {
25        Self {
26            lineages: HashMap::new(),
27            stats: LineageStats::default(),
28        }
29    }
30
31    /// Track genetic lineage
32    pub fn track_lineage(&mut self, child_id: &str, parent_ids: &[String]) {
33        self.lineages
34            .insert(child_id.to_string(), parent_ids.to_vec());
35        self.stats.total_lineages += 1;
36    }
37
38    /// Get lineage for genetics
39    pub fn get_lineage(&self, genetics_id: &str) -> Option<&Vec<String>> {
40        self.lineages.get(genetics_id)
41    }
42
43    /// Add child to lineage
44    pub fn add_child_to_lineage(&mut self, parent_id: &str, child_id: &str) {
45        if let Some(lineage) = self.lineages.get_mut(parent_id) {
46            if !lineage.contains(&child_id.to_string()) {
47                lineage.push(child_id.to_string());
48            }
49        }
50    }
51
52    /// Update lineage statistics
53    pub fn update_stats(&mut self) {
54        self.stats.total_lineages = self.lineages.len();
55
56        if !self.lineages.is_empty() {
57            let total_children: usize = self.lineages.values().map(|children| children.len()).sum();
58            self.stats.average_children_per_parent =
59                total_children as f64 / self.lineages.len() as f64;
60        }
61    }
62
63    /// Clear old lineage records
64    pub fn cleanup_old_lineages(&mut self) {
65        // In a real implementation, this would remove old lineage records
66        // For now, just update stats
67        self.update_stats();
68
69        // Create simple lineage tracking
70        let _lineage_data: std::collections::HashMap<String, String> =
71            std::collections::HashMap::new();
72        // Store lineage data would go here
73        debug!("Cleaned up old lineage records");
74    }
75
76    /// Get current statistics
77    pub fn get_stats(&self) -> &LineageStats {
78        &self.stats
79    }
80}
81
82/// Lineage tracking statistics
83#[derive(Debug, Default, Clone)]
84pub struct LineageStats {
85    pub total_lineages: usize,
86    pub max_generation: u32,
87    pub average_children_per_parent: f64,
88}
