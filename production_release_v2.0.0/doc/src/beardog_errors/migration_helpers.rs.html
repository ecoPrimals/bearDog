9/// Helper macro for creating operation outcomes
10#[macro_export]
11macro_rules! operation_outcome {
12    ($result:expr, $component:expr) => {
13        $crate::improved_results::OperationOutcome {
14            result: $result,
15            context: $crate::improved_results::OperationContext::new($component),
16            metrics: $crate::improved_results::OperationMetrics::default(),
17            warnings: vec![],
18        }
19    };
20    ($result:expr, $component:expr, $metadata:expr) => {
21        $crate::improved_results::OperationOutcome {
22            result: $result,
23            context: $crate::improved_results::OperationContext::with_metadata(
24                $component, $metadata,
25            ),
26            metrics: $crate::improved_results::OperationMetrics::default(),
27            warnings: vec![],
28        }
29    };
30}

244/// Helper for creating processing outcomes
245pub fn create_processing_outcome<T: Clone>(
246    results: T,
247    total_items: usize,
248    successful_items: usize,
249    failures: Vec<crate::improved_results::ProcessingFailure>,
250) -> crate::improved_results::ProcessingOutcome<T> {
251    let now = Utc::now();
252    let failed_items = failures.len();
253    let success_rate = if total_items > 0 {
254        (successful_items as f64 / total_items as f64) * 100.0
255    } else {
256        100.0
257    };
258
259    crate::improved_results::ProcessingOutcome {
260        result: results.clone(),
261        failures: failures.clone(),
262        statistics: crate::improved_results::ProcessingStatistics {
263            total_items,
264            successful_items,
265            failed_items,
266            success_rate,
267            avg_processing_time: std::time::Duration::from_millis(10),
268            skipped_items: 0,
269            processing_rate_per_second: total_items as f64 / 1.0, // 1 second default
270            average_item_processing_time: std::time::Duration::from_millis(10),
271        },
272        context: crate::improved_results::OperationContext {
273            operation_id: uuid::Uuid::new_v4().to_string(),
274            started_at: now,
275            completed_at: now,
276            component: "beardog-core".to_string(),
277            initiator: "system".to_string(),
278            request_id: None,
279            metadata: HashMap::new(),
280        },
281        metrics: crate::improved_results::OperationMetrics::default(),
282        warnings: vec![],
283        items: vec![results.clone()],
284        failed_items: failures,
285        configuration: crate::improved_results::ProcessingConfiguration {
286            name: "default".to_string(),
287            description: "Default processing configuration".to_string(),
288            batch_size: 100,
289            max_retries: 3,
290            timeout_seconds: 300,
291            parallel_processing: true,
292        },
293    }
294}

347/// Migration trait for easy conversion from unit results
348pub trait UnitResultExt<E> {
349    /// Convert Result<(), E> to Result<OperationOutcome<OperationSummary>, E>
350    fn to_operation_outcome(
351        self,
352        component: &str,
353        operation: &str,
354    ) -> Result<
355        crate::improved_results::OperationOutcome<crate::improved_results::OperationSummary>,
356        E,
357    >;
358}
359
360impl<E> UnitResultExt<E> for Result<(), E> {
361    fn to_operation_outcome(
362        self,
363        component: &str,
364        operation: &str,
365    ) -> Result<
366        crate::improved_results::OperationOutcome<crate::improved_results::OperationSummary>,
367        E,
368    > {
369        self.map(|_| {
370            operation_outcome!(
371                create_operation_summary(operation, "completed", 1),
372                component
373            )
374        })
375    }
376}

378/// Migration trait for converting existing outcomes
379pub trait OutcomeExt<T> {
380    /// Add timing information to an outcome
381    fn with_timing(self, start_time: chrono::DateTime<Utc>) -> Self;
382
383    /// Add metadata to an outcome
384    fn with_metadata(self, metadata: HashMap<String, serde_json::Value>) -> Self;
385
386    /// Add warnings to an outcome
387    fn with_warnings(self, warnings: Vec<crate::improved_results::OperationWarning>) -> Self;
388}
389
390impl<T> OutcomeExt<T> for crate::improved_results::OperationOutcome<T> {
391    fn with_timing(mut self, start_time: chrono::DateTime<Utc>) -> Self {
392        let end_time = Utc::now();
393        self.context.started_at = start_time;
394        self.context.completed_at = end_time;
395        self.metrics.duration = (end_time - start_time).to_std().unwrap_or_default();
396        self
397    }
398
399    fn with_metadata(mut self, metadata: HashMap<String, serde_json::Value>) -> Self {
400        self.context.metadata = metadata;
401        self
402    }
403
404    fn with_warnings(mut self, warnings: Vec<crate::improved_results::OperationWarning>) -> Self {
405        self.warnings = warnings;
406        self
407    }
408}
