80/// **Provider Type Classification**
81///
82/// Standardized classification of `HSM` provider types.
83#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
84pub enum ProviderType {
85    /// `Android` `StrongBox` hardware security
86    AndroidStrongBox,
87    /// `iOS` Secure Enclave hardware security
88    IosSecureEnclave,
89    /// Software-based `HSM` implementation
90    Software,
91    /// PKCS#11 compliant hardware security module
92    Pkcs11,
93    /// Trusted Platform Module 2.0
94    Tpm,
95    /// Cloud-based `HSM` service
96    Cloud,
97    /// Custom/proprietary `HSM` implementation
98    Custom,
99}
100
101impl std::fmt::Display for ProviderType {
102    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
103        match self {
104            ProviderType::AndroidStrongBox => write!(f, "`Android` `StrongBox`"),
105            ProviderType::IosSecureEnclave => write!(f, "`iOS` Secure Enclave"),
106            ProviderType::Software => write!(f, "Software `HSM`"),
107            ProviderType::Pkcs11 => write!(f, "PKCS#11 `HSM`"),
108            ProviderType::Tpm => write!(f, "`TPM` 2.0"),
109            ProviderType::Cloud => write!(f, "Cloud `HSM`"),
110            ProviderType::Custom => write!(f, "Custom `HSM`"),
111        }
112    }
113}

212/// **Operation Result - Universal Standard**
213///
214/// Standardized result structure for `HSM` operations.
215#[derive(Debug, Clone)]

217    /// Operation result data
218    pub data: T,

223/// **Operation Metadata**
224///
225/// Metadata about `HSM` operations for auditing and monitoring.
226#[derive(Debug, Clone)]
227pub struct OperationMetadata {
228    /// Unique operation identifier
229    pub operation_id: String,
230    /// Provider that performed the operation
231    pub provider_id: String,
232    /// Operation type
233    pub operation_type: String,
234    /// Timestamp when operation started
235    pub started_at: DateTime<Utc>,
236    /// Timestamp when operation completed
237    pub completed_at: DateTime<Utc>,
238    /// Operation duration in milliseconds
239    pub duration_ms: f64,
240    /// Whether operation was successful
241    pub success: bool,
242    /// Error message (if operation failed)
243    pub error_message: Option<String>,
244    /// Additional operation-specific data
245    pub additional_data: HashMap<String, String>,
246}
247
248impl<T> OperationResult<T> {
249    /// Create a successful operation result
250    pub fn success(
251        data: T,
252        provider_id: String,
253        operation_type: String,
254        started_at: DateTime<Utc>,
255    ) -> Self {
256        let completed_at = Utc::now();
257        let duration_ms = (completed_at - started_at).num_milliseconds() as f64;
258
259        Self {
260            data,
261            metadata: OperationMetadata {
262                operation_id: uuid::Uuid::new_v4().to_string(),
263                provider_id,
264                operation_type,
265                started_at,
266                completed_at,
267                duration_ms,
268                success: true,
269                error_message: None,
270                additional_data: HashMap::new(),
271            },
272        }
273    }
274
275    /// Create a failed operation result
276    pub fn failure(
277        error: String,
278        _provider_id: String,
279        _operation_type: String,
280        _started_at: DateTime<Utc>,
281    ) -> BearDogResult<Self> {
282        Err(BearDogError::Hsm { message: error })
283    }
284}
