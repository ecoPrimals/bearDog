5use serde::{Deserialize, Serialize};

7/// Compliance requirement types
8#[derive(Debug, Clone, Serialize, Deserialize)]
9pub enum ComplianceType {
10    /// General Data Protection Regulation (European Union)
11    ///
12    /// Protects personal data in research involving European subjects
13    /// or institutions, ensuring privacy rights for international collaboration.
14    GDPR,
15
16    /// Health Insurance Portability and Accountability Act (United States)
17    ///
18    /// Protects healthcare data in medical research, ensuring patient
19    /// privacy in biological and health-related academic studies.
20    HIPAA,
21
22    /// Sarbanes-Oxley Act (United States)
23    ///
24    /// Protects financial data in economic research, ensuring proper
25    /// handling of financial information in business and economic studies.
26    SOX,
27
28    /// National data sovereignty requirements
29    ///
30    /// Ensures research data remains within specific national boundaries
31    /// to comply with data residency laws and protect national interests.
32    DataSovereignty,
33
34    /// Export control restrictions on sensitive research
35    ///
36    /// Prevents unauthorized international sharing of research that could
37    /// have national security implications or dual-use applications.
38    ExportControl,
39}
40
41/// Geographic regions for compliance
42#[derive(Debug, Clone, Serialize, Deserialize)]
43pub enum GeographicRegion {
44    /// European Union and associated regions
45    ///
46    /// Includes EU member states and regions following European
47    /// data protection and academic collaboration frameworks.
48    EU,
49
50    /// United States and territories
51    ///
52    /// Includes regions following US regulatory frameworks and
53    /// federal research compliance requirements.
54    US,
55
56    /// China and associated regions
57    ///
58    /// Requires careful handling due to data sovereignty laws and
59    /// potential export control restrictions on research collaboration.
60    China,
61
62    /// Russia and associated regions
63    ///
64    /// Requires special consideration for geopolitical tensions and
65    /// restrictions on academic and research collaboration.
66    Russia,
67
68    /// Other geographic regions not explicitly categorized
69    ///
70    /// Covers emerging regions with specific requirements for
71    /// academic collaboration and research data handling.
72    Other(String),
73}

75/// Routing restrictions for compliance
76#[derive(Debug, Clone, Serialize, Deserialize)]
77pub enum RoutingRestriction {
78    /// Prevent data from being stored in external cloud services
79    ///
80    /// Ensures sensitive research data remains on controlled infrastructure
81    /// rather than potentially unsecured third-party cloud storage.
82    NoCloudStorage,
83
84    /// Keep all traffic within national or regional boundaries
85    ///
86    /// Prevents research data from crossing international borders
87    /// to comply with data sovereignty and export control requirements.
88    NoInternationalRouting,
89
90    /// Mandate encryption for all data transmission
91    ///
92    /// Ensures that sensitive research data is cryptographically protected
93    /// during transmission to prevent interception or tampering.
94    EncryptionRequired,
95
96    /// Require detailed logging of all data access and transmission
97    ///
98    /// Creates audit trails for sensitive research data to meet compliance
99    /// requirements and enable forensic analysis if needed.
100    AuditTrailRequired,
101
102    /// Restrict data to specific geographic regions
103    ///
104    /// Ensures research data remains within specified jurisdictions
105    /// to comply with data residency laws and institutional policies.
106    DataResidency(GeographicRegion),
107}
