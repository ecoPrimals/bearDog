1//! A scalable reader-writer lock.
2//!
3//! This implementation makes read operations faster and more scalable due to less contention,
4//! while making write operations slower. It also incurs much higher memory overhead than
5//! traditional reader-writer locks.
6

19/// A scalable read-writer lock.
20///
21/// This type of lock allows a number of readers or at most one writer at any point in time. The
22/// write portion of this lock typically allows modification of the underlying data (exclusive
23/// access) and the read portion of this lock typically allows for read-only access (shared
24/// access).
25///
26/// This reader-writer lock differs from typical implementations in that it internally creates a
27/// list of reader-writer locks called 'shards'. Shards are aligned and padded to the cache line
28/// size.
29///
30/// Read operations lock only one shard specific to the current thread, while write operations lock
31/// every shard in succession. This strategy makes concurrent read operations faster due to less
32/// contention, but write operations are slower due to increased amount of locking.
33pub struct RwLock<T> {
34    /// A list of locks protecting the internal data.
35    shards: Vec<CachePadded<parking_lot::RwLock<()>>>,
36
37    /// The internal data.
38    value: UnsafeCell<T>,
39}
40
41unsafe impl<T: Send> Send for RwLock<T> {}
42unsafe impl<T: Send + Sync> Sync for RwLock<T> {}
43
44impl<T> RwLock<T> {
45    /// Creates a new `RwLock` initialized with `value`.
46    pub fn new(value: T) -> RwLock<T> {
47        // The number of shards is a power of two so that the modulo operation in `read` becomes a
48        // simple bitwise "and".
49        let num_shards = num_cpus::get().next_power_of_two();
50
51        RwLock {
52            shards: (0..num_shards)
53                .map(|_| CachePadded::new(parking_lot::RwLock::new(())))
54                .collect(),
55            value: UnsafeCell::new(value),
56        }
57    }
58
59    /// Locks this `RwLock` with shared read access, blocking the current thread until it can be
60    /// acquired.
61    ///
62    /// The calling thread will be blocked until there are no more writers which hold the lock.
63    /// There may be other readers currently inside the lock when this method returns. This method
64    /// does not provide any guarantees with respect to the ordering of whether contentious readers
65    /// or writers will acquire the lock first.
66    ///
67    /// Returns an RAII guard which will release this thread's shared access once it is dropped.
68    pub fn read(&self) -> RwLockReadGuard<T> {
69        // Take the current thread index and map it to a shard index. Thread indices will tend to
70        // distribute shards among threads equally, thus reducing contention due to read-locking.
71        let shard_index = thread_index() & (self.shards.len() - 1);
72
73        RwLockReadGuard {
74            parent: self,
75            _guard: self.shards[shard_index].read(),
76            _marker: PhantomData,
77        }
78    }
79
80    /// Locks this rwlock with exclusive write access, blocking the current thread until it can be
81    /// acquired.
82    ///
83    /// This function will not return while other writers or other readers currently have access to
84    /// the lock.
85    ///
86    /// Returns an RAII guard which will drop the write access of this rwlock when dropped.
87    pub fn write(&self) -> RwLockWriteGuard<T> {
88        // Write-lock each shard in succession.
89        for shard in &self.shards {
90            // The write guard is forgotten, but the lock will be manually unlocked in `drop`.
91            mem::forget(shard.write());
92        }
93
94        RwLockWriteGuard {
95            parent: self,
96            _marker: PhantomData,
97        }
98    }
99}

101/// A guard used to release the shared read access of a `RwLock` when dropped.
102pub struct RwLockReadGuard<'a, T: 'a> {
103    parent: &'a RwLock<T>,
104    _guard: parking_lot::RwLockReadGuard<'a, ()>,
105    _marker: PhantomData<parking_lot::RwLockReadGuard<'a, T>>,
106}
107
108unsafe impl<'a, T: Sync> Sync for RwLockReadGuard<'a, T> {}
109
110impl<'a, T> Deref for RwLockReadGuard<'a, T> {
111    type Target = T;
112
113    fn deref(&self) -> &T {
114        unsafe { &*self.parent.value.get() }
115    }
116}

118/// A guard used to release the exclusive write access of a `RwLock` when dropped.
119pub struct RwLockWriteGuard<'a, T: 'a> {
120    parent: &'a RwLock<T>,
121    _marker: PhantomData<parking_lot::RwLockWriteGuard<'a, T>>,
122}
123
124unsafe impl<'a, T: Sync> Sync for RwLockWriteGuard<'a, T> {}
125
126impl<'a, T> Drop for RwLockWriteGuard<'a, T> {
127    fn drop(&mut self) {
128        // Unlock the shards in reverse order of locking.
129        for shard in self.parent.shards.iter().rev() {
130            unsafe {
131                shard.force_unlock_write();
132            }
133        }
134    }
135}
136
137impl<'a, T> Deref for RwLockWriteGuard<'a, T> {
138    type Target = T;
139
140    fn deref(&self) -> &T {
141        unsafe { &*self.parent.value.get() }
142    }
143}
144
145impl<'a, T> DerefMut for RwLockWriteGuard<'a, T> {
146    fn deref_mut(&mut self) -> &mut T {
147        unsafe { &mut *self.parent.value.get() }
148    }
149}

151/// Returns a `usize` that identifies the current thread.
152///
153/// Each thread is associated with an 'index'. Indices usually tend to be consecutive numbers
154/// between 0 and the number of running threads, but there are no guarantees. During TLS teardown
155/// the associated index might change.
156#[inline]
157pub fn thread_index() -> usize {
158    REGISTRATION.try_with(|reg| reg.index).unwrap_or(0)
159}
160
161/// The global registry keeping track of registered threads and indices.
162struct ThreadIndices {
163    /// Mapping from `ThreadId` to thread index.
164    mapping: HashMap<ThreadId, usize>,
165
166    /// A list of free indices.
167    free_list: Vec<usize>,
168
169    /// The next index to allocate if the free list is empty.
170    next_index: usize,
171}
172
173lazy_static! {
174    static ref THREAD_INDICES: Mutex<ThreadIndices> = Mutex::new(ThreadIndices {
175        mapping: HashMap::new(),
176        free_list: Vec::new(),
177        next_index: 0,
178    });
179}
180
181/// A registration of a thread with an index.
182///
183/// When dropped, unregisters the thread and frees the reserved index.
184struct Registration {
185    index: usize,
186    thread_id: ThreadId,
187}
188
189impl Drop for Registration {
190    fn drop(&mut self) {
191        let mut indices = THREAD_INDICES.lock().unwrap();
192        indices.mapping.remove(&self.thread_id);
193        indices.free_list.push(self.index);
194    }
195}
196
197thread_local! {
198    static REGISTRATION: Registration = {
199        let thread_id = thread::current().id();
200        let mut indices = THREAD_INDICES.lock().unwrap();
201
202        let index = match indices.free_list.pop() {
203            Some(i) => i,
204            None => {
205                let i = indices.next_index;
206                indices.next_index += 1;
207                i
208            }
209        };
210        indices.mapping.insert(thread_id, index);
211
212        Registration {
213            index,
214            thread_id,
215        }
216    };
217}
