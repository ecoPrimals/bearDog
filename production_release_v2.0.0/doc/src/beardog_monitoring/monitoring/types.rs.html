6/// Metric value types that can be recorded in the monitoring system
7#[derive(Debug, Clone, Serialize, Deserialize)]
8pub enum MetricValue {
9    /// Counter metric that only increases
10    Counter(u64),
11    /// Gauge metric that can go up or down
12    Gauge(f64),
13    /// Histogram metric containing multiple values
14    Histogram(Vec<f64>),
15    /// Timer metric representing duration
16    Timer(Duration),
17    /// String metric for categorical data
18    String(String),
19}

21/// System health status levels
