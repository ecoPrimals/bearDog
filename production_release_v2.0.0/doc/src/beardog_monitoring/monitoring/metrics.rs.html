73    /// Record metric (always free - native Rust)
74    pub async fn record_metric(&self, name: &str, value: MetricValue) {
75        let mut metrics = self.native_metrics.write().await;
76        metrics.insert(name.to_string(), value);
77
78        // Update internal collector timestamp
79        *self.internal_collector.last_updated.write().await = Utc::now();

116    /// Enable Prometheus export (requires license)
117    pub async fn enable_prometheus_export(
118        &mut self,

121        // Check license for Prometheus (external system)

131                message: "Prometheus export requires a BearDog license. Native Rust metrics are always free. Contact sales for enterprise Prometheus integration.".to_string()

136            enabled: true,

145    /// Export metrics to Prometheus (licensed feature)
146    pub async fn export_to_prometheus(&self) -> BearDogResult<String> {
147        if let Some(exporter) = &self.prometheus_exporter {
148            if !exporter.enabled {
149                return Err(BearDogError::Configuration {
150                    message: "Prometheus export is not enabled".to_string(),
151                });
152            }
153
154            // Generate Prometheus format
155            let summary = self.get_internal_summary().await;
156            let prometheus_output = format!(
157                "# HELP beardog_security_events_total Total security events processed\n\
158                 # TYPE beardog_security_events_total counter\n\
159                 beardog_security_events_total {}\n\
160                 # HELP beardog_encryption_operations_total Total encryption operations\n\
161                 # TYPE beardog_encryption_operations_total counter\n\
162                 beardog_encryption_operations_total {}\n\
163                 # HELP beardog_threat_detections_total Total threats detected\n\
164                 # TYPE beardog_threat_detections_total counter\n\
165                 beardog_threat_detections_total {}\n\
166                 # HELP beardog_compliance_checks_total Total compliance checks\n\
167                 # TYPE beardog_compliance_checks_total counter\n\
168                 beardog_compliance_checks_total {}\n\
169                 # HELP beardog_api_requests_total Total API requests\n\
170                 # TYPE beardog_api_requests_total counter\n\
171                 beardog_api_requests_total {}\n\
172                 # HELP beardog_errors_total Total errors\n\
173                 # TYPE beardog_errors_total counter\n\
174                 beardog_errors_total {}\n\
175                 # HELP beardog_active_sessions Current active sessions\n\
176                 # TYPE beardog_active_sessions gauge\n\
177                 beardog_active_sessions {}\n",
178                summary.security_events,
179                summary.encryption_operations,
180                summary.threat_detections,
181                summary.compliance_checks,
182                summary.api_requests,
183                summary.error_count,
184                summary.active_sessions
185            );
186
187            Ok(prometheus_output)
188        } else {
189            Err(BearDogError::Configuration {
190                message: "Prometheus export not configured. Use native metrics (free) or obtain a license for Prometheus integration.".to_string()
191            })
192        }
193    }

244    /// Validate monitoring license for premium features
245    async fn validate_monitoring_license(&self) -> BearDogResult<()> {
246        debug!("🔐 Validating monitoring license");
247
248        // Check for license file or environment variable
249        let license_valid = if let Ok(license_key) = std::env::var("BEARDOG_MONITORING_LICENSE") {
250            // Validate license key format
251            !license_key.is_empty() && license_key.len() >= 16
252        } else if std::path::Path::new("/etc/beardog/monitoring.license").exists() {
253            // Check for license file
254            true
255        } else {
256            // No license found, use basic monitoring
257            debug!("No monitoring license found, using basic features");
258            false
259        };
260
261        if license_valid {
262            info!("✅ Monitoring license validated - premium features enabled");
263            Ok(())
264        } else {
265            debug!("Using basic monitoring features without license");
266            Ok(()) // Don't fail, just use basic features
267        }
268    }
