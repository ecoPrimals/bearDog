7//! # Key Features
8//!
9//! * **Rule-Based Detection**: Configurable detection rules and conditions
10//! * **Behavioral Analysis**: User and entity behavior analysis for anomaly detection
11//! * **Machine Learning**: AI-enhanced anomaly detection and pattern recognition
12//! * **Threat Intelligence**: Integration with external threat feeds
13//! * **Automated Response**: Configurable incident response workflows
14//! * **MITRE ATT&CK Mapping**: Threat categorization using MITRE framework

29/// Quick access to threat detection functionality
30pub struct ThreatAPI;
31
32impl ThreatAPI {
33    /// Create a new threat detection engine with default configuration
34    pub async fn create_default() -> beardog_errors::BearDogResult<ThreatDetectionEngine> {
35        ThreatDetectionEngine::new(ThreatDetectionConfig::default()).await
36    }
37
38    /// Create a new ML-powered threat detection engine
39    pub async fn new_with_ml(
40    ) -> beardog_errors::BearDogResult<(ThreatDetectionEngine, MlThreatEngine)> {
41        let threat_engine = ThreatDetectionEngine::new(ThreatDetectionConfig::default()).await?;
42        let ml_engine = MlThreatEngine::new(MlEngineConfig::default()).await?;
43        Ok((threat_engine, ml_engine))
44    }
