1//! Threat detection engine types and structures
2//!
3//! This module contains all the types and structures used by the threat detection engine,
4//! organized into focused sub-modules for better maintainability.
5//!
6//! ## Modules
7//!
8//! - `engine`: Main threat detection engine structure
9//! - `rules`: Detection rule types and management
10//! - `conditions`: Rule condition types and evaluation logic
11//! - `ml_models`: Machine learning model types and metadata
12//!
13//! ## Example Usage
14//!
15//! ```rust
16//! use beardog::threat::types::engine::{
17//!     ThreatDetectionEngine, DetectionRule, RuleCondition, MlModel, MlModelType
18//! };
19//!
20//! // Create a new engine
21//! let engine = ThreatDetectionEngine::default();
22//!
23//! // Create a detection rule
24//! let rule = DetectionRule::new(
25//!     "brute-force-001".to_string(),
26//!     "Brute Force Detection".to_string(),
27//!     "Detects brute force login attempts".to_string(),
28//!     RuleCondition::frequency_threshold(10, 5),
29//!     ThreatType::BruteForce,
30//!     ThreatSeverity::High,
31//! );
32//!
33//! // Create an ML model
34//! let model = MlModel::new(
35//!     "anomaly-detector-v1".to_string(),
36//!     "Network Anomaly Detector".to_string(),
37//!     MlModelType::AnomalyDetection,
38//!     vec!["packet_count".to_string(), "byte_count".to_string()],
39//! );
40//! ```
41

43#[allow(clippy::module_inception)]
44pub mod engine;
45pub mod ml_models;
46pub mod rules;
