1//! Incident status management
2//!
3//! This module contains types and functionality for managing incident status
4//! throughout the incident lifecycle.
5
6use serde::{Deserialize, Serialize};
7
8/// Incident status enumeration
9///
10/// Tracks the current state of an incident through
11/// its lifecycle from detection to closure.
12#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
13pub enum IncidentStatus {
14    /// Incident opened and awaiting initial response
15    Open,
16
17    /// Incident under active investigation
18    InProgress,
19
20    /// Incident contained but not yet resolved
21    Contained,
22
23    /// Incident resolved and systems restored
24    Resolved,
25
26    /// Incident closed after post-incident review
27    Closed,
28
29    /// Incident escalated to higher security level
30    Escalated,
31}
32
33impl IncidentStatus {
34    /// Check if status is terminal
35    ///
36    /// # Returns
37    /// `true` if status represents a final state
38    ///
39    /// # Example
40    /// ```rust
41    /// use beardog::threat::types::IncidentStatus;
42    ///
43    /// assert!(IncidentStatus::Closed.is_terminal());
44    /// assert!(!IncidentStatus::Open.is_terminal());
45    /// ```
46    pub fn is_terminal(&self) -> bool {
47        matches!(self, IncidentStatus::Closed)
48    }
49
50    /// Get valid next statuses
51    ///
52    /// # Returns
53    /// Vector of valid next status transitions
54    ///
55    /// # Example
56    /// ```rust
57    /// use beardog::threat::types::IncidentStatus;
58    ///
59    /// let next_statuses = IncidentStatus::Open.valid_next_statuses();
60    /// assert!(next_statuses.contains(&IncidentStatus::InProgress));
61    /// ```
62    pub fn valid_next_statuses(&self) -> Vec<IncidentStatus> {
63        match self {
64            IncidentStatus::Open => vec![IncidentStatus::InProgress, IncidentStatus::Escalated],
65            IncidentStatus::InProgress => vec![
66                IncidentStatus::Contained,
67                IncidentStatus::Resolved,
68                IncidentStatus::Escalated,
69            ],
70            IncidentStatus::Contained => vec![IncidentStatus::Resolved, IncidentStatus::Escalated],
71            IncidentStatus::Resolved => vec![IncidentStatus::Closed],
72            IncidentStatus::Closed => vec![],
73            IncidentStatus::Escalated => vec![
74                IncidentStatus::InProgress,
75                IncidentStatus::Contained,
76                IncidentStatus::Resolved,
77            ],
78        }
79    }
80
81    /// Check if transition is valid
82    ///
83    /// # Arguments
84    /// * `new_status` - Target status
85    ///
86    /// # Returns
87    /// `true` if transition is valid
88    ///
89    /// # Example
90    /// ```rust
91    /// use beardog::threat::types::IncidentStatus;
92    ///
93    /// let current = IncidentStatus::Open;
94    /// assert!(current.can_transition_to(&IncidentStatus::InProgress));
95    /// assert!(!current.can_transition_to(&IncidentStatus::Closed));
96    /// ```
97    pub fn can_transition_to(&self, new_status: &IncidentStatus) -> bool {
98        self.valid_next_statuses().contains(new_status)
99    }
100}
101
102impl std::fmt::Display for IncidentStatus {
103    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
104        match self {
105            IncidentStatus::Open => write!(f, "Open"),
106            IncidentStatus::InProgress => write!(f, "In Progress"),
107            IncidentStatus::Contained => write!(f, "Contained"),
108            IncidentStatus::Resolved => write!(f, "Resolved"),
109            IncidentStatus::Closed => write!(f, "Closed"),
110            IncidentStatus::Escalated => write!(f, "Escalated"),
111        }
112    }
113}
