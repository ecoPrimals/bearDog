117    /// Remove a detection rule by ID
118    ///
119    /// This method removes a threat detection rule from the engine's rule set
120    /// based on its unique identifier.
121    ///
122    /// # Arguments
123    ///
124    /// * `rule_id` - The unique identifier of the rule to remove
125    ///
126    /// # Returns
127    ///
128    /// * `bool` - True if the rule was found and removed, false otherwise
129    ///
130    /// # Examples
131    ///
132    /// ```rust
133    /// use beardog::threat::handlers::ThreatDetectionEngine;
134    ///
135    /// fn main() {
136    ///     let mut engine = ThreatDetectionEngine::placeholder();
137    ///     
138    ///     let removed = engine.remove_detection_rule("brute_force_rule");
139    ///     if removed {
140    ///         println!("Rule removed successfully");
141    ///     } else {
142    ///         println!("Rule not found");
143    ///     }
144    /// }
145    /// ```
146    ///
147    /// # Impact
148    ///
149    /// Removing a rule will:
150    /// - Stop future threat detection using that rule
151    /// - Not affect existing threats detected by the rule
152    /// - Update rule statistics
153    /// - Log the removal action
154    pub fn remove_detection_rule(&mut self, rule_id: &str) -> bool {
155        let initial_len = self.detection_rules.len();
156        self.detection_rules.retain(|rule| rule.id != rule_id);
157        let removed = self.detection_rules.len() < initial_len;
158
159        if removed {
160            info!("Removed detection rule: {}", rule_id);
161        }
162
163        removed
164    }

166    /// Enable or disable a detection rule
167    ///
168    /// This method enables or disables a threat detection rule without
169    /// removing it from the system. Disabled rules are retained for
170    /// management purposes but do not participate in threat detection.
171    ///
172    /// # Arguments
173    ///
174    /// * `rule_id` - The unique identifier of the rule to modify
175    /// * `enabled` - Whether to enable (true) or disable (false) the rule
176    ///
177    /// # Returns
178    ///
179    /// * `bool` - True if the rule was found and modified, false otherwise
180    ///
181    /// # Examples
182    ///
183    /// ```rust
184    /// use beardog::threat::handlers::ThreatDetectionEngine;
185    ///
186    /// fn main() {
187    ///     let mut engine = ThreatDetectionEngine::placeholder();
188    ///     
189    ///     // Disable a rule temporarily
190    ///     let updated = engine.enable_detection_rule("brute_force_rule", false);
191    ///     if updated {
192    ///         println!("Rule disabled successfully");
193    ///     }
194    ///     
195    ///     // Re-enable the rule
196    ///     let updated = engine.enable_detection_rule("brute_force_rule", true);
197    ///     if updated {
198    ///         println!("Rule enabled successfully");
199    ///     }
200    /// }
201    /// ```
202    ///
203    /// # Use Cases
204    ///
205    /// Rule enable/disable is useful for:
206    /// - Temporarily disabling noisy rules
207    /// - Testing rule modifications
208    /// - Maintenance and troubleshooting
209    /// - Gradual rule rollout
210    pub fn enable_detection_rule(&mut self, rule_id: &str, enabled: bool) -> bool {
211        for rule in &mut self.detection_rules {
212            if rule.id == rule_id {
213                rule.enabled = enabled;
214                let status = if enabled { "enabled" } else { "disabled" };
215                info!("Detection rule {} {}", rule_id, status);
216                return true;
217            }
218        }
219        false
220    }

342    /// # Returns
343    ///
344    /// * `bool` - True if the rule was found and updated, false otherwise
345    ///
346    /// # Examples
347    ///
348    /// ```rust
349    /// use beardog::threat::handlers::ThreatDetectionEngine;
350    /// use beardog::threat::types::*;
351    ///

364    ///     if updated {
365    ///         println!("Rule updated successfully");
366    ///     }
367    /// }
368    /// ```
369    ///
370    /// # Update Considerations
371    ///
372    /// When updating rules:
373    /// - Rule ID must remain constant
374    /// - Performance history is preserved
375    /// - Changes take effect immediately
376    /// - Update is logged for audit purposes
377    /// - Existing threats are not affected
378    pub fn update_detection_rule(
379        &mut self,
380        rule_id: &str,
381        updated_rule: ThreatDetectionRule,
382    ) -> bool {
383        for rule in &mut self.detection_rules {
384            if rule.id == rule_id {
385                *rule = updated_rule;
386                info!("Updated detection rule: {}", rule_id);
387                return true;
388            }
389        }
390        false
391    }

581    ///     Ok(())
582    /// }
583    /// ```
584    ///
585    /// # Maintenance Operations
586    ///
587    /// Maintenance includes:
588    /// - **Data Cleanup**: Remove old events and stale data
589    /// - **Performance Optimization**: Defragment data structures
590    /// - **Statistics Aggregation**: Compile performance metrics
591    /// - **Health Checks**: Verify system component status
592    /// - **Resource Cleanup**: Free unused resources
593    /// - **Configuration Validation**: Check rule and config integrity
594    pub async fn perform_maintenance(&mut self) -> BearDogResult<()> {
595        info!("Starting system maintenance");
596
597        // Clean up old events from history
598        {
599            let mut history = self.event_history.write().await;
600            if history.len() > 5000 {
601                history.drain(0..1000);
602                info!("Cleaned up old event history");
603            }
604        }
605
606        // Clean up resolved incidents
607        {
608            let mut incidents = self.active_incidents.write().await;
609            let initial_count = incidents.len();
610            incidents.retain(|_, incident| incident.status != IncidentStatus::Resolved);
611            let removed = initial_count - incidents.len();
612            if removed > 0 {
613                info!("Cleaned up {} resolved incidents", removed);
614            }
615        }
616
617        // Update rule performance metrics (removed updated_at reference)
618        for _rule in &mut self.detection_rules {
619            // Rule statistics are updated elsewhere
620        }
621
622        info!("System maintenance completed");
623        Ok(())
624    }
