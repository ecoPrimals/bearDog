1//! Threat intelligence feeds management and processing
2//!
3//! This module handles external threat intelligence feeds, indicator processing,
4//! and threat feed-based detection. It provides integration with various threat
5//! intelligence sources and automated threat indicator matching.
6//!
7//! # Features
8//!
9//! - **Feed Management**: Add, update, and manage threat intelligence feeds
10//! - **Indicator Processing**: Process and normalize threat indicators
11//! - **Feed-based Detection**: Match events against threat intelligence indicators
12//! - **Feed Health Monitoring**: Track feed status and update frequencies
13//!
14//! # Supported Indicators
15//!
16//! The system supports various types of threat indicators:
17//! - IP addresses and network ranges
18//! - Domain names and URLs
19//! - File hashes (MD5, SHA1, SHA256)
20//! - Email addresses
21//! - User-agent strings
22//! - Custom indicators
23//!
24//! # Examples
25//!
26//! ```rust
27//! use beardog::threat::handlers::ThreatDetectionEngine;
28//! use beardog::threat::types::*;
29//! use std::collections::HashMap;
30//!
31//! #[tokio::main]
32//! async fn main() -> Result<(), Box<dyn std::error::Error>> {
33//!     let mut engine = ThreatDetectionEngine::placeholder();
34//!     
35//!     // Add a threat intelligence feed
36//!     let feed = ThreatIntelligenceFeed {
37//!         id: "malware_ips".to_string(),
38//!         name: "Malware IP Feed".to_string(),
39//!         source: "example.com".to_string(),
40//!         feed_type: "ip_reputation".to_string(),
41//!         // ... other fields
42//!     };
43//!     engine.add_threat_feed(feed);
44//!     
45//!     // Check event against threat feeds
46//!     let mut event_data = HashMap::new();
47//!     event_data.insert("source_ip".to_string(), "192.168.1.100".to_string());
48//!     
49//!     let feed_threats = engine.check_threat_feeds(&event_data).await?;
50//!     println!("Found {} threats from feeds", feed_threats.len());
51//!     Ok(())
52//! }
53//! ```

64    /// Check event data against threat intelligence feeds
65    ///
66    /// This method processes event data against all configured threat intelligence
67    /// feeds to identify known malicious indicators. It supports various indicator
68    /// types and provides detailed threat context from feed sources.
69    ///
70    /// # Arguments
71    ///
72    /// * `event_data` - A HashMap containing event fields and their values
73    ///
74    /// # Returns
75    ///
76    /// * `BearDogResult<Vec<ThreatEvent>>` - A vector of feed-based threats or error
77    ///
78    /// # Errors
79    ///
80    /// This function will return an error if:
81    /// - Feed processing fails
82    /// - Indicator matching fails
83    /// - Threat event creation fails
84    ///
85    /// # Examples
86    ///
87    /// ```rust
88    /// use beardog::threat::handlers::ThreatDetectionEngine;
89    /// use std::collections::HashMap;
90    ///
91    /// #[tokio::main]
92    /// async fn main() -> Result<(), Box<dyn std::error::Error>> {
93    ///     let engine = ThreatDetectionEngine::placeholder();
94    ///     
95    ///     let mut event_data = HashMap::new();
96    ///     event_data.insert("source_ip".to_string(), "192.168.1.100".to_string());
97    ///     event_data.insert("domain".to_string(), "malicious.example.com".to_string());
98    ///     
99    ///     let threats = engine.check_threat_feeds(&event_data).await?;
100    ///     for threat in threats {
101    ///         println!("Feed threat: {} from {}", threat.description, threat.source.ip_address);
102    ///     }
103    ///     Ok(())
104    /// }
105    /// ```
106    ///
107    /// # Feed Processing
108    ///
109    /// The method processes feeds in the following order:
110    /// 1. Iterate through all active threat intelligence feeds
111    /// 2. Extract relevant indicators from each feed
112    /// 3. Match event data against feed indicators
113    /// 4. Create threat events for positive matches
114    /// 5. Enrich threats with feed-specific context
115    pub async fn check_threat_feeds(
116        &self,
117        event_data: &HashMap<String, String>,
118    ) -> BearDogResult<Vec<ThreatEvent>> {
119        let mut feed_threats = Vec::new();
120
121        for feed in self.threat_feeds.values() {
122            // Check indicators in this feed
123            for indicator in &feed.indicators {
124                if self.matches_indicator(event_data, indicator) {
125                    let threat = self.create_feed_threat_event(feed, indicator, event_data)?;
126                    feed_threats.push(threat);
127                }
128            }
129        }
130
131        Ok(feed_threats)
132    }

134    /// Check if event data matches a threat indicator
135    ///
136    /// This method determines whether event data contains values that match
137    /// a specific threat indicator from intelligence feeds.
138    ///
139    /// # Arguments
140    ///
141    /// * `event_data` - The event data to check
142    /// * `indicator` - The threat indicator to match against
143    ///
144    /// # Returns
145    ///
146    /// * `bool` - True if the event matches the indicator, false otherwise
147    ///
148    /// # Examples
149    ///
150    /// ```rust
151    /// use beardog::threat::handlers::ThreatDetectionEngine;
152    /// use beardog::threat::types::*;
153    /// use std::collections::HashMap;
154    ///
155    /// # fn example() {
156    /// let engine = ThreatDetectionEngine::placeholder();
157    ///
158    /// let mut event_data = HashMap::new();
159    /// event_data.insert("source_ip".to_string(), "192.168.1.100".to_string());
160    ///
161    /// let indicator = ThreatIndicator {
162    ///     id: "malicious_ip".to_string(),
163    ///     indicator_type: "ip_address".to_string(),
164    ///     value: "192.168.1.100".to_string(),
165    ///     // ... other fields
166    /// };
167    ///
168    /// let matches = engine.matches_indicator(&event_data, &indicator);
169    /// println!("Indicator match: {}", matches);
170    /// # }
171    /// ```
172    ///
173    /// # Indicator Types
174    ///
175    /// Supported indicator types and their matching logic:
176    /// - `ip_address`: Exact IP address match
177    /// - `domain`: Domain name match
178    /// - `url`: URL pattern match
179    /// - `file_hash`: File hash match (MD5, SHA1, SHA256)
180    /// - `email`: Email address match
181    /// - `user_agent`: User-agent string match
182    pub fn matches_indicator(
183        &self,
184        event_data: &HashMap<String, String>,
185        indicator: &ThreatIndicator,
186    ) -> bool {
187        match indicator.indicator_type {
188            IndicatorType::IpAddress => {
189                (event_data.get("source_ip") == Some(&indicator.value))
190                    || (event_data.get("dest_ip") == Some(&indicator.value))
191            }
192            IndicatorType::DomainName => {
193                (event_data.get("domain") == Some(&indicator.value))
194                    || event_data
195                        .get("hostname")
196                        .is_some_and(|hostname| hostname.contains(&indicator.value))
197            }
198            IndicatorType::Url => event_data
199                .get("url")
200                .is_some_and(|url| url.contains(&indicator.value)),
201            IndicatorType::FileHash => event_data.get("file_hash") == Some(&indicator.value),
202            IndicatorType::EmailAddress => event_data.get("email") == Some(&indicator.value),
203            IndicatorType::UserAgent => event_data
204                .get("user_agent")
205                .is_some_and(|ua| ua.contains(&indicator.value)),
206            _ => false,
207        }
208    }

220    /// * `event_data` - The event data that matched the indicator
221    ///
222    /// # Returns
223    ///
224    /// * `BearDogResult<ThreatEvent>` - A new threat event or error
225    ///
226    /// # Errors
227    ///
228    /// This function will return an error if:
229    /// - UUID generation fails
230    /// - Threat event creation fails
231    /// - Feed metadata is invalid
232    ///
233    /// # Examples
234    ///
235    /// ```rust
236    /// use beardog::threat::handlers::ThreatDetectionEngine;
237    /// use beardog::threat::types::*;
238    /// use std::collections::HashMap;
239    ///
240    /// fn main() -> Result<(), Box<dyn std::error::Error>> {

242    ///     
243    ///     let feed = ThreatIntelligenceFeed {
244    ///         id: "malware_feed".to_string(),
245    ///         name: "Malware Intelligence Feed".to_string(),
246    ///         source: "security_vendor".to_string(),
247    ///         // ... other fields
248    ///     };
249    ///     
250    ///     let indicator = ThreatIndicator {
251    ///         id: "bad_ip".to_string(),
252    ///         indicator_type: "ip_address".to_string(),
253    ///         value: "192.168.1.100".to_string(),
254    ///         // ... other fields
255    ///     };
256    ///     
257    ///     let mut event_data = HashMap::new();
258    ///     event_data.insert("source_ip".to_string(), "192.168.1.100".to_string());
259    ///     
260    ///     let threat = engine.create_feed_threat_event(&feed, &indicator, &event_data)?;
261    ///     println!("Created feed threat: {}", threat.id);
262    ///     Ok(())
263    /// }
264    /// ```
265    pub fn create_feed_threat_event(
266        &self,
267        feed: &ThreatIntelligenceFeed,
268        indicator: &ThreatIndicator,
269        event_data: &HashMap<String, String>,
270    ) -> BearDogResult<ThreatEvent> {
271        let threat_id = Uuid::new_v4().to_string();
272        let timestamp = Utc::now();
273
274        // Create threat source with feed intelligence
275        let threat_source = ThreatSource {
276            id: Uuid::new_v4().to_string(),
277            ip_address: Some(event_data.get("source_ip").cloned().unwrap_or_default()),
278            hostname: Some(event_data.get("hostname").cloned().unwrap_or_default()),
279            user_agent: Some(event_data.get("user_agent").cloned().unwrap_or_default()),
280            geolocation: None, // Will be processed separately if needed
281            classification: SourceClassification::KnownMalicious,
282            reputation_score: 0.1, // Low reputation for known malicious
283            threat_actor: indicator.threat_actor.clone(),
284            confidence_score: indicator.confidence_score,
285            first_seen: Some(timestamp),
286            last_seen: Some(timestamp),
287            threat_score: indicator.confidence_score * 100.0,
288        };
289
290        // Create threat target
291        let threat_target = ThreatTarget {
292            id: Uuid::new_v4().to_string(),
293            resource_id: event_data
294                .get("resource_id")
295                .cloned()
296                .unwrap_or_else(|| format!("resource-{}", Uuid::new_v4())),
297            resource_type: event_data
298                .get("resource_type")
299                .cloned()
300                .unwrap_or_else(|| "network-endpoint".to_string()),
301            node_id: event_data.get("node_id").cloned(),
302            user_account: event_data.get("user_account").cloned(),
303            criticality: AssetCriticality::High, // Assume high for intelligence matches
304            protection_level: ProtectionLevel::Enhanced,
305            ip_address: Some(event_data.get("target_ip").cloned().unwrap_or_default()),
306            hostname: Some(
307                event_data
308                    .get("target_hostname")
309                    .cloned()
310                    .unwrap_or_default(),
311            ),
312            service: Some(event_data.get("service").cloned().unwrap_or_default()),
313            port: event_data.get("port").and_then(|p| p.parse().ok()),
314            protocol: Some(event_data.get("protocol").cloned().unwrap_or_default()),
315        };
316
317        // Create threat event
318        let threat_event = ThreatEvent {
319            id: threat_id,
320            threat_type: ThreatType::Malicious,
321            severity: ThreatSeverity::High, // Intelligence matches are typically high severity
322            status: ThreatStatus::Active,
323            description: format!(
324                "Threat intelligence match: {} indicator '{}' from feed '{}' - {}",
325                indicator.indicator_type, indicator.value, feed.name, indicator.description
326            ),
327            source: threat_source,
328            target: threat_target,
329            timestamp,
330            score: (indicator.confidence_score * 100.0) as u8,
331            detection_method: DetectionMethod::ThreatIntelligence,
332            evidence: vec![],
333            recommended_actions: vec![ThreatAction::BlockSource, ThreatAction::AlertSecurityTeam],
334            assigned_analyst: None,
335            related_events: vec![],
336            mitigation_steps: vec![],
337        };
338
339        Ok(threat_event)
340    }

373    /// # Feed Management
374    ///
375    /// - Feeds are indexed by their unique ID
376    /// - Adding a feed with an existing ID will replace the previous feed
377    /// - Feeds are immediately available for threat detection
378    /// - Feed health and update status should be monitored separately

380        self.threat_feeds.insert(feed.id.clone(), feed);
