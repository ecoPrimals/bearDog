1//! Distributed caching configuration
2//!
3//! Main configuration types for distributed caching systems.
4
5use super::redis::RedisClusterConfig;
6use super::strategies::{ConflictResolutionStrategy, SynchronizationStrategy};
7// Canonical configuration types imported as needed
8use beardog_types::canonical::configuration::storage::CacheWarmingConfig;
9use beardog_types::canonical::configuration::CachePerformanceConfig;
10use serde::{Deserialize, Serialize};
11use std::time::Duration;
12
13/// Distributed caching configuration
14#[derive(Debug, Clone, Serialize, Deserialize, Default)]
15pub struct DistributedCachingConfig {
16    /// Multi-tier caching moved to canonical CachePerformanceConfig
17    pub multi_tier: CachePerformanceConfig,
18    /// Redis cluster configuration
19    pub redis_cluster: RedisClusterConfig,
20    /// Cache warming configuration
21    pub cache_warming: CacheWarmingConfig,
22    /// Cache synchronization configuration
23    pub synchronization: CacheSynchronizationConfig,
24    /// Performance optimization configuration
25    pub performance: CachePerformanceConfig,
26}
27
28/// Cache synchronization configuration
29#[derive(Debug, Clone, Serialize, Deserialize)]
30pub struct CacheSynchronizationConfig {
31    /// Synchronization strategy
32    pub strategy: SynchronizationStrategy,
33    /// Conflict resolution strategy
34    pub conflict_resolution: ConflictResolutionStrategy,
35    /// Enable cache invalidation
36    pub enable_invalidation: bool,
37    /// Synchronization interval
38    pub sync_interval: Duration,
39}
40
41impl Default for CacheSynchronizationConfig {
42    fn default() -> Self {
43        Self {
44            strategy: SynchronizationStrategy::default(),
45            conflict_resolution: ConflictResolutionStrategy::default(),
46            enable_invalidation: true,
47            sync_interval: Duration::from_secs(30),
48        }
49    }
50}
51
52impl DistributedCachingConfig {
53    /// Create production-ready distributed caching configuration
54    pub fn production() -> Self {
55        Self {
56            multi_tier: CachePerformanceConfig::default(),
57            redis_cluster: RedisClusterConfig::production(),
58            cache_warming: CacheWarmingConfig::default(),
59            synchronization: CacheSynchronizationConfig::default(),
60            performance: CachePerformanceConfig::default(),
61        }
62    }
63
64    /// Create development-friendly distributed caching configuration
65    pub fn development() -> Self {
66        Self {
67            multi_tier: CachePerformanceConfig::default(),
68            redis_cluster: RedisClusterConfig::development(),
69            cache_warming: CacheWarmingConfig::default(),
70            synchronization: CacheSynchronizationConfig::default(),
71            performance: CachePerformanceConfig::default(),
72        }
73    }
74
75    /// Validate the distributed caching configuration
76    pub fn validate(&self) -> Result<(), String> {
77        // Validate cache warming configuration (HashMap-based)
78        if self.cache_warming.enabled && self.cache_warming.strategies.is_empty() {
79            return Err("Cache warming enabled but no strategy configured".to_string());
80        }
81
82        if self.redis_cluster.enabled && self.redis_cluster.nodes.is_empty() {
83            return Err("Redis cluster enabled but no nodes configured".to_string());
84        }
85
86        Ok(())
87    }
88}
