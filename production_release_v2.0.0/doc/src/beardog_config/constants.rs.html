5use std::env;
6use std::path::PathBuf;
7
8/// Get configurable API bind address
9pub fn api_bind_address() -> String {
10    env::var("BEARDOG_API_BIND_ADDRESS").unwrap_or_else(|_| "0.0.0.0:8080".to_string())
11}
12
13/// Get configurable metrics bind address
14pub fn metrics_bind_address() -> String {
15    env::var("BEARDOG_METRICS_BIND_ADDRESS").unwrap_or_else(|_| "0.0.0.0:9090".to_string())
16}
17
18/// Get configurable health bind address
19pub fn health_bind_address() -> String {
20    env::var("BEARDOG_HEALTH_BIND_ADDRESS").unwrap_or_else(|_| "0.0.0.0:8088".to_string())
21}
22
23/// Get configurable admin bind address
24pub fn admin_bind_address() -> String {
25    env::var("BEARDOG_ADMIN_BIND_ADDRESS").unwrap_or_else(|_| "127.0.0.1:9999".to_string())
26}

28/// Get configurable database URL
29pub fn database_url() -> String {

31}

33/// Get configurable log level

38/// Get configurable log format

43/// Get configurable data directory
44pub fn data_dir() -> PathBuf {
