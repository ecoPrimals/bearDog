1//! Test Scenarios and Load Test Types
2//!
3//! This module defines the various test scenarios and load test types
4//! that can be configured for BearDog performance testing.
5
6use serde::{Deserialize, Serialize};
7use std::time::Duration;
8
9use super::operations::*;
10use super::profiles::*;
11
12/// Test scenario
13#[derive(Debug, Clone, Serialize, Deserialize)]
14pub struct TestScenario {
15    /// Scenario name
16    pub name: String,
17    /// Scenario description
18    pub description: String,
19    /// Test type
20    pub test_type: LoadTestType,
21    /// Load profile
22    pub load_profile: LoadProfile,
23    /// Target configuration
24    pub target: TestTarget,
25    /// Success criteria
26    pub success_criteria: SuccessCriteria,
27    /// Test duration
28    pub duration: Duration,
29}
30
31/// Load test type
32#[derive(Debug, Clone, Serialize, Deserialize)]
33pub enum LoadTestType {
34    /// Database performance test
35    Database {
36        /// Database operations to test
37        operations: Vec<DatabaseOperation>,
38        /// Connection patterns
39        connection_patterns: Vec<ConnectionPattern>,
40    },
41    /// Memory optimization test
42    Memory {
43        /// Memory operations to test
44        operations: Vec<MemoryOperation>,
45        /// Memory patterns
46        memory_patterns: Vec<MemoryPattern>,
47    },
48    /// Caching performance test
49    Caching {
50        /// Cache operations to test
51        operations: Vec<CacheOperation>,
52        /// Cache patterns
53        cache_patterns: Vec<CachePattern>,
54    },
55    /// SIMD crypto acceleration test
56    SIMDCrypto {
57        /// Cryptographic operations to test
58        operations: Vec<CryptoOperation>,
59        /// Crypto patterns
60        crypto_patterns: Vec<CryptoPattern>,
61    },
62    /// Genetic algorithm performance test
63    GeneticAlgorithm {
64        /// Genetic operations to test
65        operations: Vec<GeneticOperation>,
66        /// Genetic patterns
67        genetic_patterns: Vec<GeneticPattern>,
68    },
69    /// End-to-end system test
70    EndToEnd {
71        /// System operations to test
72        operations: Vec<SystemOperation>,
73        /// Workflow patterns
74        workflow_patterns: Vec<WorkflowPattern>,
75    },
76    /// Stress test
77    Stress {
78        /// Stress operations
79        operations: Vec<StressOperation>,
80        /// Stress patterns
81        stress_patterns: Vec<StressPattern>,
82    },
83    /// Scalability test
84    Scalability {
85        /// Scalability operations
86        operations: Vec<ScalabilityOperation>,
87        /// Scaling patterns
88        scaling_patterns: Vec<ScalingPattern>,
89    },
90}
91
92/// Test target configuration
93#[derive(Debug, Clone, Serialize, Deserialize)]
94pub struct TestTarget {
95    /// Target host
96    pub host: String,
97    /// Target port
98    pub port: u16,
99    /// Target protocol
100    pub protocol: String,
101    /// Target path
102    pub path: String,
103    /// Target authentication
104    pub authentication: Option<TestAuthentication>,
105}
106
107/// Test authentication
108#[derive(Debug, Clone, Serialize, Deserialize)]
109pub struct TestAuthentication {
110    /// Authentication type
111    pub auth_type: AuthenticationType,
112    /// Username
113    pub username: String,
114    /// Password
115    pub password: String,
116    /// Additional parameters
117    pub parameters: std::collections::HashMap<String, String>,
118}
119
120/// Authentication type
121#[derive(Debug, Clone, Serialize, Deserialize)]
122pub enum AuthenticationType {
123    /// Basic authentication
124    Basic,
125    /// Bearer token
126    Bearer,
127    /// OAuth2
128    OAuth2,
129    /// Custom authentication
130    Custom { method: String },
131}
132
133/// Success criteria
134#[derive(Debug, Clone, Serialize, Deserialize)]
135pub struct SuccessCriteria {
136    /// Maximum response time
137    pub max_response_time: Duration,
138    /// Minimum throughput
139    pub min_throughput: f64,
140    /// Maximum error rate
141    pub max_error_rate: f64,
142    /// Resource utilization limits
143    pub resource_limits: ResourceLimits,
144}
145
146/// Resource limits
147#[derive(Debug, Clone, Serialize, Deserialize)]
148pub struct ResourceLimits {
149    /// Maximum CPU usage
150    pub max_cpu_usage: f64,
151    /// Maximum memory usage
152    pub max_memory_usage: f64,
153    /// Maximum disk usage
154    pub max_disk_usage: f64,
155    /// Maximum network usage
156    pub max_network_usage: f64,
157}
158
159impl Default for TestScenario {
160    fn default() -> Self {
161        Self {
162            name: "Default Test Scenario".to_string(),
163            description: "Default test scenario for load testing".to_string(),
164            test_type: LoadTestType::Database {
165                operations: Vec::new(),
166                connection_patterns: Vec::new(),
167            },
168            load_profile: LoadProfile::default(),
169            target: TestTarget::default(),
170            success_criteria: SuccessCriteria::default(),
171            duration: Duration::from_secs(60),
172        }
173    }
174}
175
176impl Default for TestTarget {
177    fn default() -> Self {
178        Self {
179            host: "localhost".to_string(),
180            port: 8080,
181            protocol: "http".to_string(),
182            path: "/".to_string(),
183            authentication: None,
184        }
185    }
186}
187
188impl Default for SuccessCriteria {
189    fn default() -> Self {
190        Self {

192            min_throughput: 100.0,
193            max_error_rate: 0.01,
194            resource_limits: ResourceLimits::default(),

197}
