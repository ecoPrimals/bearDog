1//! Health Checks Configuration
2//!
3//! This module defines health check configurations for production deployments.
4
5use serde::{Deserialize, Serialize};
6use std::time::Duration;
7
8/// Deployment health checks configuration
9#[derive(Debug, Clone, Serialize, Deserialize)]
10pub struct DeploymentHealthChecks {
11    /// Enable health checks
12    pub enabled: bool,
13    /// Health check endpoints
14    pub endpoints: Vec<HealthCheckEndpoint>,
15    /// Health check interval
16    pub interval: Duration,
17}
18
19/// Health check endpoint
20#[derive(Debug, Clone, Serialize, Deserialize)]
21pub struct HealthCheckEndpoint {
22    /// Endpoint name
23    pub name: String,
24    /// Endpoint URL
25    pub url: String,
26    /// Timeout
27    pub timeout: Duration,
28}
29
30impl Default for DeploymentHealthChecks {
31    fn default() -> Self {
32        Self {
33            enabled: true,
34            endpoints: Vec::new(),
35            interval: Duration::from_secs(30),
36        }
37    }
38}
39
40impl DeploymentHealthChecks {
41    /// Create production health checks configuration
42    pub fn production() -> Self {
43        Self {
44            enabled: true,
45            endpoints: vec![HealthCheckEndpoint {
46                name: "api_health".to_string(),
47                url: "/health".to_string(),
48                timeout: Duration::from_secs(5),
49            }],
50            interval: Duration::from_secs(10),
51        }
52    }
53
54    /// Create development health checks configuration
55    pub fn development() -> Self {
56        Self {
57            enabled: false,
58            endpoints: Vec::new(),
59            interval: Duration::from_secs(60),
60        }
61    }
62}
