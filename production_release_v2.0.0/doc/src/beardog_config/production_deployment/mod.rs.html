51/// Deployment resource estimate

117    /// Calculate deployment resources
118    pub fn calculate_deployment_resources(&self) -> DeploymentResourceEstimate {
119        let (cpu_cores, memory_gb, storage_gb, network_mbps) = match &self.strategy {
120            DeploymentStrategy::BlueGreen { .. } => {
121                // Blue-green needs double resources
122                (64, 128, 2048, 20000)
123            }
124            DeploymentStrategy::Rolling { .. } => {
125                // Rolling deployment needs extra capacity
126                (48, 96, 1536, 15000)
127            }
128            DeploymentStrategy::Canary { .. } => {
129                // Canary needs additional resources for monitoring
130                (40, 80, 1280, 12000)
131            }
132            DeploymentStrategy::Immediate { .. } => {
133                // Immediate deployment uses base resources
134                (32, 64, 1024, 10000)
135            }
136        };
137
138        DeploymentResourceEstimate {
139            cpu_cores,
140            memory_gb,
141            storage_gb,
142            network_mbps,
143            estimated_cost_per_hour: Self::estimate_deployment_cost(
144                cpu_cores, memory_gb, storage_gb,
145            ),
146        }
147    }
148
149    /// Estimate deployment cost
150    fn estimate_deployment_cost(cpu_cores: u32, memory_gb: u32, storage_gb: u32) -> f64 {
151        // AWS pricing estimates for deployment
152        let cpu_cost_per_hour = 0.048; // c5.large pricing
153        let memory_cost_per_gb_hour = 0.0058;
154        let storage_cost_per_gb_hour = 0.10 / 24.0 / 30.0;
155
156        let cpu_cost = (cpu_cores as f64 / 2.0) * cpu_cost_per_hour; // 2 vCPUs per core
157        let memory_cost = memory_gb as f64 * memory_cost_per_gb_hour;
158        let storage_cost = storage_gb as f64 * storage_cost_per_gb_hour;
159
160        cpu_cost + memory_cost + storage_cost
161    }

163    /// Generate deployment configuration summary
164    pub fn generate_deployment_summary(&self) -> String {
165        let mut summary = String::new();
166
167        summary.push_str("Production Deployment Configuration Summary\n");
168        summary.push_str("==========================================\n\n");
169
170        summary.push_str(&format!(
171            "Deployment Strategy: {}\n",
172            match &self.strategy {
173                DeploymentStrategy::BlueGreen { .. } => "Blue-Green",
174                DeploymentStrategy::Rolling { .. } => "Rolling",
175                DeploymentStrategy::Canary { .. } => "Canary",
176                DeploymentStrategy::Immediate { .. } => "Immediate",
177            }
178        ));
179
180        summary.push_str(&format!("Environment: {}\n", self.environment.name));
181        summary.push_str(&format!(
182            "Validation Checks: {}\n",
183            self.validation.checks.len()
184        ));
185        summary.push_str(&format!(
186            "Health Check Endpoints: {}\n",
187            self.health_checks.endpoints.len()
188        ));
189        summary.push_str(&format!(
190            "Security Policies: {}\n",
191            self.security.policies.len()
192        ));
193        summary.push_str(&format!(
194            "Monitoring Enabled: {}\n",
195            self.monitoring.enabled
196        ));
197        summary.push_str(&format!("Rollback Enabled: {}\n", self.rollback.enabled));
198
199        let resources = self.calculate_deployment_resources();
200        summary.push_str("\nDeployment Resource Requirements:\n");
201        summary.push_str(&format!("- CPU Cores: {}\n", resources.cpu_cores));
202        summary.push_str(&format!("- Memory: {} GB\n", resources.memory_gb));
203        summary.push_str(&format!("- Storage: {} GB\n", resources.storage_gb));
204        summary.push_str(&format!("- Network: {} Mbps\n", resources.network_mbps));
205        summary.push_str(&format!(
206            "- Estimated Cost: ${:.2}/hour\n",
207            resources.estimated_cost_per_hour
208        ));
209
210        summary
211    }
