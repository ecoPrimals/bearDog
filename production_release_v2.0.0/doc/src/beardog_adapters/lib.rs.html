34/// Universal Ecosystem Integration Result
35pub type EcosystemResult<T> = Result<T, EcosystemError>;
36
37/// Universal ecosystem integration errors
38#[derive(Debug, thiserror::Error)]
39pub enum EcosystemError {
40    #[error("Capability not found: {capability}")]
41    CapabilityNotFound { capability: String },
42
43    #[error("Service registration failed: {reason}")]
44    RegistrationFailed { reason: String },
45
46    #[error("Integration error: {message}")]
47    IntegrationError { message: String },
48
49    #[error("Configuration error: {message}")]
50    ConfigurationError { message: String },
51
52    #[error("Network error: {source}")]
53    NetworkError {
54        #[from]
55        source: reqwest::Error,
56    },
57}

59/// Universal AI-First response format following ecosystem standard
60#[derive(Debug, Clone, Serialize, Deserialize)]
61pub struct AIFirstResponse<T> {
62    /// Operation success status (machine-readable)
63    pub success: bool,
64
65    /// Strongly-typed response data
66    pub data: T,
67
68    /// AI-optimized error information
69    pub error: Option<AIFirstError>,
70
71    /// Unique request identifier for tracing and correlation
72    pub request_id: Uuid,
73
74    /// Processing time in milliseconds for performance monitoring
75    pub processing_time_ms: u64,
76
77    /// AI-specific metadata for decision making
78    pub ai_metadata: AIResponseMetadata,
79
80    /// Confidence score for AI decision making (0.0 - 1.0)
81    pub confidence_score: f64,
82
83    /// Suggested next actions for AI agents
84    pub suggested_actions: Vec<SuggestedAction>,
85}
86
87/// AI-optimized error structure with automation hints
88#[derive(Debug, Clone, Serialize, Deserialize)]
89pub struct AIFirstError {
90    /// Machine-readable error code
91    pub code: String,
92    /// Human-readable error message
93    pub message: String,
94    /// Suggested resolution for AI agents
95    pub resolution_hint: Option<String>,
96    /// Whether this error is retryable
97    pub retryable: bool,
98    /// Retry strategy if applicable
99    pub retry_strategy: Option<RetryStrategy>,
100}

102/// AI response metadata for decision making
103#[derive(Debug, Clone, Serialize, Deserialize)]
104pub struct AIResponseMetadata {
105    /// Operation type that was performed
106    pub operation_type: String,
107    /// Resource consumption metrics
108    pub resource_usage: ResourceUsage,
109    /// Performance indicators
110    pub performance_indicators: HashMap<String, f64>,
111    /// Context for next operations
112    pub context: HashMap<String, serde_json::Value>,
113}
114
115/// Suggested action for AI agents
116#[derive(Debug, Clone, Serialize, Deserialize)]
117pub struct SuggestedAction {
118    /// Action type
119    pub action_type: String,
120    /// Action description
121    pub description: String,
122    /// Priority (0-10)
123    pub priority: u8,
124    /// Parameters for the action
125    pub parameters: HashMap<String, serde_json::Value>,
126}
127
128/// Retry strategy for failed operations
129#[derive(Debug, Clone, Serialize, Deserialize)]
130pub struct RetryStrategy {
131    /// Maximum number of retries
132    pub max_retries: u32,
133    /// Base delay in milliseconds
134    pub base_delay_ms: u64,
135    /// Backoff multiplier
136    pub backoff_multiplier: f64,
137    /// Whether to use exponential backoff
138    pub exponential_backoff: bool,
139}
140
141/// Resource usage metrics
142#[derive(Debug, Clone, Serialize, Deserialize)]
143pub struct ResourceUsage {
144    /// CPU usage percentage
145    pub cpu_percent: f64,
146    /// Memory usage in MB
147    pub memory_mb: f64,
148    /// Network I/O in KB
149    pub network_kb: f64,
150    /// Disk I/O in KB
151    pub disk_kb: f64,
152}

154impl<T> Default for AIFirstResponse<T>

161            data: T::default(),
162            error: None,
163            request_id: Uuid::new_v4(),
164            processing_time_ms: 0,
165            ai_metadata: AIResponseMetadata::default(),
166            confidence_score: 1.0,
167            suggested_actions: Vec::new(),
168        }
169    }
170}
