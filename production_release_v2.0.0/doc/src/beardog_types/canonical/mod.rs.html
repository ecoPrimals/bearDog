96/// Registry of all canonical types for validation and tooling
97pub struct CanonicalTypeRegistry;
98
99impl CanonicalTypeRegistry {
100    /// Get list of all canonical status enums
101    #[must_use]
102    pub fn status_types() -> Vec<&'static str> {
103        vec![
104            "HealthStatus",
105            "ComponentStatus",
106            "OperationStatus",
107            "WorkflowStatus",
108            "KeyStatus",
109        ]
110    }
111
112    /// Get list of all canonical configuration structs
113    #[must_use]
114    pub fn config_types() -> Vec<&'static str> {
115        vec![
116            "SessionConfig",
117            "RateLimitConfig",
118            "AuthenticationConfig",
119            "PasswordPolicyConfig",
120            "MfaConfig",
121            "EncryptionConfig",
122            "ProviderConfig",
123        ]
124    }
125
126    /// Validate that a type name is canonical
127    #[must_use]
128    pub fn is_canonical_type(type_name: &str) -> bool {
129        Self::status_types().contains(&type_name)
130            || Self::config_types().contains(&type_name)
131            || matches!(
132                type_name,
133                "SecurityContext"
134                    | "SecurityAuditEvent"
135                    | "PolicyDecision"
136                    | "AuthorizationLevel"
137                    | "SecurityFlags"
138                    | "RiskLevel"
139            )
140    }
141}
