1//! # Canonical Services Types
2//!
3//! **SINGLE SOURCE OF TRUTH** for all service-related types in the `BearDog` ecosystem.
4//! This module provides unified service metadata, request/response types, and service discovery.
5
6use chrono::{DateTime, Utc};
7use serde::{Deserialize, Serialize};
8use std::collections::HashMap;
9
10/// **UNIVERSAL SERVICE METADATA** - Standard service information
11#[derive(Debug, Clone, Serialize, Deserialize)]
12pub struct UniversalServiceMetadata {
13    /// Unique service identifier
14    pub service_id: String,
15    /// Human-readable service name
16    pub service_name: String,
17    /// Service type classification
18    pub service_type: ServiceType,
19    /// Service version
20    pub version: String,
21    /// Service capabilities
22    pub capabilities: Vec<String>,
23    /// Service endpoint information
24    pub endpoints: Vec<ServiceEndpoint>,
25    /// Service health status
26    pub health_status: ServiceHealthStatus,
27    /// Service registration timestamp
28    pub registered_at: DateTime<Utc>,
29    /// Last health check timestamp
30    pub last_health_check: DateTime<Utc>,
31    /// Service metadata
32    pub metadata: HashMap<String, String>,
33}
34
35/// **SERVICE TYPE** - Classification of service types
36#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
37pub enum ServiceType {
38    /// Core `BearDog` services
39    Core,
40    /// External ecosystem services
41    External,
42    /// AI/ML services
43    Intelligence,
44    /// Storage services
45    Storage,
46    /// Security services
47    Security,
48    /// Monitoring services
49    Monitoring,
50    /// Custom service type
51    Custom(String),
52}
53
54/// **SERVICE ENDPOINT** - Service connection information
55#[derive(Debug, Clone, Serialize, Deserialize)]
56pub struct ServiceEndpoint {
57    /// Endpoint URL
58    pub url: String,
59    /// Protocol (HTTP, gRPC, etc.)
60    pub protocol: String,
61    /// Port number
62    pub port: u16,
63    /// Whether endpoint uses TLS
64    pub secure: bool,
65    /// Endpoint metadata
66    pub metadata: HashMap<String, String>,
67}
68
69/// **SERVICE HEALTH STATUS** - Current service health
70#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
71pub enum ServiceHealthStatus {
72    /// Service is healthy and operational
73    Healthy,
74    /// Service is degraded but functional
75    Degraded,
76    /// Service is unhealthy
77    Unhealthy,
78    /// Service status is unknown
79    Unknown,
80}
81
82/// **UNIVERSAL REQUEST** - Standard request format
83#[derive(Debug, Clone, Serialize, Deserialize)]
84pub struct UniversalRequest {
85    /// Request identifier
86    pub request_id: String,
87    /// Target service
88    pub target_service: String,
89    /// Request operation
90    pub operation: String,
91    /// Request payload
92    pub payload: serde_json::Value,
93    /// Request priority
94    pub priority: RequestPriority,
95    /// Request metadata
96    pub metadata: HashMap<String, String>,
97    /// Request timestamp
98    pub timestamp: DateTime<Utc>,
99}
100
101/// **REQUEST PRIORITY** - Request priority levels
102#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
103pub enum RequestPriority {
104    /// Low priority request
105    Low,
106    /// Normal priority request
107    Normal,
108    /// High priority request
109    High,
110    /// Critical priority request
111    Critical,
112}
113
114/// **UNIVERSAL RESPONSE** - Standard response format
115#[derive(Debug, Clone, Serialize, Deserialize)]
116pub struct UniversalResponse {
117    /// Response identifier (matches request)
118    pub request_id: String,
119    /// Response status
120    pub status: ResponseStatus,
121    /// Response data
122    pub data: Option<serde_json::Value>,
123    /// Error information (if any)
124    pub error: Option<String>,
125    /// Response timestamp
126    pub timestamp: DateTime<Utc>,
127    /// Response metadata
128    pub metadata: HashMap<String, String>,
129}
130
131/// **RESPONSE STATUS** - Response status types
132#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
133pub enum ResponseStatus {
134    /// Request completed successfully
135    Success,
136    /// Request failed with error
137    Error,
138    /// Request is being processed
139    Processing,
140    /// Request timed out
141    Timeout,
142}
143
144impl Default for ServiceType {
145    fn default() -> Self {
146        Self::External
147    }
148}
149
150impl Default for ServiceHealthStatus {
151    fn default() -> Self {
152        Self::Unknown
153    }
154}
155
156impl Default for RequestPriority {
157    fn default() -> Self {
158        Self::Normal
159    }
160}
161
162impl Default for ResponseStatus {
163    fn default() -> Self {
164        Self::Success
165    }
166}
