1//! # Canonical Cryptographic Types
2//!
3//! **SINGLE SOURCE OF TRUTH** for all cryptographic types in the `BearDog` ecosystem.
4//! This module consolidates key types, encryption parameters, and crypto operations.
5
6use serde::{Deserialize, Serialize};
7
8/// **UNIVERSAL CANONICAL** Key Type - Agnostic cryptographic key classification
9///
10/// This enum supports all known cryptographic key types across:
11/// - All major `HSM` vendors (Thales, `SafeNet`, Utimaco, etc.)
12/// - Cloud `HSM` services (AWS, Azure, GCP)
13/// - Mobile secure elements (iOS, `Android`)
14/// - Software `HSM` implementations
15/// - Cryptocurrency and blockchain systems
16/// - Government and military cryptographic standards
17#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
18pub enum KeyType {
19    // === UNIVERSAL ALGORITHM FAMILIES ===
20    /// Symmetric encryption keys (AES, `ChaCha20`, etc.)
21    Symmetric {
22        algorithm: String,
23        key_size: u32,
24    },
25
26    /// Asymmetric encryption keys (RSA, ECC, etc.)
27    Asymmetric {
28        algorithm: String,
29        key_size: u32,
30    },
31
32    /// Digital signature keys (RSA-PSS, ECDSA, `EdDSA`, etc.)
33    Signing {
34        algorithm: String,
35        key_size: u32,
36    },
37
38    /// Key derivation and agreement keys (ECDH, X25519, etc.)
39    KeyAgreement {
40        algorithm: String,
41        key_size: u32,
42    },
43
44    /// Authentication keys (HMAC, etc.)
45    Authentication {
46        algorithm: String,
47        key_size: u32,
48    },
49
50    // === SPECIFIC WELL-KNOWN ALGORITHMS (for performance and compatibility) ===
51    /// `AES` symmetric encryption
52    Aes128,
53    Aes192,
54    Aes256,
55
56    /// RSA asymmetric encryption/signing
57    Rsa1024,
58    Rsa2048,
59    Rsa3072,
60    Rsa4096,
61
62    /// Elliptic Curve keys
63    EccP256,
64    EccP384,
65    EccP521,
66
67    /// Edwards curve keys
68    Ed25519,
69    Ed448,
70
71    /// Curve25519 key exchange
72    X25519,
73    X448,
74
75    /// Bitcoin/cryptocurrency curves
76    Secp256k1,
77
78    /// `ChaCha20` stream cipher
79    ChaCha20,
80
81    /// Post-quantum cryptography (future-proofing)
82    PostQuantum {
83        algorithm: String,
84        parameter_set: String,
85    },
86
87    // === LEGACY AND COMPATIBILITY SUPPORT ===
88    /// Generic symmetric (for legacy systems)
89    SymmetricLegacy,
90    /// Generic asymmetric (for legacy systems)  
91    AsymmetricLegacy,
92    /// Generic signing (for legacy systems)
93    SigningLegacy,
94    /// Generic encryption (for legacy systems)
95    EncryptionLegacy,
96    /// Generic authentication (for legacy systems)
97    AuthenticationLegacy,
98    /// Generic derivation (for legacy systems)
99    DerivationLegacy,
100
101    // === VENDOR-SPECIFIC EXTENSIONS ===
102    /// Custom vendor-specific key type
103    Custom {
104        vendor: String,
105        algorithm: String,
106        parameters: std::collections::HashMap<String, String>,
107    },
108}
109
110impl Default for KeyType {
111    fn default() -> Self {
112        Self::Aes256 // Most commonly used secure symmetric key
113    }
114}

116/// **CANONICAL** Encryption Algorithm
117#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
118pub enum EncryptionAlgorithm {
119    AES256GCM,
120    AES128GCM,
121    ChaCha20Poly1305,
122    RSA2048,
123    RSA4096,
124    Ed25519,
125    ECDSA256,
126    ECDSA384,
127}
128
129impl Default for EncryptionAlgorithm {
130    fn default() -> Self {
131        Self::AES256GCM
132    }
133}

135/// **CANONICAL** Key Usage Flags
136#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
137#[allow(clippy::struct_excessive_bools)] // Key usage flags are inherently boolean
138pub struct KeyUsage {
139    pub encrypt: bool,
140    pub decrypt: bool,
141    pub sign: bool,
142    pub verify: bool,
143    pub key_agreement: bool,
144    pub key_derivation: bool,
145    pub wrap_key: bool,
146    pub unwrap_key: bool,
147}
148
149impl Default for KeyUsage {
150    fn default() -> Self {
151        Self {
152            encrypt: true,
153            decrypt: true,
154            sign: false,
155            verify: false,
156            key_agreement: false,
157            key_derivation: false,
158            wrap_key: false,
159            unwrap_key: false,
160        }
161    }
162}

164/// **CANONICAL** Cryptographic Parameters
165#[derive(Debug, Clone, Serialize, Deserialize)]
166pub struct CryptoParams {
167    pub algorithm: EncryptionAlgorithm,
168    pub key_size_bits: u32,
169    pub iv_size_bytes: Option<u32>,
170    pub tag_size_bytes: Option<u32>,
171    pub additional_data: Option<Vec<u8>>,
172}
173
174impl Default for CryptoParams {
175    fn default() -> Self {
176        Self {
177            algorithm: EncryptionAlgorithm::AES256GCM,
178            key_size_bits: 256,
179            iv_size_bytes: Some(12),  // Standard GCM IV size
180            tag_size_bytes: Some(16), // Standard GCM tag size
181            additional_data: None,
182        }
183    }
184}
