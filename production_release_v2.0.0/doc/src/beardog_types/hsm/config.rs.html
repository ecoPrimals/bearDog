25/// **KEY SOURCE ENUMERATION** - Defines how encryption keys are sourced
26#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
27pub enum KeySource {
28    /// Key is derived from other key material
