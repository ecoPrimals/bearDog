1//! # `HSM` Tier Classification and Security Levels
2//!
3//! **CONSOLIDATES FRAGMENTED TIER DEFINITIONS**
4//!
5//! This module eliminates the fragmented `HSM` tier definitions found in:
6//! - `beardog-tunnel/src/universal_hsm/traits.rs` (HsmTier - unified levels)
7//! - `beardog-tunnel/src/tunnel/hsm/types/tier.rs` (HsmTier - complex variants)  
8//! - `beardog-tunnel/src/tunnel/hsm/types/config.rs` (various level enums)

12/// **UNIFIED `HSM` TIER CLASSIFICATION** - Replaces all fragmented tier definitions
13///
14/// This combines the simple FIPS-based tiers with the complex variant-based tiers
15/// to provide a comprehensive `HSM` classification system.
16#[derive(Debug, Clone, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize, Deserialize)]
17pub enum HsmTier {
18    // Basic tier levels (from universal_hsm)
19    /// Software-based `HSM` (lowest security)
20    Software,
21    /// Hardware-backed but not certified
22    Hardware,
23    /// Common Criteria certified
24    CommonCriteria,
25    /// FIPS 140-2 Level 2
26    Fips140Level2,
27    /// FIPS 140-2 Level 3
28    Fips140Level3,
29    /// FIPS 140-2 Level 4 (highest security)
30    Fips140Level4,
31
32    // Enhanced tier levels (from tunnel/hsm/types/tier.rs)
33    /// Basic hardware `HSM`
34    BasicHardware,
35    /// Certified hardware `HSM`  
36    CertifiedHardware,
37    /// High security `HSM`
38    HighSecurity,
39    /// Premium `HSM` with human entropy
40    HumanEntropyPremium,
41
42    // Complex variant tiers with detailed configuration
43    /// Hardware `HSM` with vendor details
44    HardwareHsm {
45        vendor: String,
46        model: String,
47        certification: String,
48        tamper_resistance: TamperResistanceLevel,
49    },
50    /// Software `HSM` with enhanced features
51    SoftwareHsm {
52        implementation: SoftwareHsmType,
53        key_storage: KeyStorageType,
54        encryption_at_rest: bool,
55        memory_protection: MemoryProtectionLevel,
56    },
57    /// Smartphone-based `HSM`
58    SmartphoneHsm {
59        device_type: SmartphoneType,
60        secure_enclave: SecureEnclaveType,
61        attestation_level: AttestationLevel,
62        user_presence_required: bool,
63    },
64    /// Hybrid `HSM` combining multiple approaches
65    HybridHsm {
66        primary_tier: Box<HsmTier>,
67        fallback_tier: Box<HsmTier>,
68        selection_criteria: String,
69    },
70}

307impl HsmTier {
308    /// Get the security level of this `HSM` tier
309    #[must_use]
310    pub fn security_level(&self) -> HsmSecurityTier {
311        match self {
312            Self::Software | Self::SoftwareHsm { .. } => HsmSecurityTier::Software,
313            Self::Hardware | Self::BasicHardware | Self::HardwareHsm { .. } => {
314                HsmSecurityTier::Hardware
315            }
316            Self::CommonCriteria | Self::CertifiedHardware => HsmSecurityTier::CommonCriteria,
317            Self::Fips140Level2 => HsmSecurityTier::Fips140Level2,
318            Self::Fips140Level3 | Self::HighSecurity => HsmSecurityTier::Fips140Level3,
319            Self::Fips140Level4 | Self::HumanEntropyPremium => HsmSecurityTier::Fips140Level4,
320            Self::SmartphoneHsm {
321                attestation_level, ..
322            } => match attestation_level {
323                AttestationLevel::None | AttestationLevel::Basic => HsmSecurityTier::Hardware,
324                AttestationLevel::Enhanced => HsmSecurityTier::CommonCriteria,
325                AttestationLevel::Strong
326                | AttestationLevel::Hardware
327                | AttestationLevel::CertifiedHardware => HsmSecurityTier::Fips140Level2,
328            },
329            Self::HybridHsm { primary_tier, .. } => primary_tier.security_level(),
330        }
331    }
332
333    /// Check if this tier supports hardware operations
334    #[must_use]
335    pub const fn is_hardware_backed(&self) -> bool {
336        !matches!(self, Self::Software | Self::SoftwareHsm { .. })
337    }
338
339    /// Check if this tier requires user presence
340    #[must_use]
341    pub const fn requires_user_presence(&self) -> bool {
342        match self {
343            Self::SmartphoneHsm {
344                user_presence_required,
345                ..
346            } => *user_presence_required,
347            Self::HumanEntropyPremium => true,
348            _ => false,
349        }
350    }
351}
