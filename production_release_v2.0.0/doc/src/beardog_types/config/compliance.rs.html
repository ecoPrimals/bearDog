1//! # Canonical Compliance Configuration Types
2//!
3//! **ELIMINATES COMPLIANCE CONFIG FRAGMENTATION**
4//!
5//! This module consolidates compliance configurations from:
6//! - `beardog-compliance/src/compliance/types.rs` (ComplianceConfig, ReportingConfig)
7//! - Multiple audit and reporting configurations across modules
8//!
9//! ## Unified Compliance Architecture
10//!
11//! 1. **Consolidated Compliance Standards** - Single comprehensive compliance configuration
12//! 2. **Unified Reporting System** - Common reporting patterns across all compliance modules
13//! 3. **Standardized Audit Trail** - Consistent audit logging and retention
14//! 4. **Enterprise Compliance Features** - Advanced compliance monitoring and alerting
15
16use serde::{Deserialize, Serialize};
17use std::collections::HashMap;
18use std::time::Duration;

21/// **CANONICAL COMPLIANCE CONFIGURATION** - Main compliance system configuration
22///
23/// Consolidates compliance configurations from beardog-compliance::ComplianceConfig
24#[derive(Debug, Clone, Serialize, Deserialize)]
25pub struct ComplianceConfig {
26    // Compliance standards
27    pub enabled_standards: Vec<ComplianceStandard>,
28    pub monitoring_interval: Duration,
29    pub audit_retention: Duration,
30    pub dashboard_refresh_interval: Duration,
31
32    // Reporting configuration
33    pub reporting: ComplianceReportingConfig,
34
35    // Advanced compliance features
36    pub automated_remediation: bool,
37    pub compliance_scoring: bool,
38    pub risk_assessment: ComplianceRiskAssessment,
39    pub notification_thresholds: ComplianceNotificationThresholds,
40}
41
