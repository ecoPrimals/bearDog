61/// **COMPREHENSIVE CANONICAL FAILOVER CONFIGURATION**
62///
63/// **ELIMINATES 6 DUPLICATE FAILOVER CONFIGURATIONS**
64///
65/// This unified FailoverConfig consolidates features from:
66/// - `beardog-types/src/config/network.rs::FailoverConfig` (network failover with recovery)
67/// - `beardog-config/src/caching.rs::FailoverConfig` (Redis failover with retries)  
68/// - `beardog-tunnel/src/tunnel/hsm/manager/config.rs::FailoverConfig` (`HSM` manager failover)
69/// - `beardog-tunnel/src/tunnel/hsm/config.rs::FailoverConfig` (`HSM` config failover)
70/// - `beardog-config/src/load_testing/mod.rs::FailoverConfig` (load testing failover)
71/// - Plus the original production failover configuration
72///
73/// ## Comprehensive Feature Set
74/// - **Endpoint Management**: Primary/secondary/tertiary endpoint configuration
75/// - **Health Monitoring**: Intervals, timeouts, failure thresholds
76/// - **Circuit Breaker**: Protection with thresholds and reset timeouts
77/// - **Retry Logic**: Configurable retry attempts with delays
78/// - **Recovery Strategies**: Multiple recovery approaches (immediate, gradual, manual)
79/// - **Load Balancing**: Multiple strategies during failover scenarios
80/// - **Auto-failback**: Automatic recovery with configurable delays
81/// - **Generic Thresholds**: Support for both count-based and percentage-based thresholds
82#[derive(Debug, Clone, Serialize, Deserialize)]
83pub struct FailoverConfig {
84    // ================================================================================
85    // CORE ENABLEMENT & BEHAVIOR
86    // ================================================================================
87    /// Whether failover is enabled (from network, caching, `HSM` implementations)
88    pub enabled: bool,
89
90    /// Generic failover threshold - can be failure count or percentage (from network, load testing)
91    pub threshold: f64,
92
93    /// Initial delay before attempting failover (from network, load testing)
94    pub failover_delay: Duration,
95
96    // ================================================================================
97    // ENDPOINT MANAGEMENT (from production implementation)
98    // ================================================================================
99    /// Primary service endpoint
100    pub primary_endpoint: Option<String>,
101
102    /// Secondary service endpoint  
103    pub secondary_endpoint: Option<String>,
104
105    /// Optional tertiary endpoint for triple redundancy
106    pub tertiary_endpoint: Option<String>,
107
108    // ================================================================================
109    // HEALTH MONITORING & DETECTION
110    // ================================================================================
111    /// Interval between health checks (from production, caching)
112    pub health_check_interval: Duration,
113
114    /// Timeout for individual health checks (from production)
115    pub health_check_timeout: Duration,
116
117    /// Number of consecutive failures before triggering failover (from production)
118    pub consecutive_failure_threshold: u32,
119
120    // ================================================================================
121    // CIRCUIT BREAKER PROTECTION (from production, `HSM` implementations)
122    // ================================================================================
123    /// Enable circuit breaker pattern (from production, `HSM`)
124    pub circuit_breaker_enabled: bool,
125
126    /// Threshold for activating circuit breaker (from `HSM`)
127    pub circuit_breaker_threshold: u32,
128
129    /// Timeout before attempting to reset circuit breaker (from `HSM`)
130    pub circuit_breaker_timeout: Duration,
131
132    // ================================================================================
133    // RETRY LOGIC (from caching, `HSM` implementations)
134    // ================================================================================
135    /// Maximum number of retry attempts (from caching, `HSM`)
136    pub max_retry_attempts: u32,
137
138    /// Delay between retry attempts (from caching, `HSM`)
139    pub retry_delay: Duration,
140
141    /// Exponential backoff multiplier for retry delays
142    pub retry_backoff_multiplier: f64,
143
144    // ================================================================================
145    // RECOVERY & FAILBACK (from production, network, load testing)
146    // ================================================================================
147    /// Enable automatic failback to primary when it recovers (from production)
148    pub auto_failback_enabled: bool,
149
150    /// Delay before attempting failback (from production)
151    pub failback_delay: Duration,
152
153    /// Recovery strategy for failed services (from network, load testing)
154    pub recovery_strategy: FailoverRecoveryStrategy,
155
156    /// Timeout for recovery operations (from network, load testing)
157    pub recovery_timeout: Duration,
158
159    // ================================================================================
160    // LOAD BALANCING DURING FAILOVER (from production)
161    // ================================================================================
162    /// Load balancing strategy during failover scenarios
163    pub load_balancing_strategy: FailoverLoadBalancingStrategy,
164
165    // ================================================================================
166    // MONITORING & OBSERVABILITY
167    // ================================================================================
168    /// Enable detailed failover event logging
169    pub enable_failover_logging: bool,
170
171    /// Enable failover metrics collection
172    pub enable_failover_metrics: bool,
173
174    /// Custom tags for failover monitoring
175    pub monitoring_tags: HashMap<String, String>,
176}

178/// **RECOVERY STRATEGIES** - Consolidates RecoveryStrategy from load testing
179///
180/// Incorporates recovery strategies from:
181/// - `beardog-config/src/load_testing/mod.rs::RecoveryStrategy`
182/// - `beardog-types/src/config/network.rs::RecoveryConfig`
183#[derive(Debug, Clone, Serialize, Deserialize)]
184pub enum FailoverRecoveryStrategy {
185    /// Immediate recovery attempt when service becomes available
186    Immediate,
187
188    /// Gradual recovery with increasing load
189    Gradual {
190        /// Step percentage for gradual recovery
191        step_percentage: f64,
192        /// Interval between recovery steps
193        step_interval: Duration,
194    },
195
196    /// Manual recovery requiring administrator intervention
197    Manual,
198
199    /// Health-based recovery triggered by health check success
200    HealthBased {
201        /// Number of consecutive successful health checks required
202        success_threshold: u32,
203    },
204}
