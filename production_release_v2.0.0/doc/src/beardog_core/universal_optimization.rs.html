1//! # Universal Optimization Service
2//!
3//! This module provides a unified interface for requesting optimization from any
4//! available ecosystem modules, completely eliminating hardcoded primal references.
5//! It uses capability-based discovery to find the best available optimization
6//! modules and automatically falls back to local implementations when needed.

23/// Universal optimization request types
24#[derive(Debug, Clone, Serialize, Deserialize)]
25pub enum OptimizationRequest {
26    /// Genetic algorithm optimization
27    GeneticAlgorithm {
28        input_data: Vec<u8>,
29        optimization_target: GeneticTarget,
30        quality_requirements: GeneticQualityRequirements,
31    },
32
33    /// Performance acceleration
34    PerformanceAcceleration {
35        workload_type: WorkloadType,
36        current_metrics: PerformanceMetrics,
37        target_improvement: f64, // 0.0 to 1.0
38    },
39
40    /// Cryptographic optimization
41    CryptographicOptimization {
42        algorithm_type: CryptoAlgorithmType,
43        security_requirements: SecurityRequirements,
44        performance_constraints: CryptoPerformanceConstraints,
45    },
46
47    /// Gaming-specific optimization
48    GamingOptimization {
49        game_type: GameType,
50        latency_requirements: LatencyRequirements,
51        throughput_requirements: ThroughputRequirements,
52    },
53
54    /// Machine learning optimization
55    MLOptimization {
56        model_type: MLModelType,
57        training_data_size: u64,
58        accuracy_target: f64,
59    },
60}

237/// Universal optimization service trait
238#[async_trait]
239pub trait UniversalOptimizationService: Send + Sync {
240    /// Request optimization from available ecosystem modules
241    async fn request_optimization(
242        &self,
243        request: OptimizationRequest,
244    ) -> BearDogResult<OptimizationResponse>;
245
246    /// Get available optimization capabilities in the ecosystem
247    async fn get_available_capabilities(&self) -> BearDogResult<Vec<CapabilityType>>;
248
249    /// Check health of optimization modules
250    async fn check_optimization_health(&self) -> BearDogResult<HashMap<String, String>>;
251}

278    pub fn new(
279        discovery_service: Arc<dyn UniversalCapabilityDiscovery>,
280        fallback_optimizer: Arc<dyn LocalOptimizer>,
281    ) -> Self {
282        Self {
283            discovery_service,
284            fallback_optimizer,

288    /// Determine required capabilities for optimization request
289    fn determine_required_capabilities(
290        &self,
291        request: &OptimizationRequest,
292    ) -> Vec<CapabilityType> {
293        match request {
294            OptimizationRequest::GeneticAlgorithm { .. } => vec![
295                CapabilityType::GeneticAlgorithms,
296                CapabilityType::ComputeOptimization,
297            ],
298            OptimizationRequest::PerformanceAcceleration { .. } => vec![
299                CapabilityType::PerformanceAcceleration,
300                CapabilityType::ParallelProcessing,
301            ],
302            OptimizationRequest::CryptographicOptimization { .. } => vec![
303                CapabilityType::ComputeOptimization,
304                CapabilityType::PerformanceAcceleration,
305            ],
306            OptimizationRequest::GamingOptimization { .. } => vec![
307                CapabilityType::Gaming {
308                    low_latency: true,
309                    simd_acceleration: true,
310                },
311                CapabilityType::PerformanceAcceleration,
312            ],
313            OptimizationRequest::MLOptimization { .. } => vec![
314                CapabilityType::MachineLearning,
315                CapabilityType::ComputeOptimization,
316            ],
317        }
318    }
319
320    /// Create quality requirements based on optimization request
321    fn create_quality_requirements(&self, request: &OptimizationRequest) -> QualityRequirements {
322        match request {
323            OptimizationRequest::GeneticAlgorithm {
324                quality_requirements,
325                ..
326            } => QualityRequirements {
327                minimum_accuracy: quality_requirements.minimum_fitness,
328                minimum_reliability: 0.95,
329                minimum_availability: 0.99,
330                fault_tolerance_required: false,
331            },
332            OptimizationRequest::GamingOptimization { .. } => QualityRequirements {
333                minimum_accuracy: 0.99,
334                minimum_reliability: 0.999,
335                minimum_availability: 0.9999,
336                fault_tolerance_required: true,
337            },
338            _ => QualityRequirements::default(),
339        }
340    }
341
342    /// Create performance requirements based on optimization request
343    fn create_performance_requirements(
344        &self,
345        request: &OptimizationRequest,
346    ) -> PerformanceRequirements {
347        match request {
348            OptimizationRequest::GamingOptimization {
349                latency_requirements,
350                throughput_requirements,
351                ..
352            } => PerformanceRequirements {
353                max_response_time_ms: latency_requirements.max_crypto_latency_ms,
354                minimum_throughput: throughput_requirements.min_operations_per_second,
355                resource_constraints: crate::universal_discovery::ResourceConstraints {
356                    max_cpu_usage: Some(60.0),
357                    max_memory_mb: Some(2048),
358                    max_network_bandwidth: Some(throughput_requirements.min_network_bandwidth_mbps),
359                    priority: OperationPriority::RealTime,
360                },
361            },
362            OptimizationRequest::PerformanceAcceleration { .. } => {
363                PerformanceRequirements::high_performance()
364            }
365            _ => PerformanceRequirements::default(),
366        }
367    }
368}

370#[async_trait]
371impl UniversalOptimizationService for EcosystemOptimizationService {
372    async fn request_optimization(
373        &self,
374        request: OptimizationRequest,
375    ) -> BearDogResult<OptimizationResponse> {
376        let start_time = Instant::now();
377
378        info!("🔧 Requesting universal optimization: {:?}", request);
379
380        // Determine required capabilities
381        let required_capabilities = self.determine_required_capabilities(&request);
382        let quality_requirements = self.create_quality_requirements(&request);
383        let performance_requirements = self.create_performance_requirements(&request);
384
385        // Try to discover modules with required capabilities
386        match self
387            .discovery_service
388            .discover_by_capabilities(
389                &required_capabilities,
390                Some(quality_requirements),
391                Some(performance_requirements),
392            )
393            .await
394        {
395            Ok(modules) if !modules.is_empty() => {
396                info!("✅ Found {} optimization modules", modules.len());
397
398                // Create universal module request
399                let module_request = UniversalModuleRequest {
400                    request_id: Uuid::new_v4(),
401                    source_module: "beardog-optimization".to_string(),
402                    target_capability: required_capabilities[0].clone(),
403                    operation: "optimize".to_string(),
404                    payload: serde_json::to_value(&request)?,
405                    quality_requirements: self.create_quality_requirements(&request),
406                    performance_requirements: self.create_performance_requirements(&request),
407                    metadata: HashMap::new(),
408                    timestamp: Utc::now(),
409                };
410
411                // Send request to best module
412                match self
413                    .discovery_service
414                    .request_by_capability(module_request)
415                    .await
416                {
417                    Ok(response) => {
418                        let processing_time = start_time.elapsed();
419                        info!("🎯 Optimization completed in {:?}", processing_time);
420
421                        // Parse response into OptimizationResponse
422                        Ok(OptimizationResponse {
423                            request_id: response.request_id,
424                            optimization_type: format!("{request:?}"),
425                            success: response.success,
426                            improvement_factor: response.actual_performance.resource_efficiency,
427                            optimized_parameters: HashMap::new(),
428                            performance_metrics: PerformanceMetrics {
429                                latency_ms: response.actual_performance.average_response_time_ms,
430                                throughput_ops_per_sec: response
431                                    .actual_performance
432                                    .throughput_ops_per_second,
433                                cpu_utilization: 0.0, // Would be provided by module
434                                memory_utilization: 0.0,
435                                network_utilization: 0.0,
436                                error_rate: 1.0 - response.actual_quality.accuracy,
437                            },
438                            recommendations: vec![], // Would be parsed from response
439                            estimated_duration: processing_time,
440                            confidence_score: response.actual_quality.accuracy,
441                        })
442                    }
443                    Err(e) => {
444                        warn!("Module optimization failed: {}, falling back to local", e);
445                        self.fallback_to_local_optimization(request).await
446                    }
447                }
448            }
449            _ => {
450                info!("No optimization modules found, using local implementation");
451                self.fallback_to_local_optimization(request).await
452            }
453        }
454    }
455
456    async fn get_available_capabilities(&self) -> BearDogResult<Vec<CapabilityType>> {
457        let all_optimization_capabilities = vec![
458            CapabilityType::GeneticAlgorithms,
459            CapabilityType::PerformanceAcceleration,
460            CapabilityType::ComputeOptimization,
461            CapabilityType::Gaming {
462                low_latency: true,
463                simd_acceleration: true,
464            },
465            CapabilityType::MachineLearning,
466        ];
467
468        let mut available_capabilities = Vec::new();
469
470        for capability in all_optimization_capabilities {
471            match self
472                .discovery_service
473                .check_capability_health(capability.clone())
474                .await
475            {
476                Ok(health_statuses) if !health_statuses.is_empty() => {
477                    available_capabilities.push(capability);
478                }
479                _ => continue,
480            }
481        }
482
483        Ok(available_capabilities)
484    }
485
486    async fn check_optimization_health(&self) -> BearDogResult<HashMap<String, String>> {
487        let mut health_map = HashMap::new();
488
489        let capabilities_to_check = vec![
490            CapabilityType::GeneticAlgorithms,
491            CapabilityType::PerformanceAcceleration,
492            CapabilityType::ComputeOptimization,
493        ];
494
495        for capability in capabilities_to_check {
496            match self
497                .discovery_service
498                .check_capability_health(capability.clone())
499                .await
500            {
501                Ok(health_statuses) => {
502                    for (module_id, health_status) in health_statuses {
503                        health_map.insert(
504                            format!("{}:{}", capability.as_capability_string(), module_id),
505                            format!("{health_status:?}"),
506                        );
507                    }
508                }
509                Err(e) => {
510                    health_map.insert(
511                        capability.as_capability_string(),
512                        format!("Check failed: {e}"),
513                    );
514                }
515            }
516        }
517
518        Ok(health_map)
519    }
520}

522impl EcosystemOptimizationService {
523    /// Fallback to local optimization when no modules are available
524    async fn fallback_to_local_optimization(
525        &self,
526        request: OptimizationRequest,
527    ) -> BearDogResult<OptimizationResponse> {
528        info!("🔄 Falling back to local optimization");
529
530        match request {
531            OptimizationRequest::GeneticAlgorithm { .. } => {
532                self.fallback_optimizer
533                    .local_genetic_optimization(request)
534                    .await
535            }
536            OptimizationRequest::PerformanceAcceleration { .. } => {
537                self.fallback_optimizer
538                    .local_performance_optimization(request)
539                    .await
540            }
541            OptimizationRequest::CryptographicOptimization { .. } => {
542                self.fallback_optimizer
543                    .local_crypto_optimization(request)
544                    .await
545            }
546            OptimizationRequest::GamingOptimization { .. } => {
547                self.fallback_optimizer
548                    .local_performance_optimization(request)
549                    .await
550            }
551            OptimizationRequest::MLOptimization { .. } => {
552                self.fallback_optimizer
553                    .local_performance_optimization(request)
554                    .await
555            }
556        }
557    }
