151/// Universal Primal Provider trait for ecosystem integration
152#[async_trait]
153pub trait UniversalPrimalProvider: Send + Sync {
154    /// Get primal metadata for ecosystem registration
155    fn metadata(&self) -> &PrimalMetadata;
156
157    /// Get list of capabilities this primal provides
158    fn capabilities(&self) -> &[PrimalCapability];
159
160    /// Get list of services this primal exposes
161    async fn services(&self) -> BearDogResult<Vec<PrimalService>>;
162
163    /// Register with the ecosystem through Songbird
164    async fn register_with_ecosystem(&self, songbird_endpoint: &str) -> BearDogResult<()>;
165
166    /// Handle incoming service request from ecosystem
167    async fn handle_service_request(
168        &self,
169        service_id: &str,
170        request_data: Vec<u8>,
171        context: ServiceContext,
172    ) -> BearDogResult<Vec<u8>>;
173
174    /// Health check for ecosystem monitoring
175    async fn health_check(&self) -> BearDogResult<ServiceHealth>;
176
177    /// Shutdown notification from ecosystem
178    async fn shutdown(&self) -> BearDogResult<()>;
179}
