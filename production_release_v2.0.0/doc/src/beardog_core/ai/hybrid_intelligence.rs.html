288    /// Route large-scale AI analysis to Squirrel via universal adapter
289    pub async fn analyze_with_squirrel_ai(
290        &self,
291        ai_request: ExternalAIRequest,
292    ) -> BearDogResult<ExternalAIResult> {
293        info!("🐿️ Routing AI analysis to Squirrel via universal adapter");
294
295        // `BearDog` doesn't know it's going to Squirrel - just requests AI capability
296        let capability_request = CapabilityRequest::AIIntelligence(ai_request.clone());
297
298        let response = self
299            .universal_adapter
300            .lock()
301            .await
302            .route_capability_request("intelligence".to_string(), capability_request)
303            .await?;
304
305        if !response.success {
306            return Err(BearDogError::ExternalServiceError {
307                service: "ai_intelligence".to_string(),
308                message: "AI capability request failed".to_string(),
309            });
310        }
311
312        // Parse response from AI capability provider (Squirrel)
313        let ai_result: ExternalAIResult = serde_json::from_value(response.data).map_err(|e| {
314            BearDogError::DeserializationError {
315                message: e.to_string(),
316            }
317        })?;
318
319        debug!("✅ Received AI analysis from capability provider");
320        Ok(ai_result)
321    }

399    /// Execute sequential workflow: `BearDog` first, then Squirrel
400    async fn execute_sequential_workflow(
401        &self,
402        workflow: &HybridAIWorkflow,
403        input_data: &WorkflowInputData,
404    ) -> BearDogResult<HybridWorkflowResult> {
405        // Step 1: `BearDog` internal ML processing
406        let mut internal_results = Vec::new();
407        for step in &workflow.steps {
408            if matches!(step.step_type, AIStepType::InternalML(_)) {
409                let result = self.execute_internal_step(step, input_data).await?;
410                internal_results.push(result);
411            }
412        }
413
414        // Step 2: Route to Squirrel via universal adapter with `BearDog` context
415        let mut external_results = Vec::new();
416        for step in &workflow.steps {
417            if matches!(step.step_type, AIStepType::ExternalAI(_)) {
418                let internal_values: Vec<serde_json::Value> = internal_results
419                    .iter()
420                    .map(|result| result.result_data.clone())
421                    .collect();
422                let enhanced_input =
423                    self.enhance_input_with_beardog_context(input_data, &internal_values);
424                let result = self.execute_external_step(step, &enhanced_input).await?;
425                external_results.push(result);
426            }
427        }
428
429        // Step 3: Combine results
430        let mut all_results = Vec::new();
431        all_results.extend(internal_results);
432        all_results.extend(external_results);
433
434        Ok(HybridWorkflowResult {
435            workflow_id: workflow.workflow_id.clone(),
436            results: all_results,
437            overall_confidence: 0.85,
438            processing_time_ms: 100,
439        })
440    }
