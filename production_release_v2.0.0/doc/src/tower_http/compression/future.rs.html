18pin_project! {
19    /// Response future of [`Compression`].
20    ///
21    /// [`Compression`]: super::Compression
22    #[derive(Debug)]
23    pub struct ResponseFuture<F, P> {
24        #[pin]
25        pub(crate) inner: F,
26        pub(crate) encoding: Encoding,
27        pub(crate) predicate: P,
28        pub(crate) quality: CompressionLevel,
29    }
30}
31
32impl<F, B, E, P> Future for ResponseFuture<F, P>
33where
34    F: Future<Output = Result<Response<B>, E>>,
35    B: Body,
36    P: Predicate,
37{
38    type Output = Result<Response<CompressionBody<B>>, E>;
39
40    #[allow(unreachable_code, unused_mut, unused_variables)]
41    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
42        let res = ready!(self.as_mut().project().inner.poll(cx)?);
43
44        // never recompress responses that are already compressed
45        let should_compress = !res.headers().contains_key(header::CONTENT_ENCODING)
46            && self.predicate.should_compress(&res);
47
48        let (mut parts, body) = res.into_parts();
49
50        let body = match (should_compress, self.encoding) {
51            // if compression is _not_ support or the client doesn't accept it
52            (false, _) | (_, Encoding::Identity) => {
53                return Poll::Ready(Ok(Response::from_parts(
54                    parts,
55                    CompressionBody::new(BodyInner::identity(body)),
56                )))
57            }
58
59            #[cfg(feature = "compression-gzip")]
60            (_, Encoding::Gzip) => {
61                CompressionBody::new(BodyInner::gzip(WrapBody::new(body, self.quality)))
62            }
63            #[cfg(feature = "compression-deflate")]
64            (_, Encoding::Deflate) => {
65                CompressionBody::new(BodyInner::deflate(WrapBody::new(body, self.quality)))
66            }
67            #[cfg(feature = "compression-br")]
68            (_, Encoding::Brotli) => {
69                CompressionBody::new(BodyInner::brotli(WrapBody::new(body, self.quality)))
70            }
71            #[cfg(feature = "compression-zstd")]
72            (_, Encoding::Zstd) => {
73                CompressionBody::new(BodyInner::zstd(WrapBody::new(body, self.quality)))
74            }
75            #[cfg(feature = "fs")]
76            (true, _) => {
77                // This should never happen because the `AcceptEncoding` struct which is used to determine
78                // `self.encoding` will only enable the different compression algorithms if the
79                // corresponding crate feature has been enabled. This means
80                // Encoding::[Gzip|Brotli|Deflate] should be impossible at this point without the
81                // features enabled.
82                //
83                // The match arm is still required though because the `fs` feature uses the
84                // Encoding struct independently and requires no compression logic to be enabled.
85                // This means a combination of an individual compression feature and `fs` will fail
86                // to compile without this branch even though it will never be reached.
87                //
88                // To safeguard against refactors that changes this relationship or other bugs the
89                // server will return an uncompressed response instead of panicking since that could
90                // become a ddos attack vector.
91                return Poll::Ready(Ok(Response::from_parts(
92                    parts,
93                    CompressionBody::new(BodyInner::identity(body)),
94                )));
95            }
96        };
97
98        parts.headers.remove(header::CONTENT_LENGTH);
99
100        parts
101            .headers
102            .insert(header::CONTENT_ENCODING, self.encoding.into_header_value());
103
104        let res = Response::from_parts(parts, body);
105        Poll::Ready(Ok(res))
106    }
107}
